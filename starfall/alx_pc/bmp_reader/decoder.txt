function BMP.bmpDecodeStep()
    local File = BMP.file
    
    if not BMP.pixelStart then

        File:seek(BMP.pixelOffset)

        BMP.pixelStart = true

        BMP.x = 0
        BMP.y = 0
        
        local MaxSize = 1024
        if BMP.SizeLimit>0 then
            MaxSize = math.clamp(BMP.SizeLimit,1,MaxSize)    
        end
        
        local WH = math.max(BMP.width,BMP.height)
        
        if MaxSize==1024 and WH>1024  then
            local Scale = WH / MaxSize
            local ResizedWidth  = math.floor(BMP.width / Scale)
            local ResizedHeight = math.floor(BMP.height / Scale)
                            
            print("Your image is larger than Starfall Screen supports\n"..
                "Your image will be resized to: "..ResizedWidth.."x"..ResizedHeight)
        end
        
        local scale = math.min(
            MaxSize / BMP.width,
            MaxSize / BMP.height,
            1
        )
    
        BMP.scale = scale
        BMP.lastX = -1
        BMP.lastY = -1
        
        BMP.RenderW = math.floor(BMP.width * BMP.scale)
        BMP.RenderH = math.floor(BMP.height * BMP.scale) 
        
        local sequential = true     
        -- RLE compression delta might shift
        if BMP.compression == 1 or BMP.compression == 2 then
            sequential = false 
        end
        
        BMP.Renderer.Init(BMP.RenderW, BMP.RenderH, BMP.SizeLimit, BMP.NoKeepAR, BMP.UrlMaterial, sequential, BMP.topDown)

        if BMP.compression==1 or BMP.compression==2 then
            BMP.rleState = {
                count = 0,
                mode = 0,
                value = 0,
                absoluteRemaining = 0,
                absoluteBytesRead = 0,
                rle4Odd = false
            }    
        end
    end

    return BMP.readPixel()
end

function BMP.OutputTransformPixel(x, y)
    local sx = math.floor(x * BMP.RenderW / BMP.width)
    local sy = math.floor(y * BMP.RenderH / BMP.height)

    if BMP.lastFlushedY == sy then
        return nil, sy
    end

    if sx == BMP.lastX and sy == BMP.lastY then
        return nil, sy
    end

    BMP.lastX = sx
    BMP.lastY = sy

    return sx, sy
end

function BMP.readPixelRLE8()
    local st = BMP.rleState
    local File = BMP.file

    if st.count == 0 and st.absoluteRemaining == 0 then
        local count = File:readByte()
        local command = File:readByte()

        if count == 0 then
            if command == 0 then
                --BMP.x = 0
                --BMP.y = BMP.y + 1
                --BMP.Renderer.Flush()
                return
            elseif command == 1 then
                BMP.Renderer.Finish()
                BMP.finished = true
                return 1
            elseif command == 2 then 
                local dx = File:readByte()
                local dy = File:readByte()
                BMP.x = BMP.x + dx
                BMP.y = BMP.y + dy
                return
            else
                st.absoluteRemaining = command
                st.mode = 1
                st.absoluteBytesRead = 0
            end
        else 
            st.count = count
            st.value = command
            st.mode = 0
        end
    end

    if st.mode == 0 and st.count > 0 then
        st.count = st.count - 1
        local color = BMP.palette[st.value]
        if color then
            return BMP.processPixel(color.r, color.g, color.b)
        end
    elseif st.mode == 1 and st.absoluteRemaining > 0 then
        st.absoluteRemaining = st.absoluteRemaining - 1
        local index = File:readByte()
        st.absoluteBytesRead = st.absoluteBytesRead + 1

        if st.absoluteRemaining == 0 and bit.band(st.absoluteBytesRead, 1) != 0 then
            File:skip(1)
        end

        local color = BMP.palette[index]
        if color then
            return BMP.processPixel(color.r, color.g, color.b)
        end
    end
end

function BMP.readPixelRLE4()
    local st = BMP.rleState
    local File = BMP.file

    if st.count == 0 and st.absoluteRemaining == 0 then
        local count = File:readByte()
        local command = File:readByte()

        if count == 0 then
            if command == 0 then
                --BMP.x = 0
                --BMP.y = BMP.y + 1
                --BMP.Renderer.Flush()
                st.rle4Odd = false
                return
            elseif command == 1 then 
                BMP.Renderer.Finish()
                BMP.finished = true
                return 1
            elseif command == 2 then 
                local dx = File:readByte()
                local dy = File:readByte()
                BMP.x = BMP.x + dx
                BMP.y = BMP.y + dy
                
                while BMP.x >= BMP.width and BMP.width > 0 do
                    BMP.x = BMP.x - BMP.width
                    BMP.y = BMP.y + 1
                    BMP.Renderer.Flush()
                end
                
                st.rle4Odd = false 
                return
            else
                st.absoluteRemaining = command
                st.mode = 1
                st.absoluteBytesRead = 0
                st.rle4Odd = false
            end
        else
            st.count = count
            st.value = command 
            st.mode = 0
            st.rle4Odd = false
        end
    end

    if st.mode == 0 and st.count > 0 then
        st.count = st.count - 1
        local index
        if not st.rle4Odd then
            index = bit.band(bit.rshift(st.value, 4), 0x0F)
        else
            index = bit.band(st.value, 0x0F)
        end
        st.rle4Odd = not st.rle4Odd

        local color = BMP.palette[index]
        if color then
            return BMP.processPixel(color.r, color.g, color.b)
        end
    elseif st.mode == 1 and st.absoluteRemaining > 0 then
        st.absoluteRemaining = st.absoluteRemaining - 1
        
        local index
        if not st.rle4Odd then
            st.value = File:readByte()
            st.absoluteBytesRead = st.absoluteBytesRead + 1
            index = bit.band(bit.rshift(st.value, 4), 0x0F)
            st.rle4Odd = true
        else
            index = bit.band(st.value, 0x0F)
            st.rle4Odd = false
        end

        if st.absoluteRemaining == 0 and bit.band(st.absoluteBytesRead, 1) != 0 then
            File:skip(1)
        end

        local color = BMP.palette[index]
        if color then
            return BMP.processPixel(color.r, color.g, color.b)
        end
    end
end

function BMP.processPixel(r,g,b,a)
    
    local DrawY
    
    if BMP.topDown then
        DrawY = BMP.y
    else
        DrawY = BMP.height - 1 - BMP.y
    end
    
    local sx, sy = BMP.OutputTransformPixel(BMP.x, DrawY)

    if sx!=nil then        
        BMP.Renderer.setPixel(
            sx,
            sy,
            r,
            g,
            b,
            a
        )
    end

    BMP.x = BMP.x + 1

    if BMP.x >= BMP.width then

        BMP.x = 0
        BMP.y = BMP.y + 1
        
        BMP.lastX = -1

        if BMP.compression != 1 and BMP.compression != 2 then
            --  padding
            local padding = BMP.rowSize - BMP.rowBytes
    
            if padding > 0 then
                BMP.file:skip(padding)
            end
        end
        
        if BMP.lastFlushedY != sy then
            BMP.Renderer.Flush()
            BMP.lastFlushedY = sy
        end
    end

    if BMP.y >= BMP.height then
        BMP.Renderer.Finish()
        BMP.bmpFinish()
        BMP.finished = true
        return 1
    end
end

function BMP.readPackedPixels(Bits)
    local File = BMP.file

    local Byte = File:readByte()

    local PixelsPerByte = 8 / Bits
    local Mask = bit.lshift(1, Bits) - 1

    for i = 0, PixelsPerByte - 1 do

        if BMP.x >= BMP.width then
            break
        end

        local Shift = 8 - Bits * (i + 1)

        local Index = bit.band(
            bit.rshift(Byte, Shift),
            Mask
        )

        local Color = BMP.palette[Index]
        if BMP.processPixel(Color.r, Color.g, Color.b) == 1 then
            return 1
        end
    end
end

function BMP.unpackField(pixel, field)

    if not field then
        return 255
    end

    local v = bit.band(pixel, field.mask)
    v = bit.rshift(v, field.shift)

    return math.floor(v * 255 / field.max + 0.5)
end

function BMP.readPixel1()
    return BMP.readPackedPixels(1)
end

function BMP.readPixel2()
    return BMP.readPackedPixels(2)
end

function BMP.readPixel4()
    return BMP.readPackedPixels(4)
end

function BMP.readPixel8()
    local index = BMP.file:readByte()

    local color = BMP.palette[index]

    if not color then
        return nil
    end

    return BMP.processPixel(color.r, color.g, color.b)
end

function BMP.readPixel16()

    local pixel = BMP.file:readUShort()

    if BMP.compression == 0 then
        --  BI_RGB = RGB555

        local r = bit.band(bit.rshift(pixel,10),0x1F)
        local g = bit.band(bit.rshift(pixel,5),0x1F)
        local b = bit.band(pixel,0x1F)

        r = math.floor(r * 255 / 31 + 0.5)
        g = math.floor(g * 255 / 31 + 0.5)
        b = math.floor(b * 255 / 31 + 0.5)

        return BMP.processPixel(r,g,b)

    elseif BMP.compression == 3 then

        local r = BMP.unpackField(pixel, BMP.rField)
        local g = BMP.unpackField(pixel, BMP.gField)
        local b = BMP.unpackField(pixel, BMP.bField)
        local a = BMP.unpackField(pixel, BMP.aField)

        return BMP.processPixel(r,g,b,a)

    else
        print("Unsupported 16-bit compression")
    end
end

function BMP.readPixel24()
    local file = BMP.file

    local b = file:readByte()
    local g = file:readByte()
    local r = file:readByte()

    return BMP.processPixel(r, g, b)
end

function BMP.readPixel32()
    local file = BMP.file

    local b = file:readByte()
    local g = file:readByte()
    local r = file:readByte()
    file:readByte()

    return BMP.processPixel(r, g, b)
end

function BMP.readPixel32Alpha()
    local file = BMP.file
    
    local b = file:readByte()
    local g = file:readByte()
    local r = file:readByte()
    local a = file:readByte()

    local AlphaFactor = a / 255
    
    r = math.floor(r * AlphaFactor)
    g = math.floor(g * AlphaFactor)
    b = math.floor(b * AlphaFactor)

    return BMP.processPixel(r, g, b)
end

function BMP.readPixel32Inversed()
    local file = BMP.file

    file:readByte()
    local b = file:readByte()
    local g = file:readByte()
    local r = file:readByte()

    return BMP.processPixel(r, g, b)
end
