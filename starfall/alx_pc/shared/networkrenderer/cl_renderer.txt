local Renderer = {}

local SafeNet = require("SafeNet")

local function toBool(val)
    if type(val) == "boolean" then
        return val
    elseif type(val) == "number" then
        return val > 0
    end
    return false
end

function Renderer.Init(w,h,SizeLimit,NoKeepAR,isMaterialRender,sequential,topDown,alpha)
    Renderer.queue = {}
    Renderer.netBytesAccumulated = 0
    
    if SafeNet.cancelAll!=nil then
        SafeNet.cancelAll()
    end

    SafeNet.start("image_render_init")
    SafeNet.writeUInt(w,16)
    SafeNet.writeUInt(h,16)
    SafeNet.writeUInt(SizeLimit,16)
    SafeNet.writeBool(toBool(NoKeepAR))
    SafeNet.writeBool(toBool(isMaterialRender))
    SafeNet.writeBool(toBool(sequential))
    SafeNet.writeBool(toBool(topDown))
    SafeNet.writeBool(toBool(alpha))
    SafeNet.send()
    Renderer.sequential = sequential
    Renderer.hasAlpha = toBool(alpha)
end

function Renderer.Flush()
    Renderer.FlushBuffer()
end

function Renderer.FlushBuffer()
    local Queue = Renderer.queue
    if #Queue == 0 then return end
    
    local IsSafeStream = SafeNet.stringstream!=nil
    
    local Stream 
    if IsSafeStream then
        Stream = SafeNet.stringstream()    
    else
        Stream = bit.stringstream() 
    end
    
    Stream:writeUInt16(#Queue)
    
    for i=1,#Queue do
        local p = Queue[i]
        
        if not Renderer.sequential then
            Stream:writeUInt16(p.x)
            Stream:writeUInt16(p.y)
        end
            
        Stream:writeUInt8(p.r)
        Stream:writeUInt8(p.g)
        Stream:writeUInt8(p.b)
        if Renderer.hasAlpha then
            Stream:writeUInt8(p.a)
        end
    end
    
    local RawData = Stream:getString()
    local Data = bit.compress(RawData)
    local Size = Data:len()
        
    SafeNet.start("image_render_flush")
    SafeNet.writeUInt(Size,16)
    SafeNet.writeData(Data, Size)
    SafeNet.send()
    
    Renderer.queue = {}
    Renderer.netBytesAccumulated = 0
end

function Renderer.Finish()
    Renderer.FlushBuffer()
    
    SafeNet.start("image_render_finish")
    SafeNet.send()
end

function Renderer.Abort()    
    SafeNet.start("image_render_abort")
    SafeNet.send()
end

function Renderer.setPixel(x,y,r,g,b,a)
    Renderer.queue[#Renderer.queue + 1] = {
        x = x,
        y = y,
        r = r,
        g = g,
        b = b,
        a = a or 255
    }
    
    local pixelSize = Renderer.hasAlpha and 4 or 3
    if not Renderer.sequential then
        pixelSize = pixelSize + 4
    end
    
    Renderer.netBytesAccumulated = Renderer.netBytesAccumulated + pixelSize
    
    if Renderer.netBytesAccumulated >= 19000 then
        Renderer.FlushBuffer()
    end
    
end

return Renderer
