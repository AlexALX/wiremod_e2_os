--@name Network for Renderer
--@author AlexALX
--@shared
--@include https://raw.githubusercontent.com/Jacbo1/Public-Starfall/main/SafeNet/safeNet.lua as SafeNet
--@include alx_pc/shared/cl_renderer.txt

SafeNet = require("SafeNet")

if SERVER then

    SafeNet.receive("image_render_init", function()
        if SafeNet.cancelAll!=nil then
            SafeNet.cancelAll()
        end
            
        SafeNet.start("image_render_init")
        SafeNet.writeUInt(SafeNet.readUInt(16),16)
        SafeNet.writeUInt(SafeNet.readUInt(16),16)
        SafeNet.writeUInt(SafeNet.readUInt(16),16)
        SafeNet.writeBool(SafeNet.readBool())
        SafeNet.writeBool(SafeNet.readBool())
        SafeNet.writeBool(SafeNet.readBool())
        SafeNet.writeBool(SafeNet.readBool())
        SafeNet.writeBool(SafeNet.readBool())
        SafeNet.send()
    end)
    
    SafeNet.receive("image_render_flush", function()
        local Size = SafeNet.readUInt(16)
        local CompData = SafeNet.readData(Size)
        
        SafeNet.start("image_render_flush")
        SafeNet.writeUInt(Size,16)
        SafeNet.writeData(CompData,Size)        
        SafeNet.send()
    end)
    
    SafeNet.receive("image_render_finish", function()
        SafeNet.start("image_render_finish")
        SafeNet.send()
    end)
    
    SafeNet.receive("image_render_abort", function()        
        SafeNet.start("image_render_abort")
        SafeNet.send()
    end)

else
    
    local NetworkRenderer = {
        sequential = false,
        TopDown = false,
        hasAlpha = false, 
        width = 0,
        height = 0,  
        x = 0,
        y = 0 
    }

    local Renderer = require("alx_pc/shared/cl_renderer.txt")
    
    SafeNet.receive("image_render_init", function()
        NetworkRenderer.width = SafeNet.readUInt(16)
        NetworkRenderer.height = SafeNet.readUInt(16)
        NetworkRenderer.SizeLimit = SafeNet.readUInt(16)
        NetworkRenderer.NoKeepAR = SafeNet.readBool()
        local isMaterialRender = SafeNet.readBool()
        NetworkRenderer.sequential = SafeNet.readBool()
        NetworkRenderer.TopDown = SafeNet.readBool()
        NetworkRenderer.hasAlpha = SafeNet.readBool()
        
        NetworkRenderer.x = 0
        NetworkRenderer.y = 0
        
        if not NetworkRenderer.TopDown then
            NetworkRenderer.y = NetworkRenderer.height - 1
        end
        
        Renderer.Init(
            NetworkRenderer.width, 
            NetworkRenderer.height, 
            NetworkRenderer.SizeLimit,
            NetworkRenderer.NoKeepAR,
            isMaterialRender
        )    
    end)
    
    SafeNet.receive("image_render_flush", function()
        local Size = SafeNet.readUInt(16)
        local CompData = SafeNet.readData(Size)
        
        local Data = bit.decompress(CompData)
        
        local IsSafeStream = SafeNet.stringstream!=nil
        
        local Stream 
        if IsSafeStream then
            Stream = SafeNet.stringstream(Data)    
        else
            Stream = bit.stringstream(Data)
        end
        
        try(function()
            
            local QueueCount = Stream:readUInt16()
            
            local x, y, a = NetworkRenderer.x,NetworkRenderer.y,255

            for I=1,QueueCount do
                if not NetworkRenderer.sequential then
                    x = Stream:readUInt16()
                    y = Stream:readUInt16()
                end
            
                local r = Stream:readUInt8()
                local g = Stream:readUInt8()
                local b = Stream:readUInt8()
                if NetworkRenderer.hasAlpha then
                    a = Stream:readUInt8()
                end

                Renderer.setPixel(x,y,r,g,b,a)
                
                if NetworkRenderer.sequential then 
                    x = x+1
                    if x>=NetworkRenderer.width then
                        if not NetworkRenderer.TopDown then
                            y = y-1
                        else
                            y = y+1
                        end
                        x = 0 
                    end    
                end
            end
            
            NetworkRenderer.x = x
            NetworkRenderer.y = y
            
            Renderer.Flush()
        end)
    end)
    
    SafeNet.receive("image_render_finish", function()
        Renderer.Finish()
    end)
    
    SafeNet.receive("image_render_abort", function()
        Renderer.Abort()
    end)
    
end
