function initGpu() {
    local Info = dsGetTable()
    NB = Info[1,string]
    Devices = Info[2,table]
    BOOT_HDD = Info[3,table]
    E2_CPU = Devices["cpu",entity]
    E2_SPU = Devices["spu",entity]
    Keyboard = Devices["keyboard",table]
    HDD_IO = Devices["hdd_io",table]
    HDDS = Info[5,table]
    FS_TYPES = Info[6,table]
    TickQuota = Info[7,number]
    Init = 1
    if (!Info[4,number]) {
        dsSendDirect("init",BIOS,"gpu")
        egpClear()
        dsClear()
        csClear()
    }
    GpuData = table()
    GpuPersistData = table()
    
    # add to global BIOS table
    Devices["GPU_Screens",table] = Screens
}

function ready() {
    SysFS = dsGetTable()
    init()
}

function gpuShutdown() {
    if (ShutHandler!="") {
        ShutHandler(dsGetArray())   
        ShutHandler = ""
    }
    stopAllTimers()
    GpuData = table()
    GpuPersistData = table()
    Init = 0
    PlugHandler = ""
    egpClear()
    dsClear()
    csClear()
    EGPEvent = function() {}
    DSEvent = function() {}
    CSEvent = function() {}
}

function screenDataClear() {
    Screens:clear()
    EGP = nowirelink()
    EGP_ID = -1
    EGPContainer[1,wirelink] = EGP
    DS = nowirelink()
    DS_ID = -1
    DSContainer[1,wirelink] = DS
    CS = nowirelink()
    CS_ID = -1
    CSContainer[1,wirelink] = CS
}

function pcie_init() {
    dsSendDirect("pcie",BIOS,table("gpu",entity(),Device_Name))
    screenDataClear()
    for (I=0,ScreenSockets-1) {
        dsSendDirect("wirelink",entity(ioGetInputNumber("ScreenSockG_"+I)),entity())    
    }
}

function shutdown() {
    gpuShutdown()
    dsSendDirect("SHUTDOWN",BIOS,dsGetArray())
    screenDataClear()
}

function number findscreen(ID) {
    for (I=0,ScreenSockets-1) {
        if (ID==ioGetInputNumber("ScreenSockG_"+I)) {
            return I+1
        }
    }
    return -1
}

function screen_connect() {
    local Info = dsGetTable()
    local ID = Info[1,number]
    local WL = Info[2,wirelink]    
    local WLI = WL:entity():id()
    if (WL) {
        local Type = WL:entity():type()
        if (Type=="gmod_wire_expression2") {
            local ScreenType = gpuScreenType(WL)
            if (ScreenType==2) {
                Type = "gmod_wire_digitalscreen"
            } elseif (ScreenType==1) {
                Type = "gmod_wire_consolescreen"
            }
        }

        if (!EGP:entity():isValid() & Type=="gmod_wire_egp") {
            EGP = WL
            EGP_ID = WLI
            EGPContainer[1,wirelink] = WL
            Screens[WLI,number] = findscreen(ID)
            Screens["EGP_WL",wirelink] = WL
            EGPEvent()
        } elseif (!CS:entity():isValid() & Type=="gmod_wire_consolescreen") {       
            CS = WL
            CS_ID = WLI
            CSContainer[1,wirelink] = WL
            Screens[WLI,number] = findscreen(ID)
            Screens["CS_WL",wirelink] = WL
            CSEvent()
        } elseif (!DS:entity():isValid() & Type=="gmod_wire_digitalscreen") {
            DS = WL
            DS_ID = WLI
            DSContainer[1,wirelink] = WL
            Screens[WLI,number] = findscreen(ID)
            Screens["DS_WL",wirelink] = WL
            DSEvent()
        }
        
        if (GpuData["BSOD",array]:count()) {
            alxos_bsod(GpuData["BSOD",array])    
        }
        
    }
}

if (inputClk()) {
    if (!Init) { exit() }
    local Name = inputClkName()
    if (Name:find("ScreenSockG")) {
        local Ent = entity(ioGetInputNumber(Name))
        local ID = Name:right(1):toNumber()+1
        if (Ent) {
            dsSendDirect("wirelink",Ent,entity())    
        } else {
            local BSOD = 3
            
            if (Screens[EGP_ID,number]==ID) {
                EGP = nowirelink()
                if (EGP_ID>0) {
                    Screens[EGP_ID,number] = 0
                }
                EGP_ID = -1
                EGPContainer[1,wirelink] = EGP
                Screens["EGP_WL",wirelink] = EGP
                EGPEvent()
                
                if ((DS:entity():isValid() | CS:entity():isValid()) & CurrentProg=="alxos_console") {
                    #local IsConsole = "alxos_console_isconsole"()[number]
                    #if (IsConsole) { BSOD = 0 }
                    BSOD = 0
                }
            } 
            
            if (Screens[DS_ID,number]==ID) {
                DS = nowirelink()
                if (DS_ID>0) {
                    Screens[DS_ID,number] = 0
                }
                DS_ID = -1
                DSContainer[1,wirelink] = DS
                Screens["DS_WL",wirelink] = DS
                DSEvent()
                
                BSOD = 5
                if (EGP:entity():isValid() | CS:entity():isValid()) {
                    BSOD = 0
                }
            }

            if (Screens[CS_ID,number]==ID) {
                CS = nowirelink()
                if (CS_ID>0) {
                    Screens[CS_ID,number] = 0
                }
                CS_ID = -1
                CSContainer[1,wirelink] = CS
                Screens["CS_WL",wirelink] = CS
                CSEvent()
                
                BSOD = 6
                if (EGP:entity():isValid() | DS:entity():isValid()) {
                    BSOD = 0
                }
            }

            if (BSOD) {
                stopAllTimers()
                dsSendDirect("BSOD",BIOS,array(BSOD))
            }
        }
    }    
    exit()
}

            PlugHandler(Name)
        }
    } elseif (DsHandler!="") {
        DsHandler(Name)
    }
