function table midiSelectSampleSfz(Instrument:table, Program:number, Note:number)
{
    local Samples = Instrument["samples", table]
    local Result = table()
    
    #         
    local PathKey = toString(Note) + "_path"
    local RootKey = toString(Note) + "_root"
    
    if (Samples[PathKey, string] != "") {
        Result["sample", string] = Samples[PathKey, string]
        Result["rootNote", number] = Samples[RootKey, number]
    } else {
        local SynthInstrument = midiProgramSynth(Program)
        
        Result = midiSelectSample(SynthInstrument, Program, Note)
    }
    
    return Result
}
