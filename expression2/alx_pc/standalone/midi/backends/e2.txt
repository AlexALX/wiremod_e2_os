function table midiGetInstrument(Program:number)
{
    local Cache = Midi["instrumentCache",table]

    if (Cache[Program,table]:count() > 0) {
        return Cache[Program,table]
    }

    local Instrument = MidiProgram(Program)[table]

    Cache[Program,table] = Instrument

    return Instrument
}

function table midiGetSample(Instrument:table, Program:number, Note:number)
{
    local Index = Program * 128 + Note

    local Cache = Midi["sampleCache",table]

    if (Cache[Index,table]:count() > 0) {
        return Cache[Index,table]
    }
    
    local Sample = MidiSample(Instrument,Program,Note)[table]

    Cache[Index,table] = Sample

    return Sample
}

function midiUpdateVoices(Backend:table) {
    local MaxVoices = 16
    
    VoicesArray = array()
    for (I=1,MaxVoices) {
        local Ent = ioInputEntity("Voice"+I)
        if (Ent!=noentity()) {
            VoicesArray:pushEntity(Ent)  
        }             
    }

    Backend["voiceCount",number] = VoicesArray:count()
}

function midiBackendE2SetMode(Mode:string) {
    MidiPlayMode = Mode:lower()
    
    if (Mode=="addon") {
        MidiProgram = function(Program:number) { 
            return midiProgramAddon(Program) 
        }
        MidiSample = function(Instrument:table,Program:number,Note:number) {
            return midiSelectSampleSfz(Instrument,Program,Note)    
        }
    } elseif (Mode=="instrumental") {
        MidiProgram = function(Program:number) { 
            return midiProgramInstrument(Program) 
        }
        MidiSample = function(Instrument:table,Program:number,Note:number) {
            return midiSelectSample(Instrument,Program,Note)    
        }
    } else {
        MidiProgram = function(Program:number) { 
            return midiProgramSynth(Program) 
        }
        MidiSample = function(Instrument:table,Program:number,Note:number) {
            return midiSelectSample(Instrument,Program,Note)    
        }
    }  
}

function table midiBackendE2(Mode:string) {
    local Backend = table()
    
    midiBackendE2SetMode(Mode)
    
    midiUpdateVoices(Backend)
    
    Backend["Init",function] = function() {
        Midi["instrumentCache",table] = table()
        Midi["sampleCache",table] = table()
        MidiWarning = ""
    }
    
    Backend["Check",function] = function() {
        return 1
    }
    
    Backend["Play",function] = function() {
        ActiveNotes = 0
        NoSoundWarning = 0
    }
    
    Backend["Pause",function] = function() {
        soundPurge()
    }
    
    Backend["Stop",function] = function() {
        soundPurge()
        MidiWarning = ""
        NoSoundWarning = 0
    }

    Backend["TickStart",function] = function() {        
        if (MidiWarning=="" & NoSoundWarning) {
            MidiWarning = "Sound playback may be throttled or muted\n"+
            "due to Expression2 ConVar limits.\n"+
            "Check Recommended E2 settings for midi."
            
            if (!MidiNoPrint) {
                print(MidiWarning)   
                
                print(
                    "\nRecommended e2 settings for better playback:\n"+
                    "wire_expression2_maxsounds 128\n"+
                    "wire_expression2_sound_burst_max 64\n"+
                    "wire_expression2_sound_burst_rate 0\n"+     
                    "And then do wire_expression2_reload or reload map"
                )        
            }
        } 
    }

    Backend["TickEnd",function] = function() {
        #
    }

    Backend["VoiceOn",function] = function(Voice:number)
    {
        local V = Midi["voices",table][Voice,table]
    
        local SoundId = Voice + 1
        local Ent = VoicesArray[SoundId,entity]
        if (!Ent:isValid()) { return }
    
        local Instrument = midiGetInstrument(V["program",number])
    
        local Sample = midiGetSample(
            Instrument,
            V["program",number],
            V["note",number]
        )
        
        Ent:soundPlay(
            SoundId,
            0,
            Sample["sample",string]
        )
        
        ActiveNotes++
    
        if (!soundPlaying(SoundId)) { 
            NoSoundWarning = 1
            return 
        }
        
        local Pitch = round(
            100 * (2 ^ (
                (V["note",number] - Sample["rootNote",number]) / 12
            ))
        )

        soundPitch(SoundId, clamp(Pitch, 0, 255))
    
        local Volume =
            (V["velocity",number] / 127) *
            (V["volume",number] / 127) *
            (V["expression",number] / 127) *
            Instrument["gain",number]
    
        soundVolume(SoundId, clamp(Volume, 0, 1))
    }

    Backend["VoiceOff",function] = function(Voice:number)
    {
        local SoundId = Voice + 1
        soundStop(SoundId)
        
        ActiveNotes--
    }

    Backend["VoiceVolume",function] = function(Voice:number)
    {
        local V = Midi["voices",table][Voice,table]
    
        local Volume =
            (V["velocity",number] / 127) *
            (V["volume",number] / 127) *
            (V["expression",number] / 127)
            
        if (!soundPlaying(Voice + 1)) { return }
    
        soundVolume(Voice + 1, clamp(Volume, 0, 1))
    }
    
    Backend["VoiceExpression",function] = function(Voice:number)
    {
        Backend["VoiceVolume",function](Voice)
    }
    
    Backend["VoicePan",function] = function(Voice:number)
    {
        #  Wire Sound does support pan.
    }

    return Backend
}

event input(Name:string) {
    if (Name:find("Voice")==1) {
        midiUpdateVoices(CurrentBackend)
    }
}
