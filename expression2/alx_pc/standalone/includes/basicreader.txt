#include "alx_pc/standalone/includes/screen"

function number checkScreen(DS:wirelink) {
    if (DS==nowirelink()) {
        print(_HUD_PRINTTALK,"Digital Screen is not connected.") 
        return 0
    }

    local ScreenName = screenName(DS)
    
    if (ScreenName=="") {
        print(_HUD_PRINTTALK,"Entity what you did connected is not Digital Screen.\n"+
        "It is located in Wire -> Visuals -> Screens -> Digital Screen.\nThis is NOT EGPv3 screen.") 
        return 0    
    }
    return 1  
}

function readerChatCommand(Command:string, Cmd:string, Url:string) {            
    if (Cmd==Command+"abort" | Cmd==Command & Url:lower() == "abort") {        
        Funcs["readerAbort", function]()
    } elseif (Cmd == Command) {
        if (!Funcs["readerCheck",function]()[number]) { exit() }            
        #ifndef httpCanRequest()
            print(_HUD_PRINTTALK,"HTTP extension is not enabled. Please enable via wire_expression2_extension_enable http") 
        #else
        
            if (Url=="") {
                print(_HUD_PRINTTALK,
                    "Load file from the web."
                ) 
                exit()
            }
        
            if (!httpCanRequest()) {
                print(_HUD_PRINTTALK, "HTTP Busy. Please wait...")
                exit()
            }
                
            timer(0.01, function() {
                httpRequest(Url) 
            })
        #endif
    } elseif (Cmd == Command+"file") {
        if (!Funcs["readerCheck",function]()[number]) { exit() }
        
        #ifndef fileCanLoad()
            print(_HUD_PRINTTALK,"FILE extension is not enabled. Please enable via wire_expression2_extension_enable file") 
        #else
        
            if (Url=="") {
                print(_HUD_PRINTTALK,
                    "Load file from the disk in garrysmod/data/e2files/ folder."+
                    (ExtWarning!="" ? "\n"+ExtWarning : "")
                )
                exit()
            }
        
            if (!fileCanLoad()) {
                print(_HUD_PRINTTALK, "File Busy. Please wait...")
                exit()
            }
            
            timer(0.01, function() {
                try {
                    fileLoad(Url) 
                } catch(E:string) {
                    Funcs["readerError", function](E + (ExtWarning!="" ? "\n"+ExtWarning : ""), FileName)  
                }
            })
        #endif
    }
}

event chat(Player:entity, Said:string, Team:number) {
    if (!Command) { exit() }
    
    if (Player==owner()) { 
        local Find = Said:find(" ")
        
        local Cmd = Said:lower()
        local Url = ""        
        if (Find) {
            Cmd = Said:left(Find-1):lower()
            Url = Said:sub(Find+1)
        }
        
        readerChatCommand(Command, Cmd, Url)
    }
}

if (~LongURL & LongURL!="") {
    if (!Funcs["readerCheck",function]()[number]) { exit() }

    #ifndef httpCanRequest()
        print(_HUD_PRINTTALK,"HTTP extension is not enabled. Please enable via wire_expression2_extension_enable http")   
        exit()
    #else
        if (!httpCanRequest()) {
            print(_HUD_PRINTTALK,"HTTP Request Canceled - Please wait a bit...")   
            exit()
        }
        httpRequest(LongURL) 
    #endif
}

if (~FileName & FileName!="") {
    if (!Funcs["readerCheck",function]()[number]) { exit() }

    #ifndef fileCanLoad()
        print(_HUD_PRINTTALK,"FILE extension is not enabled. Please enable via wire_expression2_extension_enable file")   
        exit()
    #else
        if (!fileCanLoad()) {
            print(_HUD_PRINTTALK, "File Busy. Please wait...") 
            exit()
        }

        try {
            fileLoad(FileName)
        } catch(E:string) {
            Funcs["readerError", function](E + (ExtWarning!="" ? "\n"+ExtWarning : ""), FileName) 
        }
    #endif
}

#ifdef httpCanRequest()
event httpLoaded(Body:string, Size:number, Url:string)
{
    Funcs["readerLoad", function](Body, Size, Url)
}

event httpErrored(Error:string, Url:string)
{
    Funcs["readerError", function](Error, Url)
}
#endif

#ifdef fileLoad(string)
event fileLoaded(Filepath:string,Data:string) {
    Funcs["readerLoad", function](Data, Data:length(), Filepath) 
}

event fileErrored(FilePath:string, Status:number) {
    local Error = ""
    switch(Status) {
        case _FILE_404,
            Error = "File Error: Not Found (too big? max "+fileMaxSize()+"kb)\n"+
            "You can change size via wire_expression2_file_max_size convar"
            break     
        case _FILE_TIMEOUT,
            Error = "File Error: Timeout"
            break 
        case _FILE_TRANSFER_ERROR,
            Error = "File Error: Transfer Error" 
            break 
        default,
            Error = "File Error: Unknown Error ("+Status+")" 
            break 
    }
    
    Funcs["readerError", function](Error, FilePath) 
}
#endif
