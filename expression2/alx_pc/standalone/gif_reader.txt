@name GIF Reader By AlexALX
@inputs AnimateMode:string NextFrame:number GifMinDelay:number
@persist Animate:number MinDelay:number
@strict

#[
    GIF Reader
    Created by AlexALX (c) 2026
    -------------------
    GIF Parser, specifications:
    - Max resolution is unlimited, but will be automatically resized to 512x512 (DS limit) or 1024x1024 (Quad DS)
    - Baseline and Interlace files supported
    - Animation support with few modes
    - Loop and Disposal support
    - Due to digital screen speed issues animations 
       work properly only on very small images
       or you can resize output by GifSizeLimit

    Usage:
    - Spawn one GIF Reader (this file)
    - Connect this E2 to Digital Screen (this is not EGP screen! located in Wire -> Visuals -> Screens -> Digital Screen)
    - Or connect "DS" input to Quad Digital Screen E2 VirtualScreen output (this is custom device dupe)
    
    Chat commands:
    !gif http://url_with_gif_file - open gif file from the web
    Note: long url may be cut, use input LongURL instead for such case (via string value) 
    
    !giffile filename.gif.txt - open gif file from a garrysmod/data/e2files/ folder, file extension must be renamed to .txt
    
    !gifabort - abort rendering
    
    !gifmode - animation mode

    Animation Modes:
     normal - no animation, just first frame
     manual - play animation manually via input (button or command)
     animate - play animation, but it's slow (1 frame per few seconds)
    
    !gifnext - play next frame in manual mode
    
    !gifsize - set maximum image dimensions before scaling (1 to 512 for DS, 1024 for Quad DS)
    !gifdelay - set minimum delay between animation frames (min 0.1)
    !gifkeepar - set image Aspect Ratio mode (0 or 1)
    
    Additional Wire Inputs:
    GifMinDelay - Minimum delay between animations (2 sec default)
    ImageSizeLimit - Maximum image dimensions before scale
     This can be used to scale down any image before drawing
    ImageNoKeepAR - Do not keep Aspect Ratio of image
    
]#

#include "alx_pc/standalone/includes/imagereader"
#include "alx_pc/standalone/gif_reader/_init_"

function animateMode(ModeString:string) {
    local Mode = 0
    if (ModeString=="manual") {
        Mode = 1
    } elseif (ModeString=="animate") {
        Mode = 2    
    }        
    return Mode
}

if (first() | duped()) {
    Command = "!gif"
    
    ExtWarning = "File extension must be changed to .txt!\nOr file will not load."
    
    Funcs = table()
    
    Funcs["readerAbort", function] = function() {
        print(_HUD_PRINTTALK,"GIF rendering canceled.")     
        gifAbort()
    }
    
    Funcs["readerLoad", function] = function(Body:string, Size:number, Path:string) {
        gifInit(Body,Animate,SizeLimit,NoKeepAR,MinDelay)
    }
    
    Funcs["readerCheck", function] = function() {
        return checkScreen(DS)
    } 
    
    Funcs["readerError", function] = function(Error:string, Path:string) {
        print(_HUD_PRINTTALK, Error)
    }
    
    Animate = animateMode(AnimateMode)
    MinDelay = GifMinDelay
}

function gifNextCommand() {
    local Result = gifNextManualFrame()
    
    if (Result==1) {    
        print(_HUD_PRINTTALK,
            "Processing next frame - "+(Gif["frameProcessed",number]+1)+"/"+Gif["frameTotal",number]
        )
    } elseif (Result==2) {    
        print(_HUD_PRINTTALK,
            "Gif animation already finished."
        )
    } elseif (Result==-2) {
        print(_HUD_PRINTTALK,
            "Gif is not running"
        ) 
    } elseif (Result==-1) {
        print(_HUD_PRINTTALK,
            "Current Gif mode is not manual\n"+
            "Please restart gif reader."
        ) 
    } else {
        print(_HUD_PRINTTALK,
            "GIF Reader is busy, please wait."
        )
    }
}

event chat(Player:entity, Said:string, Team:number) {
    if (!Command) { exit() }
    
    if (Player==owner()) { 
        
        local Find = Said:lower():find(" ")
        
        local Cmd = Said:lower()
        local Mode = ""        
        if (Find) {
            Cmd = Said:left(Find-1):lower()
            Mode = Said:sub(Find+1):lower()
        }

        if (Cmd==Command+"mode") {
            if (Mode=="") {
                print(_HUD_PRINTTALK,
                    "Available Gif Modes:\n"+
                    "normal - no animation, just first frame\n"+
                    "manual - play animation manually via input (button or command)\n"+
                    "animate - play animation, but it's slow (1 frame per few seconds)"
                )
                exit()
            }
            
            if (Mode == "manual") {
                Animate = 1
                print(_HUD_PRINTTALK,
                    "Gif mode set to manual\n"+
                    "Next frame will be read after input"
                )
            } elseif (Mode == "animate") {
                Animate = 2
                print(_HUD_PRINTTALK,
                    "Gif mode set to animated\n"+
                    "Every frame read will be rendered automatically\n"+
                    "This mode is very slow\n"+
                    "STRONGLY recommend set !gifsize 128 for this mode!"
                ) 
            } elseif(Mode == "normal") {
                Animate = 0
                print(_HUD_PRINTTALK,
                    "Gif mode set to normal\n"+
                    "No animation will be displayed, rendering only the first frame"
                ) 
            } else {
                print(_HUD_PRINTTALK,
                    "Invalid GIF mode"
                ) 
            }
        } elseif (Cmd==Command+"delay") {
            if (Mode=="") {
                print(_HUD_PRINTTALK,
                    "Gif delay command usage: !gifdelay 0.2"
                )
                exit()
            }
            
            local Size = Mode:toNumber()
            MinDelay = max(Size,0.1)
            print(_HUD_PRINTTALK,
                "Minimum delay between animations set to "+MinDelay
            )
        } elseif (Said:lower()=="!gifnext") {
            gifNextCommand()
        }
    }
}

if (~AnimateMode) {
    Animate = animateMode(AnimateMode)
}

if (~GifMinDelay) {
    MinDelay = GifMinDelay
}

if (~NextFrame & NextFrame) {
    gifNextCommand()
}
