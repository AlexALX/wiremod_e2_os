@name JPG Reader By AlexALX
@inputs ProgressivePreview:number
@persist PreviewMode:number
@strict

#[
    JPG Reader
    Created by AlexALX (c) 2026
    -------------------
    JPG Parser, specifications:
    - Max resolution is unlimited, but will be automatically resized to 512x512 (DS limit) or 1024x1024 (Quad DS)
    - Support Baseline (8/12bits) and Progressive JPG
    - Support YUV 4:2:0 4:2:2 4:2:2v 4:4:4
    - Slow, no multithread
    - Support progressive preview mode
    
    Usage:
    - Spawn one JPG Reader (this file)
    - Connect this E2 to Digital Screen (this is not EGP screen! located in Wire -> Visuals -> Screens -> Digital Screen)
    - Or connect "DS" input to Quad Digital Screen E2 VirtualScreen output (this is custom device dupe)
    
    Chat commands:
    !jpg http://url_with_jpg_file - open jpg file from the web
    Note: long url may be cut, use input LongURL instead for such case (via string value) 
    
    !jpgfile filename.jpg - open jpg file from a garrysmod/data/e2files/ folder
    
    !jpgabort - abort rendering
    
    !jpgmode - normal, preview, fullpreview

    Modes:
     normal - no preview, fastest render
     preview - first AC scan preview
     fullpreview - every scan rerender (slow)
    Note: this mode works only for progressive images 
    
    !jpgsize - set maximum image dimensions before scaling (1 to 512 for DS, 1024 for Quad DS)
    !jpgkeepar - set image Aspect Ratio mode (0 or 1)
    
    Additional Wire Inputs:
    ProgressivePreview - Set jpg mode
    ImageSizeLimit - Maximum image dimensions before scale
     This can be used to scale down any image before drawing
    ImageNoKeepAR - Do not keep Aspect Ratio of image
]#

#include "alx_pc/standalone/includes/imagereader"
#include "alx_pc/standalone/jpg_reader/_init_"

if (first() | duped()) {
    Command = "!jpg"
    
    Funcs = table()
    
    Funcs["readerAbort", function] = function() {
        print(_HUD_PRINTTALK,"JPG rendering canceled.")     
        jpgAbort()
    }
    
    Funcs["readerLoad", function] = function(Body:string, Size:number, Path:string) {
        jpgInit(Body, PreviewMode, SizeLimit, NoKeepAR)
    }
    
    Funcs["readerCheck", function] = function() {
        return checkScreen(DS)  
    } 
    
    Funcs["readerError", function] = function(Error:string, Path:string) {
        print(_HUD_PRINTTALK, Error)
    }
    
    PreviewMode = ProgressivePreview
}

event chat(Player:entity, Said:string, Team:number) {
    if (!Command) { exit() }
    
    if (Player==owner()) { 
        
        local Find = Said:lower():find(" ")
        
        local Cmd = Said:lower()
        local Mode = ""        
        if (Find) {
            Cmd = Said:left(Find-1):lower()
            Mode = Said:sub(Find+1):lower()
        }

        if (Cmd==Command+"mode") { 
            if (Mode=="") {
                print(_HUD_PRINTTALK,
                    "Available Jpg Modes:\n"+
                    "normal - no preview, fastest render\n"+
                    "preview - first AC scan preview\n"+
                    "fullpreview - every scan rerender (slow)\n"+
                    "Note: this mode works only for progressive images" 
                )
                exit()
            }
            
            if (Mode == "fullpreview") {
                PreviewMode = 2
                print(_HUD_PRINTTALK,
                    "Jpg mode set to fullpreview\n"+
                    "Every scan read will rerender image\n"+
                    "This mode is very slow\n"+
                    "Note: this mode works only for progressive images"
                ) 
            } elseif (Mode == "preview") {
                PreviewMode = 1    
                print(_HUD_PRINTTALK,
                    "Jpg mode set to preview\n"+
                    "Image preview will display once after the 2nd scan\n"+
                    "This mode balances speed and visualization\n"+
                    "Note: this mode works only for progressive images"
                ) 
            } else {
                PreviewMode = 0 
                print(_HUD_PRINTTALK,
                    "Jpg mode set to normal\n"+
                    "No preview will be displayed, rendering only the final image\n"+
                    "This mode is fastest"
                ) 
            }
        }
    }
}

if (~ProgressivePreview) {
    PreviewMode = ProgressivePreview
}
