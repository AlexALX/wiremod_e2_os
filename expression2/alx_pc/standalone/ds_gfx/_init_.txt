#include "alx_pc/standalone/ds_gfx/_shared_"
#include "alx_pc/standalone/ds_gfx/text"

if (first() | duped()) {
    
    DsCurrentFont = table()
    DsText = table()
    
    DSWLContainer = table()
    
    DSCache = table()
}

# DS:table used for store reference without need track changes everywhere
# used in GPU with DSContainer table
function dsInit(DSTable:table, Width:number, Height:number) {
    DSWLContainer = DSTable
    DsWidth = Width
    DsHeight = Height
    DsCacheDelay = 0
    
    local DS = DSTable[1,wirelink]
    local DsShift = DS["DsShifted",number]
    
    DS[1048569+DsShift] = 2
    DS[1048574+DsShift] = 0
    DS[1048575+DsShift] = 1
    DS[1048573+DsShift] = Width
    DS[1048572+DsShift] = Height

    dsStopQueues()
}

function dsInit(DS:wirelink, Width:number, Height:number) {
    dsInit(table(DS), Width, Height)    
}

function dsSetFont(Font:table)
{
    DsFont = Font["glyphs",table]
    DsFontWidth = Font["width",number]
    DsFontHeight = Font["height",number]
    DsFontSpacing = Font["spacing",number]
    DsFontLineHeight = DsFontHeight + DsFontSpacing
    DsFontCharWidth = DsFontWidth + DsFontSpacing
}

function dsCacheDelay(Seconds:number) {
    DsCacheDelay = Seconds    
}
