@persist DSTextQueue:table DSCurrentStep:string DSLastStep:string DSCache:table

#include "alx_pc/standalone/ds_gfx/_shared_"
#include "alx_pc/standalone/ds_gfx/text"

function dsDrawTextStart(X:number, Y:number, Text:string)
{
    local DsText = table()

    DsText["text", string] = Text:replace("\r", "")
    DsText["x", number] = X
    DsText["y", number] = Y

    DsText["startX", number] = X
    DsText["index", number] = 1

    DsText["type", string] = "normal"
    
    DSTextQueue:pushTable(DsText)
    DSCurrentStep = "executeQueue"
}

function number dsDrawTextStep(DsText:table)
{
    local Text = DsText["text", string]
    local Index = DsText["index", number]

    if (Index > Text:length()) {
        return 1
    }

    local Char = Text:sub(Index, Index)
    
    if (Char == "\n") {

        DsText["x", number] = DsText["startX", number]
        DsText["y", number] = DsText["y", number] + DsFontHeight + DsFontSpacing

    } else {

        dsDrawChar(
            DsText["x", number],
            DsText["y", number],
            Char
        )

        DsText["x", number] = DsText["x", number] + DsFontWidth + DsFontSpacing
    }

    DsText["index", number] = Index + 1

    return 0
}

function dsDrawCachedTextStep(Item:table)
{
    if (Item["clearLine", number] == 1) {
        dsTextClearLine(Item["y", number])
        Item["clearLine", number] = 0    
    }    
    
    local Index = Item["index",number]
    local Text = Item["text",string]

    if (Index > Text:length()) {
        return 1
    }

    local Char = Text:sub(Index, Index)

    dsDrawChar(
        Item["x",number] + (Index-1)*(DsFontWidth+DsFontSpacing),
        Item["y",number],
        Char
    )

    Item["index",number] = Index + 1

    return 0
}

function dsDrawCachedText(X:number, Y:number, Text:string)
{
    local Key = X + ":" + Y
    
    if (DSCache[Key,string] == Text) {
        return
    }
    
    DSCache[Key,string] = Text
    
    local Item = table()
    
    Item["x",number] = X
    Item["y",number] = Y
    Item["text",string] = Text
    Item["index",number] = 1
    Item["clearLine", number] = 1
    
    DSTextQueue:pushTable(Item)
    DSCurrentStep = "executeQueue"
}

function dsClearCache() {
    DSCache = table()   
    DSTextQueue = table()
}

function dsStopQueues() {
    dsClearCache()
    DSCurrentStep = ""  
    DSLastStep = ""  
}

event tick() {
    if (DSCurrentStep) {
        while (perf(90)) { 
            if (DSTextQueue:count() <= 0) {
                DSCurrentStep = ""
                break
            }

            local CurrentItem = DSTextQueue[1, table]
            
            if (CurrentItem["type", string] == "normal") {
                if (dsDrawTextStep(CurrentItem)) {
                    DSLastStep = "drawText"
                    DSTextQueue:remove(1)   
                }
            } else {
                if (dsDrawCachedTextStep(CurrentItem)) {
                    DSLastStep = "drawCachedText"
                    DSTextQueue:remove(1)   
                }
            }            
        }
    }
}
