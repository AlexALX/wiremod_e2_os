@persist DsFont:table DsFontWidth:number DsFontHeight:number DsFontSpacing:number DsFontLineHeight:number DsFontCharWidth:number
@persist DsWidth:number DsHeight:number
@persist DsText:table DsCacheDelay:number DSWLContainer:table

function dsDrawPixel(DS:wirelink, X:number, Y:number, ColorNum:number)
{
    if (X < 0 | Y < 0 | X >= DsWidth | Y >= DsHeight) { return }
    DS[X + Y * DsWidth] = ColorNum
}

function dsDrawPixel(DS:wirelink, X:number, Y:number)
{
    if (X < 0 | Y < 0 | X >= DsWidth | Y >= DsHeight) { return }
    DS[X + Y * DsWidth] = 16777215 # 0xFFFFFF
}
