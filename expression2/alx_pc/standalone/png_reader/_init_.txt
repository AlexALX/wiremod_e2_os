@name PNG Reader Main Chip

#include "alx_pc/standalone/png_reader/_shared_"
#include "alx_pc/standalone/png_reader/parser"
#include "alx_pc/standalone/png_reader/decompress"
#include "alx_pc/standalone/png_reader/draw"
#include "alx_pc/standalone/png_reader/progress"

function pngInit(Data:string,MaxSize:number,NoKeepAR:number)
{
    Png = table()
    Png["start",number] = curtime()
    
    Error = ""
    Progress = table()
    Processing = 0
    Warning = ""
    
    Png["MaxSize", number] = MaxSize
    Png["NoKeepAR", number] = NoKeepAR
    
    Png["data",string] = Data
    Png["size",number] = Data:length()
    Png["pos",number] = 1
    Png["state",string] = ""
    
    Png_CompressedData = ""
    
    deflateInit()
    
    local SigValid = 1
    local ValidSig = array(137, 80, 78, 71, 13, 10, 26, 10)
    for(I = 1, 8) {
        if(toByte(Data, I) != ValidSig[I, number]) {
            SigValid = 0
            break
        }
    }
    
    if (!SigValid) {
        timer(0.001, function() {
            Error = "Not a valid PNG file"
            if (!PNGNoPrint) {
                print(Error)
            }
        })
        return
    }
    
    Progress["stepTotal",number] = 5
    Png["pos",number] = 9
    
    timer(0.001, function() {
        Processing = 1
        Png["state",string] = "parse_chunks"    
    })
    
    if (!PNGNoPrint) {
        print("PNG detected")
    }
}

# clear e2 memory
function pngClear() {
    Png = table()    
    Png_CompressedData = ""
}

function pngAbort() {
    Png["state",string] = ""
    CurrentState = ""
    Progress = table()
    ProgressText = ""
    Error = ""
    Processing = 0
    pngClear()
}

function pngDone() {
    Png["state", string] = "done"
    
    pngUpdateProgress()
    Processing = 0
    
    local RenderTime = round(curtime() - Png["start", number],2)
    
    Progress["renderTime",number] = RenderTime
    
    if (!PNGNoPrint) {
        print("PNG Finished, render time: "+RenderTime+"s")
    }
    
    # clear memory
    timer(0.001, function() {
        pngClear()
    })
}

event tick()
{  
    
    if (Png["state",string] == "" | Png["state",string] == "done" | Error) {
        exit()
    }
    
    if (DeflateInitState != "ready") {
        deflateInitTablesStep()
        exit() 
    }
    
    #print(Png["state",string], Png["decSubState", string])
    
    while(perf(95)) {
        
        if (Png["state",string] == "parse_chunks") {
            
            if (pngParseChunksStep()) {                
                if (PNGDebug) {
                    print("PNG Parsing Finished!")
                    print("Total compressed IDAT bytes: " + Png_CompressedData:length())
                }
                
                pngStartDecompress()
            }
            
        } elseif (Png["state",string] == "decompress") {            
            local Res = deflateDecompressStep(Png)            
            if (Res == 1) {
                
                if (PNGDebug) {
                    print("Deflate finished! Uncompressed bytes: " + Png["outputLength", number])
                }
                
                Png["state",string] = "filter" 
                pngStartFilter()
            } elseif (Res == -1) {
                Png["state",string] = ""
                print("Error: " + Error)
            }
        } elseif (Png["state",string] == "filter") {
            #    
            if (pngFilterStep()) {
                if (PNGDebug) {
                    print("PNG De-filtering Finished! Ready to draw.")
                }
                
                Png["state",string] = "draw"
                pngStartDraw()
            }
        } elseif (Png["state",string] == "draw") {
            #     
            if (pngDrawStep()) {     
                pngDone() 
            }
        }
    }
    
    pngUpdateProgress()
}
