@persist Png:table PNGDebug PNGNoPrint
@outputs CurrentState:string Progress:table ProgressText:string Processing:number Error:string Warning:string
@inputs DS:wirelink

@persist Png_CompressedData:string

if (first() | duped()) {

    PNGDebug = 0
    PNGNoPrint = 0
    
    dsSetScope(2)
    
    Png_CompressedData = ""
}

function pngError(Msg:string) {
    Error = Msg
    Processing = 0
    Png["state",string] = ""
    
    if (!PNGNoPrint) {
        print(Error)
    }
}
