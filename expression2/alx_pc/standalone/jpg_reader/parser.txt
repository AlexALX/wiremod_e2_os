function string jpgMarkerName(Marker:number) {
    switch (Marker) {
        case 0xFFD8,
            return "SOI"

        case 0xFFD9,
            return "EOI"

        case 0xFFC0,
            return "SOF0"

        case 0xFFC1,
            return "SOF1"

        case 0xFFC2,
            return "SOF2"

        case 0xFFC3,
            return "SOF3"

        case 0xFFC4,
            return "DHT"

        case 0xFFDB,
            return "DQT"

        case 0xFFDA,
            return "SOS"

        case 0xFFFE,
            return "COM"
            
        case 0xFFDD,
            return "DRI"

        default,
            if (Marker >= 0xFFE0 & Marker <= 0xFFEF) {
                return "APP" + (Marker - 0xFFE0)
            }

            if (Marker >= 0xFFD0 & Marker <= 0xFFD7) {
                return "RST" + (Marker - 0xFFD0)
            }

            return "UNKNOWN (0x" + Marker + ")"
    }
}

function number jpgHasLength(Marker:number)
{
    switch (Marker) {

        case 0xFFD8, # SOI
        case 0xFFD9, # EOI
            return 0

        case 0xFFD0, case 0xFFD1, case 0xFFD2, case 0xFFD3,
        case 0xFFD4, case 0xFFD5, case 0xFFD6, case 0xFFD7,
            return 0

        case 0xFF01, # TEM
            return 0
    }

    return 1
}

function jpgParseDRI()
{
    local Length = jpgU16()
    local Interval = jpgU16()

    Jpg["restartInterval",number] = Interval

    if (JPGDebug) {
        print("Restart interval =", Interval)
    }
}

function jpgBuildMCU()
{
    local MaxH = Jpg["maxH",number]
    local MaxV = Jpg["maxV",number]

    local Width = Jpg["width",number]
    local Height = Jpg["height",number]

    local MCUWidth = MaxH * 8
    local MCUHeight = MaxV * 8

    local MCUX = ceil(Width / MCUWidth)
    local MCUY = ceil(Height / MCUHeight)

    Jpg["mcuWidth",number] = MCUWidth
    Jpg["mcuHeight",number] = MCUHeight

    Jpg["mcuX",number] = MCUX
    Jpg["mcuY",number] = MCUY

    Jpg["mcuCount",number] = MCUX * MCUY

    if (JPGDebug) {
        print("--------------------------")
        print("MCU")
        print("--------------------------")
        print("MCU Size: " + MCUWidth + "x" + MCUHeight)
        print("Grid: " + MCUX + " x " + MCUY)
        print("Total MCU: " + MCUX * MCUY)   
    }
}

function number jpgAllocatePlanesStep()
{
    local State = Jpg["allocPlanes",table]

    if (State["active",number] == 0) {

        Jpg["planes",table] = table()
        Jpg["planesFull",table] = table()

        State["active",number] = 1
        State["component",number] = 1

        if (JPGDebug) {
            print("--------------------------")
            print("Planes")
            print("--------------------------")
        }
    }

    local Components = Jpg["components",table]
    
    local ID = State["component",number]
    
    if (ID > Jpg["componentCount",number]) {   
        State["active",number] = 0
        return 1
    }
    
    local Component = Components[ID,table]

    if (!Jpg["planes",table]:exists(ID)) {

        local Width = Jpg["width",number]
        local Height = Jpg["height",number]

        local MaxH = Jpg["maxH",number]
        local MaxV = Jpg["maxV",number]

        local H = Component["h",number]
        local V = Component["v",number]

        Component["width",number] = ceil(Width * H / MaxH)
        Component["height",number] = ceil(Height * V / MaxV)
        
        Jpg["planeWidth",table][ID,number] =
            Component["width",number]
        
        Jpg["planeHeight",table][ID,number] =
            Component["height",number]

        Jpg["planes",table][ID,table] = table()

        if (JPGDebug) {
            print(
                "Component " + ID +
                " = " +
                Component["width",number] +
                "x" +
                Component["height",number]
            )
        }
    }
     
    State["component",number] = ID + 1

    return 0
}

function jpgParseSOF(Type:number)
{
    local Length = jpgU16()
    local Precision = jpgU8()
    local Height = jpgU16()
    local Width = jpgU16()
    
    local WH = Width > Height ? Width : Height
    
    if (WH>512) {
        jpgError("Your image are too big\nMaximum resolution is 512x512\nYour image: "+Width+"x"+Height)
        return -1
    }
    
    Jpg["progressive",number] = (Type == 2)
    if (!Jpg["progressive",number]) {
        Jpg["progressivePreview",number] = 0
    }
    
    if (!JPGNoPrint) {
        print(Jpg["progressive",number] ? "Decoding Progressive JPEG" : "Decoding Baseline JPEG")    
    }

    local Components = jpgU8()
    
    local ComponentTable = Jpg["components",table]
    
    local MaxH = 0
    local MaxV = 0
    
    Jpg["frameComponents",table] = table()

    for (I = 1, Components) {
    
        local ID = jpgU8()
    
        local Sampling = jpgU8()
    
        local QT = jpgU8()
    
        local H = Sampling >> 4
        local V = Sampling && 15
    
        local Component = table()
    
        Component["id",number] = ID
        Component["h",number] = H
        Component["v",number] = V
        Component["qt",number] = QT
    
        ComponentTable[ID,table] = Component
    
        if (H > MaxH) {
            MaxH = H
        }
    
        if (V > MaxV) {
            MaxV = V
        }
    
        if (JPGDebug) {
            print(
                "Component " + I +
                "  ID=" + ID +
                "  H=" + H +
                "  V=" + V +
                "  QT=" + QT
            )
        }
        
        Jpg["frameComponents",table][I,table] = Component
    }
    
    Jpg["maxH",number] = MaxH
    Jpg["maxV",number] = MaxV
    
    Jpg["width",number] = Width
    Jpg["height",number] = Height
    
    Jpg["componentCount",number] = Components
    
    jpgBuildMCU()
    
    Jpg["allocBlocks",table] = table()
    Jpg["state",string] = "alloc_blocks"
    
    return 1
}

function afterAllocateBlocks() {
    Jpg["blockIndex",table] = table()  
    Jpg["state",string] = "parse_scan"   
}

function number jpgParseSOSStep()
{
    switch (Jpg["scanState",number]) {

        case 0,

            Jpg["scanLength",number] = jpgU16()

            local Components = jpgU8()

            Jpg["scanComponentsCount",number] = Components
            Jpg["scanComponentIndex",number] = 1
            Jpg["scanComponents",table] = table()
            Jpg["scanIsSingleComponent",number] = (Components == 1)
            
            if (Jpg["firstScanOffset", number] == 0) {
                Jpg["firstScanOffset", number] = jpgTell()
            }

            if (JPGDebug) {
                print("--------------------------------")
                print("Start Of Scan")
                print("--------------------------------")
                print("Components: " + Components)
            }

            Jpg["scanState",number] = 1
            Jpg["scanEOBRun",number] = 0
            break

        case 1,

            if (Jpg["scanComponentIndex",number] <= Jpg["scanComponentsCount",number]) {

                local ID = jpgU8()
                local Tables = jpgU8()

                local Component = table()

                Component["id",number] = ID
                Component["dc",number] = Tables >> 4
                Component["ac",number] = Tables && 15

                Jpg["scanComponents",table][Jpg["scanComponentIndex",number],table] = Component

                if (JPGDebug) {
                    print(
                        "Component ID=" + ID +
                        " DC=" + Component["dc",number] +
                        " AC=" + Component["ac",number]
                    )
                }

                Jpg["scanComponentIndex",number] = Jpg["scanComponentIndex",number] + 1
                break
            }

            Jpg["scanState",number] = 2
            break

        case 2,

            local Ss = jpgU8()
            local Se = jpgU8()
            local AhAl = jpgU8()

            Jpg["scanSs",number] = Ss
            Jpg["scanSe",number] = Se
            Jpg["scanAh",number] = AhAl >> 4
            Jpg["scanAl",number] = AhAl && 15
            Jpg["scanEOBRun",number] = 0
            Jpg["scanStart",number] = jpgTell()

            if (JPGDebug) {
                print(
                    "Spectral: Ss=" + Ss +
                    " Se=" + Se +
                    " Ah=" + Jpg["scanAh",number] +
                    " Al=" + Jpg["scanAl",number]
                )

                print("Scan Mode: " + jpgScanMode())
            }

            Jpg["scanMCU",number] = 0
            Jpg["restartPending",number] = 0
            Jpg["scanState",number] = 0

            local ScanCount = Jpg["scanComponentsCount",number]
    
            if (ScanCount == 1) {

                local Scan = Jpg["scanComponents",table][1,table]
                local ID = Scan["id",number]

                local Component = Jpg["components",table][ID,table]

                Jpg["scanMCUCount",number] =
                    Jpg["mcuCount",number] *
                    Component["h",number] *
                    Component["v",number]

            } else {
                Jpg["scanMCUCount",number] = Jpg["mcuCount",number]
            }
            
            if (JPGDebug) {
                print(
                    "Scan MCU count: " +
                    Jpg["scanMCUCount",number]
                )
            }

            return 1
            break
    }

    return 0
}


function jpgSaveScan()
{
    local Count = Jpg["scanSavedCount",number] + 1
    Jpg["scanSavedCount",number] = Count

    local Scan = table()

    Scan["pos",number] = Jpg["scanStart",number]
    Scan["size",number] = Jpg["scanBytes",number]

    Scan["components",number] = Jpg["scanComponentsCount",number]

    Scan["Ss",number] = Jpg["scanSs",number]
    Scan["Se",number] = Jpg["scanSe",number]
    Scan["Ah",number] = Jpg["scanAh",number]
    Scan["Al",number] = Jpg["scanAl",number]

    local Scans = Jpg["scans",table]
    Scans[Count,table] = Scan
}

function table jpgNormalizeQT(QT:table)
{
    local Result = table()

    for (I = 0, 63) {

        local Index = JpgZigZag[I + 1,number]

        Result[Index,number] = QT[I,number]
    }

    Result["precision",number] = QT["precision",number]

    return Result
}

function jpgParseDQT()
{
    local End = jpgTell() + jpgU16() - 2

    local DQT = Jpg["dqt",table]

    while (jpgTell() < End) {

        local Info = jpgU8()

        local Precision = Info >> 4
        local ID = Info && 15

        local QT = table()

        QT["precision",number] = Precision

        if (Precision == 0) {

            for (I = 0, 63) {
                QT[I,number] = jpgU8()
            }

        } else {

            for (I = 0, 63) {
                QT[I,number] = jpgU16()
            }

        }

        DQT[ID,table] = jpgNormalizeQT(QT)

        #print("DQT #" + ID + " (" + (Precision == 0 ? "8" : "16") + " bit)")
    }
}

function number jpgParseDHT()
{
    if (Jpg["DHTactive",number]==0) {
        local Length = jpgU16()
        
        if (JPGDebug) {
            print("--------------------------------")
            print("DHT")
            print("--------------------------------")
            print("DHT length = " + Length)
        }
        
        local End = jpgTell() + Length - 2
    
        local DHT = Jpg["dht",table]
        
        Jpg["dhtLength",number] = Length
        Jpg["dhtEnd",number] = End
        Jpg["DHTactive",number] = 1
    }
    
    local Length = Jpg["dhtLength",number]
    
    local End = Jpg["dhtEnd",number]

    local DHT = Jpg["dht",table]

    while (jpgTell() < End) {
        if (!perf(90)) {         
            return 0    
        }
        
        local Info = jpgU8()

        local Class = Info >> 4
        local ID = Info && 15

        local HT = table()

        HT["class",number] = Class

        local Total = 0

        local Lengths = table()

        for (I = 1, 16) {

            local Count = jpgU8()

            Lengths[I,number] = Count

            Total = Total + Count
        }
        
        HT["lengths",table] = Lengths

        local Symbols = table()

        for (I = 0, Total - 1) {
            Symbols[I,number] = jpgU8()
        }

        HT["symbols",table] = Symbols
        
        local Key = Class * 16 + ID

        DHT[Key,table] = HT
        
        jpgBuildHuffman(HT)

        if (JPGDebug) {
            if (Class == 0) {
                print("DHT DC #" + ID)
            } else {
                print("DHT AC #" + ID)
            }
    
            print(
                "Remaining = " + (End - jpgTell())
            )
        }
    }
    
    return 1
}

function number jpgParse()
{
    while (perf(90)) {      
        switch (Jpg["parseState",number]) {

            #
            # Read next marker
            #
            case 0,
                if (jpgEOF()) {
                    return 1
                }

                Jpg["markerPos",number] = jpgTell()
                Jpg["marker",number] = jpgU16()

                if (JPGDebug) {
                    print("[" + Jpg["markerPos",number] + "] " + jpgMarkerName(Jpg["marker",number]))
                }

                Jpg["parseState",number] = 1
                break

            #
            # Process marker
            #
            case 1,
    
                local Marker = Jpg["marker",number]

                switch (Marker) {

                    case 0xFFC0,
                    case 0xFFC2,
                    
                        local Result = jpgParseSOF(Marker==0xFFC0 ? 0 : 2)
                    
                        if (Result == -1) {
                            return 0
                        }
                    
                        if (Result == 1) {
                            Jpg["parseState",number] = 0
                            return 0
                        }
                    
                        break

                    case 0xFFDB,
                        jpgParseDQT()
                        break

                    case 0xFFC4,
                        if (jpgParseDHT()) {              
                            Jpg["DHTactive",number] = 0  
                        } else {
                            return 0   
                        }
                        break

                    case 0xFFDD,
                        jpgParseDRI()
                        break

                    case 0xFFDA,
                        if (jpgParseSOSStep()) {
                            jpgBitInit()
                            jpgResetScanState()
                            
                            if (Jpg["scanCount",number] == 0) {
                                Jpg["state",string] = "alloc_planes"
                            } else {
                                Jpg["state",string] = "scan"
                            }
                            
                            Jpg["scanCount",number] = Jpg["scanCount",number] + 1    
                            Jpg["parseState",number] = 0
                        }
                        return 0
                        break
                        
                    case 0xFFD9,
                    
                        if (Jpg["progressive",number]) {
                            Jpg["state",string] = "progressive_build"
                        } else {
                            Jpg["state",string] = "done"
                        }
    
                        Jpg["parseState",number] = 0
                    
                        return 1
                        break
                        
                    default,

                        if (jpgHasLength(Marker)) {

                            local Length = jpgU16()

                            jpgSkip(Length - 2)
                        }

                        break
                }

                Jpg["marker",number] = 0
                Jpg["parseState",number] = 0
                break
        }
    }

    return 0
}
