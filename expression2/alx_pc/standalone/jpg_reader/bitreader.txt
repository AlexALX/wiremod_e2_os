#include "alx_pc/standalone/jpg_reader/_shared_"
#include "alx_pc/standalone/jpg_reader/reader"

# ----------------------------------------------------
# JPEG Bit Reader
# ----------------------------------------------------

function jpgBitInit()
{
    Jpg["bitBuffer",number] = 0
    Jpg["bitCount",number] = 0
}

function number jpgReadByte()
{
    local Byte = jpgU8()

    if (Byte != 0xFF) {
        return Byte
    }

    local Next = jpgU8()

    # Byte stuffing (FF 00 -> FF)
    if (Next == 0x00) {
        return 0xFF
    }

    # Restart markers
    if (Next >= 0xD0 & Next <= 0xD7) {
        Jpg["restartMarker",number] = Next
        return -2
    }

    # End of entropy stream
    Jpg["earlyMarker",number] = 0xFF00 || Next

    return -1
}

function number jpgBit()
{
    local Buffer = Jpg["bitBuffer",number]
    local Count  = Jpg["bitCount",number]

    if (Count == 0) {

        local Pos = jpgTell()
        local Byte = jpgReadByte()

        if (Byte < 0) {
            return Byte
        }

        Buffer = Byte
        Count = 8
    }

    Count = Count - 1

    Jpg["bitCount",number] = Count
    Jpg["bitBuffer",number] = Buffer

    return (Buffer >> Count) && 1
}

function number jpgBits(Bits:number)
{
    local Value = 0
    
    local Buffer = Jpg["bitBuffer",number]
    local Count  = Jpg["bitCount",number]

    for (I = 1, Bits) {
        if (Count == 0) {
            local Byte = jpgReadByte()
            if (Byte < 0) {
                Jpg["bitCount",number] = 0
                Jpg["bitBuffer",number] = Buffer
                return Byte
            }
            Buffer = Byte
            Count = 8
        }

        Count = Count - 1

        local Bit = (Buffer >> Count) && 1
        Value = (Value << 1) || Bit
    }

    Jpg["bitCount",number] = Count
    Jpg["bitBuffer",number] = Buffer

    return Value
}

function jpgAlignByte()
{
    Jpg["bitBuffer",number] = 0
    Jpg["bitCount",number] = 0
}

function jpgDebugBytes(Pos:number, Count:number)
{
    local Save = Jpg["pos",number]

    Jpg["pos",number] = Pos

    print("Bytes DUMP", Pos, Count)

    for (I = 0, Count - 1) {

        local Byte = jpgReadByte()
        
        print(
            Pos + I,
            Byte
        )
    }

    Jpg["pos",number] = Save
}

function jpgDebugRaw(Start:number, Count:number)
{
    jpgSeek(Start)

    for (I = 0, Count - 1) {
        print(Start + I, jpgU8())
    }
}
