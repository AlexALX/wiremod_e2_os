@persist JpgIDCT:table JPG_INV_SQRT2:number JPG_C1:number JPG_C2:number JPG_C3:number JPG_C5:number JPG_C6:number JPG_C7:number

#include "alx_pc/standalone/jpg_reader/_shared_"

if (first() | duped()) {
    JpgIDCT = table()
    
    JPG_INV_SQRT2 = 1.0 / sqrt(2.0)
    JPG_C1 = cosr(1 * _PI / 16)
    JPG_C2 = cosr(2 * _PI / 16)
    JPG_C3 = cosr(3 * _PI / 16)
    JPG_C5 = cosr(5 * _PI / 16)
    JPG_C6 = cosr(6 * _PI / 16)
    JPG_C7 = cosr(7 * _PI / 16)
}

function number jpgIDCTStep(Block:table)
{
    local State = Jpg["idct",table]

    #
    # INIT
    #
    if (State["active",number] == 0) {
        State["active",number] = 1
        State["stage",number] = 0
        State["index",number] = 0
        State["temp",table] = table()
        
        # Fast path for solid color
        local IsSolidColor = 1
        for (I = 1, 63) { 
            if (abs(Block[I,number]) > 0.0001) {
                IsSolidColor = 0
                break
            }
        }
        
        if (IsSolidColor) {    
            local DCValue = (Block[0,number] / 8) + Jpg["levelShift",number]
            if (Jpg["outputShift",number] > 0) {
                DCValue = floor(DCValue / Jpg["outputShift",number])  
            }
            DCValue = clamp(DCValue, 0, 255)

            for (I = 0, 63) {
                Block[I,number] = DCValue
            }
            return 1
        }
    }

    local Temp = State["temp",table]

    #
    # STAGE 0: ROW PASS (Inlined Loeffler - Zero Garbage Allocations)
    #
    if (State["stage",number] == 0) {
        local Y = State["index",number]
        local O = Y * 8

        # AC Zero check optimization
        if (Block[O + 1, number] == 0 & Block[O + 2, number] == 0 & Block[O + 3, number] == 0 & Block[O + 4, number] == 0 & Block[O + 5, number] == 0 & Block[O + 6, number] == 0 & Block[O + 7, number] == 0) {
            local DC = Block[O, number] * JPG_INV_SQRT2
            Temp[O+0, number] = DC, Temp[O+1, number] = DC
            Temp[O+2, number] = DC, Temp[O+3, number] = DC
            Temp[O+4, number] = DC, Temp[O+5, number] = DC
            Temp[O+6, number] = DC, Temp[O+7, number] = DC
        } else {
            # Direct scalar readings
            local X0 = Block[O+0, number], local X1 = Block[O+1, number]
            local X2 = Block[O+2, number], local X3 = Block[O+3, number]
            local X4 = Block[O+4, number], local X5 = Block[O+5, number]
            local X6 = Block[O+6, number], local X7 = Block[O+7, number]

            # Even stage
            local A0 = (X0 + X4) * JPG_INV_SQRT2
            local A1 = (X0 - X4) * JPG_INV_SQRT2
            local A2 = X2 * JPG_C2 + X6 * JPG_C6
            local A3 = X2 * JPG_C6 - X6 * JPG_C2

            local Even0 = A0 + A2, local Even1 = A1 + A3
            local Even2 = A1 - A3, local Even3 = A0 - A2

            # Odd stage
            local Odd3 = X1 * JPG_C1 + X3 * JPG_C3 + X5 * JPG_C5 + X7 * JPG_C7
            local Odd2 = X1 * JPG_C3 - X3 * JPG_C7 - X5 * JPG_C1 - X7 * JPG_C5
            local Odd1 = X1 * JPG_C5 - X3 * JPG_C1 + X5 * JPG_C7 + X7 * JPG_C3
            local Odd0 = X1 * JPG_C7 - X3 * JPG_C5 + X5 * JPG_C3 - X7 * JPG_C1

            # Output directly to Temp
            Temp[O+0, number] = Even0 + Odd3
            Temp[O+1, number] = Even1 + Odd2
            Temp[O+2, number] = Even2 + Odd1
            Temp[O+3, number] = Even3 + Odd0
            Temp[O+4, number] = Even3 - Odd0
            Temp[O+5, number] = Even2 - Odd1
            Temp[O+6, number] = Even1 - Odd2
            Temp[O+7, number] = Even0 - Odd3
        }

        State["index",number] = Y + 1
        if (State["index",number] >= 8) {
            State["stage",number] = 1
            State["index",number] = 0
        }
        return 0
    }

    #
    # STAGE 1: COLUMN PASS (Inlined Loeffler - Zero Garbage Allocations)
    #
    local X = State["index",number]
    local OutShift = Jpg["outputShift",number]
    local LvlShift = Jpg["levelShift",number]

    # AC Zero check optimization
    if (Temp[X+8, number] == 0 & Temp[X+16, number] == 0 & Temp[X+24, number] == 0 & Temp[X+32, number] == 0 & Temp[X+40, number] == 0 & Temp[X+48, number] == 0 & Temp[X+56, number] == 0) {
        local DC = Temp[X, number] * JPG_INV_SQRT2 * 0.25 + LvlShift
        if (OutShift > 0) { DC = floor(DC / OutShift) }
        local Value = clamp(floor(DC), 0, 255)

        Block[X+0,  number] = Value, Block[X+8,  number] = Value
        Block[X+16, number] = Value, Block[X+24, number] = Value
        Block[X+32, number] = Value, Block[X+40, number] = Value
        Block[X+48, number] = Value, Block[X+56, number] = Value
    } else {
        local X0 = Temp[X+0,  number], local X1 = Temp[X+8,  number]
        local X2 = Temp[X+16, number], local X3 = Temp[X+24, number]
        local X4 = Temp[X+32, number], local X5 = Temp[X+40, number]
        local X6 = Temp[X+48, number], local X7 = Temp[X+56, number]

        local A0 = (X0 + X4) * JPG_INV_SQRT2
        local A1 = (X0 - X4) * JPG_INV_SQRT2
        local A2 = X2 * JPG_C2 + X6 * JPG_C6
        local A3 = X2 * JPG_C6 - X6 * JPG_C2

        local Even0 = A0 + A2, local Even1 = A1 + A3
        local Even2 = A1 - A3, local Even3 = A0 - A2

        local Odd3 = X1 * JPG_C1 + X3 * JPG_C3 + X5 * JPG_C5 + X7 * JPG_C7
        local Odd2 = X1 * JPG_C3 - X3 * JPG_C7 - X5 * JPG_C1 - X7 * JPG_C5
        local Odd1 = X1 * JPG_C5 - X3 * JPG_C1 + X5 * JPG_C7 + X7 * JPG_C3
        local Odd0 = X1 * JPG_C7 - X3 * JPG_C5 + X5 * JPG_C3 - X7 * JPG_C1

        local V0 = (Even0 + Odd3) * 0.25 + LvlShift
        local V1 = (Even1 + Odd2) * 0.25 + LvlShift
        local V2 = (Even2 + Odd1) * 0.25 + LvlShift
        local V3 = (Even3 + Odd0) * 0.25 + LvlShift
        local V4 = (Even3 - Odd0) * 0.25 + LvlShift
        local V5 = (Even2 - Odd1) * 0.25 + LvlShift
        local V6 = (Even1 - Odd2) * 0.25 + LvlShift
        local V7 = (Even0 - Odd3) * 0.25 + LvlShift

        if (OutShift > 0) {
            V0 = floor(V0 / OutShift), V1 = floor(V1 / OutShift)
            V2 = floor(V2 / OutShift), V3 = floor(V3 / OutShift)
            V4 = floor(V4 / OutShift), V5 = floor(V5 / OutShift)
            V6 = floor(V6 / OutShift), V7 = floor(V7 / OutShift)
        }

        Block[X+0,  number] = clamp(floor(V0), 0, 255)
        Block[X+8,  number] = clamp(floor(V1), 0, 255)
        Block[X+16, number] = clamp(floor(V2), 0, 255)
        Block[X+24, number] = clamp(floor(V3), 0, 255)
        Block[X+32, number] = clamp(floor(V4), 0, 255)
        Block[X+40, number] = clamp(floor(V5), 0, 255)
        Block[X+48, number] = clamp(floor(V6), 0, 255)
        Block[X+56, number] = clamp(floor(V7), 0, 255)
    }

    State["index",number] = X + 1
    if (State["index",number] >= 8) {
        State["active",number] = 0
        State["temp",table] = table()
        return 1
    }

    return 0
}
