#include "alx_pc/standalone/jpg_reader/_shared_"
#include "alx_pc/standalone/jpg_reader/reader"

function jpgProgress(Task:string, Current:number, Max:number)
{
    Progress["progressTask",string] = Task
    Progress["progressCurrent",number] = Current
    Progress["progressMax",number] = Max
    
    local Percent = 0
    
    ProgressText = Progress["progressTask",string]
    
    if (Max > 0) {
        Percent = floor(Current * 100 / Max)
    
        ProgressText += 
            " - " +
            Progress["progressCurrent",number] +
            "/" +
            Progress["progressMax",number] +
            " ("+Percent + "%)"
    }
    
    Progress["progressPercent",number] = Percent
}


function jpgUpdateProgress()
{
    CurrentState = Jpg["state",string]
    
    local IsPreviewRender = Jpg["progressivePreview",number] & (Jpg["marker",number] != 0xFFD9)

    switch (CurrentState) {

        case "parse",
            Progress["step",number] = 1
            
            jpgProgress("Parsing JPEG", 0, 0)
            break

        case "alloc_blocks",
            Progress["step",number] = 2
            
            local S = Jpg["allocBlocks",table]

            jpgProgress("Allocating block "+S["component",number],
                S["component",number],
                S["componentCount",number]
            )
            break 

        case "alloc_planes",
            Progress["step",number] = 3

            local S = Jpg["allocPlanes",table]
            
            local Component = S["component",number]

            if (Component > Jpg["componentCount",number]) {
                Component = Jpg["componentCount",number]
            }

            jpgProgress(
                "Allocating plane "+Component,
                Component,
                Jpg["componentCount",number]
            )

            break
            
        case "parse_scan",
            if (!Jpg["progressive",number]) {
                Progress["step",number] = 4
                jpgProgress("Decoding scan", 0, 0)
            }
            break

        case "scan",
            Progress["step",number] = 5
            
            if (Jpg["progressive",number]) {
                local ScanCount = " - Scan "+Jpg["scanCount",number]    
                
                local FirstOffset = Jpg["firstScanOffset", number]
                
                local TotalCurrent = jpgTell() - FirstOffset
                local TotalMax = Jpg["size", number] - FirstOffset
                
                jpgProgress("Decoding JPEG"+ScanCount,
                    TotalCurrent,
                    TotalMax
                )
                
            } else {
                local PreciptionTxt = (Jpg["precision",number]!=8 ? Jpg["precision",number]+"bit " : "")
                
                jpgProgress("Decoding Baseline "+PreciptionTxt+"JPEG",
                    Jpg["scanMCU",number],
                    Jpg["scanMCUCount",number]
                )
            }
            break

        case "progressive_build",
            Progress["step",number] = (IsPreviewRender ? 5 : 6)
        
            local S = Jpg["progressiveBuild",table]
        
            if (IsPreviewRender) {
                local CompID = S["component",number]
                local CurrentCompMax = Jpg["components",table][CompID,table]["blockCount",number]
                
                # Preview mode: shows current block index inside the active component
                jpgProgress(
                    "Building progressive blocks C"+CompID,
                    S["block",number],
                    CurrentCompMax
                )
            } else {
                jpgProgress(
                    "Building progressive blocks C"+S["component",number],
                    S["totalBlocks",number],
                    Jpg["totalBlocks",number]
                )
            }
        
            break

        case "assemble",
            Progress["step",number] = (IsPreviewRender ? 5 : 7)

            local S = Jpg["assemble",table]

            jpgProgress(
                "Assembling image",
                S["processed",number],
                Jpg["totalBlocks",number]
            )
            
            break

        case "fullplanes",
            Progress["step",number] = (IsPreviewRender ? 5 : 8)

            jpgProgress(
                "Expanding planes",
                Jpg["full",table]["index",number],
                Jpg["width",number] * Jpg["height",number]
            )
            
            break

        case "draw",
            Progress["step",number] = (IsPreviewRender ? 5 : 9)

            local S = Jpg["draw",table]

            jpgProgress(
                "Drawing image",
                S["index",number],
                Jpg["width",number] * Jpg["height",number]
            )
            
            break

        case "done",
            jpgProgress("Done", 0, 0)
            break

        default,

            jpgProgress("", 0, 0)
    }
}
