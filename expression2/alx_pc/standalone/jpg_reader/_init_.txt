@name JPG Reader Main Chip

#include "alx_pc/standalone/jpg_reader/_shared_"
#include "alx_pc/standalone/jpg_reader/parser"
#include "alx_pc/standalone/jpg_reader/decoder"
#include "alx_pc/standalone/jpg_reader/draw"
#include "alx_pc/standalone/jpg_reader/progress"

function jpgInit(Data:string, ProgressivePreview:number, MaxSize:number, NoKeepAR:number)
{
    Processing = 0
    
    Jpg = table()
    Jpg["start", number] = curtime()
    Jpg["progressivePreview",number] = ProgressivePreview
    Jpg["MaxSize", number] = MaxSize
    Jpg["NoKeepAR", number] = NoKeepAR
    
    Jpg["data",string] = Data
    Jpg["size",number] = Data:length()
    
    Jpg["pos",number] = 1
    Jpg["active",number] = 0
    
    Jpg["scanCount",number] = 0
    Jpg["scanSavedCount",number] = 0
    Jpg["scanBytes",number] = 0
    Jpg["scans",table] = table()
    Jpg["state",string] = ""
    
    Jpg["dqt",table] = table()
    Jpg["dht",table] = table()
    
    Jpg["lastDC",table] = table()
    Jpg["components",table] = table()
    Jpg["scanComponents",table] = table()
    Jpg["blocks",table] = table()
    Jpg["blockIndex",table] = table()
    Jpg["coefficients",table] = table()
    
    Jpg["dirtyBlocks",table] = table()
    
    Jpg["progressiveBuild",table] = table()
    
    Jpg["decodedMCU",number] = 0
    Jpg["scanMCU",number] = 0
    Jpg["mcuDecode",table] = table()
    
    Jpg["blockDecode",table] = table()
    Jpg["idct",table] = table()
    
    Jpg["assemble",table] = table()
    Jpg["upsample",table] = table()
    Jpg["pixels",table] = table()
    Jpg["rgb",table] = table()
    
    Jpg["planes",table] = table()
    Jpg["allocPlanes",table] = table()
    Jpg["planesFull",table] = table()
    Jpg["full",table] = table()
    Jpg["allocPixels",table] = table()
    
    Jpg["planeWidth",table] = table()
    Jpg["planeHeight",table] = table()
    
    Jpg["draw",table] = table()
    
    Progress = table()
    Progress["step",number] = 1
    Progress["stepTotal",number] = 9
    
    Error = ""
    Warning = ""
    
    if (jpgU16() != 0xFFD8) {
        timer(0.001,function() {
            Error = "Not JPEG"
            if (!JPGNoPrint) {
                print(Error)    
            }
        })
        return
    }
    
    jpgUpdateProgress()
    
    # we must delay a bit or it will not trigger proper outputs
    timer(0.001,function() {
        Processing = 1
        Jpg["state",string] = "parse"    
    })
    if (!JPGNoPrint) {
        print("JPEG detected")
    }
}

# clear e2 memory
function jpgClear() {
    Jpg = table()    
}

function jpgAbort() {
    Jpg["state",string] = ""
    CurrentState = ""
    Progress = table()
    ProgressText = ""
    Error = ""
    Processing = 0
    jpgClear()
}

function jpgDone() {
    Jpg["state", string] = "done"
    
    jpgUpdateProgress()
    Processing = 0
    
    local RenderTime = round(curtime() - Jpg["start", number],2)
    
    Progress["renderTime",number] = RenderTime
    
    if (!JPGNoPrint) {
        print("JPG Finished, render time: "+RenderTime+"s")
    }
    
    # clear memory
    timer(0.001, function() {
        jpgClear()
    })
}

event tick()
{
    if (Jpg["state",string]=="" | Jpg["state",string]=="done" | Error) {
        exit()
    }
    
    while(perf(90)) {
        if (Jpg["state",string] == "parse" | Jpg["state",string] == "parse_scan") {
            jpgParse()
        } elseif (Jpg["state",string] == "alloc_blocks") {
    
            if (jpgAllocateBlocksStep()) {
                afterAllocateBlocks()      
            }
    
        } elseif (Jpg["state",string] == "alloc_planes") {
    
            if (jpgAllocatePlanesStep()) {
                Jpg["state",string] = "scan"
            }
    
        } elseif (Jpg["state",string] == "scan") {
            local Result = jpgDecodeScan()
            if (Result!=0) {
                jpgAfterDecodeScan(Result)    
            }
        } elseif (Jpg["state",string] == "progressive_build") {
    
            if (jpgProgressiveBuildBlocksStep()) {
                Jpg["state",string] = "assemble"
            }    
        } elseif (Jpg["state",string] == "assemble") {
        
            if (jpgAssembleImageStep()) {
                Jpg["state",string] = "fullplanes"
            }
        
        } elseif (Jpg["state",string] == "fullplanes") {
        
            if (jpgInitFullPlanes()) {
                Jpg["state",string] = "draw"
            }
            
        } elseif (Jpg["state",string] == "draw") {
    
            if (jpgDrawStep()) {
                if (Jpg["progressivePreview",number] & Jpg["marker",number] != 0xFFD9) {      
                    Jpg["dirtyBlocks",table] = table()      
                    Jpg["state",string] = "parse_scan"
                } else {
                    jpgDone()
                }
            }
    
        }        
    }
    
    if (Jpg["state",string] & Jpg["state",string]!="done") {
        jpgUpdateProgress()
    }
}
