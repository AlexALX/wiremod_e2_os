function number jpgExtend(Value:number, Size:number)
{
    if (Size == 0) {
        return 0
    }

    local Limit = 1 << (Size - 1)

    if (Value >= Limit) {
        return Value
    }

    return Value - ((1 << Size) - 1)
}

function number jpgDecodeDC(HT:table, Previous:number)
{
    local Category = jpgHuffmanRead(HT)

    if (Category < 0) {
        return Previous
    }

    local Diff = jpgBits(Category)

    Diff = jpgExtend(Diff, Category)

    return Previous + Diff
}

function number jpgNextMCUBlock()
{
    local State = Jpg["mcuDecode",table]
    local Components = Jpg["scanComponents",table]
    
    if (Jpg["scanIsSingleComponent",number]) {
        State["y",number] = 1
        State["x",number] = 1
        State["component",number] = 1
        State["active",number] = 0
        return 1
    }
    local Scan = Components[State["component",number],table]

    local ID = Scan["id",number]
    local Component = Jpg["components",table][ID,table]

    local H = Component["h",number]
    local V = Component["v",number]

    State["x",number] = State["x",number] + 1


    if (State["x",number] <= H) {
        return 0
    }


    State["x",number] = 1
    State["y",number] = State["y",number] + 1


    if (State["y",number] <= V) {
        return 0
    }


    #
    # component finished
    #

    State["y",number] = 1
    State["component",number] = State["component",number] + 1


    if (State["component",number] <= Components:count()) {
        return 0
    }


    #
    # MCU finished
    #

    State["component",number] = 1
    State["x",number] = 1
    State["y",number] = 1

    State["active",number] = 0

    return 1
}

function jpgDecodeBlockBegin(
    ComponentID:number,
    BlockIndex:number,
    DCHT:table,
    ACHT:table
)
{
    
    local S = Jpg["blockDecode",table]
    
    S["stage",number] = 0
    S["component",number] = ComponentID
    S["dcht",table] = DCHT
    S["acht",table] = ACHT
    S["blockIndex"] = BlockIndex

    if (Jpg["progressive",number]) {
        
        local Block = table()
        if (BlockIndex>=0) {
            local Coeff = Jpg["coefficients",table][ComponentID,table][BlockIndex,table]
    
            for (I = 0,63) {
                Block[I,number] = Coeff[I,number]
            }
        }

        S["block",table] = Block
        
        S["Ss",number] = Jpg["scanSs",number]
        S["Se",number] = Jpg["scanSe",number]
        S["Al",number] = Jpg["scanAl",number]
        S["Ah",number] = Jpg["scanAh",number]
        
    } else {
        S["block",table] = table()
    }

    S["dequantIndex",number] = 0

    S["active",number] = 1
    S["idct",number] = 0
    S["saved",number] = 0
}

function number jpgDecodeBlockBaseline()
{
    local S = Jpg["blockDecode",table]
    local Stage = S["stage",number]
    local Block = S["block",table]

    #
    # Decode DC
    #

    if (Stage == 0) {        
        local Size = jpgHuffmanRead(S["dcht",table])

        if (Size < 0) {
            return -1
        }

        local Diff = 0

        if (Size > 0) {

            Diff = jpgBits(Size)

            if (Diff < 0) {
                #print("jpgReceive -1")
                return -1
            }

            Diff = jpgExtend(Diff, Size)
        }

        local Component = S["component",number]
        
        local DC =
            Jpg["lastDC",table][Component,number] + Diff


        Jpg["lastDC",table][Component,number] = DC

        Block[0,number] = DC

        S["block",table] = Block
        S["k",number] = 1
        S["stage",number] = 1
        State = 1
    }

    #
    # Decode AC
    #

    if (S["stage",number] == 1) {
            
        local ACHT = S["acht",table]

        Block = S["block",table]

        local K = S["k",number]

        while (K < 64) {

            if (!perf(90)) {
                S["k",number] = K
                return 0
            }

            local Symbol =
                jpgHuffmanRead(ACHT)

            if (Symbol < 0) {
                #print("jpgHuffmanRead acht -1")
                return -1
            }

            #
            # EOB
            #

            if (Symbol == 0) {
                S["stage",number] = 2                
                return 1
            }

            #
            # ZRL
            #

            if (Symbol == 0xF0) {

                K += 16
                continue
            }

            local Run = Symbol >> 4
            local Size = Symbol && 15

            K += Run

            if (K >= 64) {
                break
            }

            local Value = 0

            if (Size > 0) {

                Value = jpgBits(Size)

                if (Value < 0) {
                    #print("jpgReceive jpgReceive -1")
                    return -1
                }

                Value = jpgExtend(Value, Size)
            }

            Block[JpgZigZag[K + 1,number],number] = Value

            K++
        }

        S["k",number] = K
        S["stage",number] = 2

        return 1
    }

    return 1
}

function jpgMarkBlockDirty()
{    
    if (!Jpg["progressivePreview",number]) { return }
    
    local S = Jpg["blockDecode",table]
    local Component = S["component",number]
    local BlockIndex = S["blockIndex", number]

    if (BlockIndex >= 0) {
        local Key = Component * 10000 + BlockIndex
        Jpg["dirtyBlocks",table][Key,number] = 1
    }
}

function number jpgDecodeBlockDCFirstStep()
{
    local S = Jpg["blockDecode",table]

    local Block = S["block",table]

    #
    # Decode DC
    #

    if (S["stage",number] == 0) {

        #
        # Progressive block starts empty
        #
        if (Block:count() == 0) {
            Block = table()
        }

        local Size = jpgHuffmanRead(S["dcht",table])

        if (Size < 0) {
            #print("HUFF DC", Size)
            return Size
        }


        local Diff = 0

        if (Size > 0) {

            Diff = jpgBits(Size)

            if (Diff < 0) {
                #print("DIFF DC", Diff)
                return Diff
            }

            Diff = jpgExtend(Diff, Size)
        }


        local Component = S["component",number]

        local DC =
            Jpg["lastDC",table][Component,number] + Diff


        Jpg["lastDC",table][Component,number] = DC


        #
        # Progressive shift
        #
        DC = DC << Jpg["scanAl",number]


        Block[0,number] = DC
        jpgMarkBlockDirty()

        S["block",table] = Block
        S["stage",number] = 1


        return 1
    }


    return 1
}

function number jpgDecodeBlockACFirstStep()
{
    local S = Jpg["blockDecode",table]
    local Block = S["block",table]

    local Ss = S["Ss",number]
    local Se = S["Se",number]
    local Al = S["Al",number]

    #
    # Progressive AC EOB run
    #

    local EOBRun = Jpg["scanEOBRun",number]

    if (EOBRun > 0) {
        Jpg["scanEOBRun",number] = EOBRun - 1

        S["stage",number] = 0
        S["block",table] = Block

        return 1
    }


    #
    # Initialize scan position
    #

    if (S["stage",number] == 0) {

        S["stage",number] = 1
        S["k",number] = Ss
    }


    local K = S["k",number]


    while (K <= Se) {


        if (!perf(90)) {
            S["k",number] = K
            return 0
        }

        local Symbol = jpgHuffmanRead(S["acht",table])
        if (Symbol < 0) {  
            S["k",number] = K       
            return Symbol
        }


        local Run = Symbol >> 4
        local Size = Symbol && 15

        #
        # EOB / EOBRUN
        #

        if (Size == 0) {

            #
            # ZRL
            #

            if (Run == 15) {
            
                K += 16
            
                if (K > Se) {
                    K = Se + 1
                    break
                }
            
                continue
            }


            #
            # EOB run
            #

            local Extra = 0

            if (Run > 0) {

                Extra = jpgBits(Run)

                if (Extra < 0) {
                    print("Extra < 0")
                    return -1
                }
            }


            local Count = (1 << Run)
            
            if (Run > 0) {
                Count += Extra
            }
        
            Jpg["scanEOBRun",number] = Count - 1

            S["stage",number] = 0
            S["k",number] = Ss
            S["block",table] = Block 
                    
            return 1
        }


        #
        # Normal coefficient
        #

        K += Run


        if (K > Se) {

            print(
                "K>",
                K,
                Se,
                "Symbol",
                Symbol
            )

            return -1
        }


        local Value = jpgBits(Size)


        if (Value < 0) {

            print("RECEIVE", Value)
            return -1
        }


        Value = jpgExtend(Value, Size)

        local BlockIndex = JpgZigZag[K+1,number] 

        Block[BlockIndex,number] = Value << Al
        jpgMarkBlockDirty()


        K++
    }


    S["k",number] = K
    S["block",table] = Block
    S["stage",number] = 2

    return 1
}


function number jpgDecodeBlockDCRefineStep()
{
    local S = Jpg["blockDecode",table]
    local Block = S["block",table]

    if (S["stage",number] == 0) {
        local Bit = jpgBit()
        if (Bit < 0) {
            return Bit
        }

        if (Bit == 1) {
            local Al = S["Al",number]
            Block[0,number] = Block[0,number] || (1 << Al)
            jpgMarkBlockDirty()
        }

        S["block",table] = Block
        S["stage",number] = 1
        return 1
    }

    return 1
}

function number jpgRefineCoefficient(
    Block:table,
    Index:number,
    Al:number
)
{
    if (Block[Index,number] == 0) {
        return 1
    }

    local Delta = 1 << Al

    if (Block[Index,number] && Delta) {
        return 1
    }

    local Bit = jpgBit()

    if (Bit < 0) {
        return Bit
    }

    if (Bit == 0) {
        return 1
    }

    if (Block[Index,number] > 0) {
        Block[Index,number] = Block[Index,number] + Delta
    } else {
        Block[Index,number] = Block[Index,number] - Delta
    }
    jpgMarkBlockDirty()

    return 1
}

function number jpgRefineRange(
    Block:table,
    Start:number,
    End:number,
    Al:number
)
{
    local K = Start
    while (K <= End) {
        local Index = JpgZigZag[K + 1,number]
        if (Block[Index,number] != 0) {
            local Result = jpgRefineCoefficient(Block, Index, Al)
            if (Result < 0) {
                return Result
            }
        }
        K++
    }
    return -1000
}

function number jpgRefineSkipRun(
    Block:table,
    Start:number,
    Run:number,
    Se:number,
    Al:number
)
{
    local K = Start
    local ZeroCount = 0
    
    while (K <= Se) {
    
        local Index = JpgZigZag[K + 1,number]
    
        if (Block[Index,number] != 0) {
    
            local Result = jpgRefineCoefficient(Block, Index, Al)
            if (Result < 0) {
                return Result
            }
    
        } else {
            if (Run == 0) {
                return K
            }
            Run--
        }
    
        K++
    }
    
    if (ZeroCount != Run) {
        return -2
    }
    
    return K
}

function number jpgDecodeBlockACRefineStep()
{
    local S = Jpg["blockDecode",table]
    local Block = S["block",table]

    local Ss = S["Ss",number]
    local Se = S["Se",number]
    local Al = S["Al",number]
    local ZigZag = JpgZigZag

    #
    # Existing EOB run
    #

    local EOBRun = Jpg["scanEOBRun",number]

    if (EOBRun > 0) {

        local K = S["k",number]
        if (S["stage",number] != 3) {
            K = Ss
            S["stage",number] = 3
        }

        while (K <= Se) {

            if (!perf(90)) {
                S["k",number] = K
                return 0
            }

            local Index = ZigZag[K + 1,number]

            if (Block[Index,number] != 0) {

                local Res = jpgRefineCoefficient(Block, Index, Al)
                if (Res < 0) {
                    return Res
                }
            }

            K++
        }

        Jpg["scanEOBRun",number] = EOBRun - 1

        S["stage",number] = 0
        S["block",table] = Block

        return 1
    }

    #
    # Init
    #

    local CurStage = S["stage",number]
    local K = S["k",number]

    if (CurStage == 0) {
        CurStage = 1
        K = Ss
    }
    
    local Run = S["symbolRun",number]
    local Size = S["symbolSize",number]
    local SymbolVal = S["symbolValue",number]

    while (K <= Se) {

        if (!perf(90)) {
            S["k",number] = K
            S["stage",number] = CurStage
            S["symbolRun",number] = Run
            return 0
        }

        # Stage 1: Read Huffman Symbol
        if (CurStage == 1) {

            local Symbol = jpgHuffmanRead(S["acht",table])
            if (Symbol < 0) {
                S["k",number] = K
                S["stage",number] = CurStage
                S["symbolRun",number] = Run
                return Symbol
            }

            Run = Symbol >> 4
            Size = Symbol && 15
            S["symbolSize",number] = Size

            if (Size == 0) {

                if (Run == 15) {
                    CurStage = 5
                } else {
                    local Extra = 0
                    if (Run > 0) {
                        Extra = jpgBits(Run)
                        if (Extra < 0) {
                            return Extra
                        }
                    }
                    local Count = (1 << Run)
                    if (Run > 0) {
                        Count += Extra
                    }
                    S["eobCount",number] = Count
                    CurStage = 4
                }

            } else {

                local Value = jpgBits(Size)
                if (Value < 0) {
                    return Value
                }
                SymbolVal = jpgExtend(Value, Size)
                S["symbolValue",number] = SymbolVal
                CurStage = 5
            }
        }

        local Index = ZigZag[K + 1,number]

        # Stage 4: Process EOBRun step
        if (CurStage == 4) {

            if (Block[Index,number] != 0) {
                local Res = jpgRefineCoefficient(Block, Index, Al)
                if (Res < 0) {
                    return Res
                }
            }

            K++

            if (K > Se) {
                Jpg["scanEOBRun",number] = S["eobCount",number] - 1
                S["stage",number] = 0
                S["k",number] = 0 #     
                S["block",table] = Block
                return 1
            }

            continue
        }

        # Stage 5: Process ZRL or New Coefficient step
        if (CurStage == 5) {

            if (Size == 0) { # ZRL Case (Run == 15)

                if (Block[Index,number] != 0) {
                    local Res = jpgRefineCoefficient(Block, Index, Al)
                    if (Res < 0) {
                        return Res
                    }
                } else {
                    Run--
                    if (Run < 0) {
                        CurStage = 1
                    }
                }

                K++
                continue

            } else { # New Coefficient Case

                if (Block[Index,number] != 0) {
                    local Res = jpgRefineCoefficient(Block, Index, Al)
                    if (Res < 0) {
                        return Res
                    }
                    K++
                } else {
                    if (Run == 0) {
                        Block[Index,number] = SymbolVal << Al
                        CurStage = 1
                        K++
                    } else {
                        Run--
                        K++
                    }
                }
                continue
            }
        }
    }

    S["k",number] = 0 #     
    S["stage",number] = 0
    S["symbolRun",number] = Run
    S["block",table] = Block

    return 1
}


function number jpgDecodeBlockStep() {
    local Mode = jpgScanMode()
    
    if (Mode == 0) {
        return jpgDecodeBlockBaseline()
    }
    
    #print("Enter Mode",Mode)
    
    switch (Mode) {
        case 1,
            return jpgDecodeBlockDCFirstStep()
        break
    
        case 2,
            return jpgDecodeBlockACFirstStep()
        break
    
        case 3,
            return jpgDecodeBlockDCRefineStep()
        break
    
        case 4,
            return jpgDecodeBlockACRefineStep()
        break
    } 
    
    print("Unknown Mode", Mode)

    return 1
}

function number jpgAllocateBlocksStep()
{
    local State = Jpg["allocBlocks",table]

    switch (State["stage",number]) {

        #
        # Initialize
        #
        case 0,

            State["blocks",table] = table()
            State["coefficients",table] = table()
            State["component",number] = 0
            State["stage",number] = 1

            Jpg["totalBlocks",number] = 0

            break

        #
        # Prepare next component
        #
        case 1,

            State["component",number] = State["component",number] + 1

            if (State["component",number] > Jpg["componentCount",number]) {
                Jpg["blocks",table] = State["blocks",table]
                Jpg["coefficients",table] = State["coefficients",table]
                Jpg["allocBlocks",table] = table()

                return 1
            }

            local Width = Jpg["width",number]
            local Height = Jpg["height",number]

            local MaxH = Jpg["maxH",number]
            local MaxV = Jpg["maxV",number]

            local Components = Jpg["components",table]
            local C = Components[State["component",number],table]

            local H = C["h",number]
            local V = C["v",number]

            local BlocksX = ceil(Width * H / (8 * MaxH))
            local BlocksY = ceil(Height * V / (8 * MaxV))
            local Count = BlocksX * BlocksY

            C["blocksX",number] = BlocksX
            C["blocksY",number] = BlocksY
            C["blockCount",number] = Count

            if (JPGDebug) {
                print(
                    "Component " + State["component",number] +
                    " blocks = " +
                    BlocksX + " x " +
                    BlocksY + " = " +
                    Count
                )
            }

            State["storage",table] = table()
            State["coeffStorage",table] = table()
            State["count",number] = Count
            State["index",number] = 0
            
            State["blocks",table][State["component",number],table] = table()
            State["coefficients",table][State["component",number],table] = table()

            Jpg["totalBlocks",number] = Jpg["totalBlocks",number] + Count

            break

    }

    return 0
}

function number jpgDequantizeBlockStep(Block:table, QT:table, State:table)
{
    local I = State["dequantIndex",number]

    while (I < 64) {

        if (!perf(90)) {
            State["dequantIndex",number] = I
            return 0
        }
        
        local Val = Block[I,number]
        
        if (Val!=0) {
            Block[I,number] = Val * QT[I,number]
        }

        I++
    }

    State["dequantIndex",number] = 0

    return 1
}

function number jpgDecodeMCU()
{
    local State = Jpg["mcuDecode",table]

    if (State["active",number] == 0) {
        jpgResetScanState()
    }

    local Components = Jpg["scanComponents",table]
    local C = State["component",number]
    local CompCount = Components:count()  

    if (C > CompCount) {
        State["active", number] = 0
        return 1
    }

    local Scan = Components[C,table]
    local ID = Scan["id",number]
    local Component = Jpg["components",table][ID,table]
        
    local Index = -1
    local IsSingle = Jpg["scanIsSingleComponent",number] 
    
    if (IsSingle) {
        local BlocksX = Component["blocksX",number]
        local ScanMCU = Jpg["scanMCU",number]
        local BlockX = ScanMCU % BlocksX
        local BlockY = floor(ScanMCU / BlocksX)
        
        local H = Component["h",number]
        local V = Component["v",number]
        
        local TargetMCU = floor(BlockY / V) * Jpg["mcuX",number] + floor(BlockX / H)
        local InternalBlock = (BlockY % V) * H + (BlockX % H)
        
        Index = jpgComponentBlockIndex(Component, TargetMCU, InternalBlock)
    } else {
        local InternalBlockOffset = (State["y",number] - 1) * Component["h",number] + (State["x",number] - 1)
        Index = jpgComponentBlockIndex(Component, Jpg["scanMCU",number], InternalBlockOffset)
    }

    local Decode = Jpg["blockDecode",table]
    local CurStage = State["stage",number]

    if (CurStage == 0) {
        local DCHT = Jpg["dht",table][Scan["dc",number],table]
        local ACHT = Jpg["dht",table][16 + Scan["ac",number],table]
        
        jpgDecodeBlockBegin(ID, Index, DCHT, ACHT)
        State["stage",number] = 1
        CurStage = 1
    }

    if (CurStage == 1) {
        local Result = jpgDecodeBlockStep()

        if (Result <= 0) {
            return Result
        }
    
        if (Jpg["progressive",number]) {
            if (Index >= 0) {
                local Blocks = Jpg["coefficients",table][ID,table]
                local Coeff = Blocks[Index,table]
                
                if (!Coeff:count()) {
                    Blocks[Index,table] = table()
                    Coeff = Blocks[Index,table]
                }
                
                local Block = Decode["block",table]
                
                for (I = 0, 63) {
                    Coeff[I,number] = Block[I,number]
                }
            }
        
            State["stage",number] = 0
            return jpgNextMCUBlock()
        }

        State["stage",number] = 2
        CurStage = 2
    }

    if (CurStage == 2) {
        local QT = Jpg["dqt",table][Component["qt",number],table]

        if (!jpgDequantizeBlockStep(Decode["block",table], QT, State)) {
            return 0
        }

        State["stage",number] = 3
        CurStage = 3
    }

    if (CurStage == 3) {
        if (!jpgIDCTStep(Decode["block",table])) {
            return 0
        }
        State["stage",number] = 4
        CurStage = 4
    }

    if (CurStage == 4) {
        local Storage = Jpg["blocks",table][ID,table]
        
        if (Index >= 0) {
            local BlockTable = Decode["block",table]
            
            BlockTable["x",number] = State["x",number]
            BlockTable["y",number] = State["y",number]
            BlockTable["mcu",number] = Jpg["scanMCU",number]
            
            Storage[Index,table] = BlockTable
            Decode["block",table] = table()
        }

        State["stage",number] = 0
        return jpgNextMCUBlock()
    }

    return 0
}

function number jpgConsumeRestart()
{
    local Byte = jpgReadByte()

    if (Byte != -2) {
        return 0
    }

    jpgAlignByte()

    for (I = 1, Jpg["componentCount",number]) {
        Jpg["lastDC",table][I,number] = 0
    }

    Jpg["scanEOBRun",number] = 0

    return 1
}

function number jpgDecodeScan()
{
    while (perf(90)) {        
        if (Jpg["restartPending",number]) {
        
            if (!jpgConsumeRestart()) {
                return 0
            }
        
            Jpg["restartPending",number] = 0
        }
                
        local Result = jpgDecodeMCU()
        
        if (Result==-100) {
            error("JPG ERROR?")    
        }
        
        if (Result == -1) {
            jpgAlignByte()
            return -1
        }
                        
        if (Result>0) {
                        
            Jpg["scanMCU",number] = Jpg["scanMCU",number] + 1
            Jpg["decodedMCU",number] = Jpg["decodedMCU",number] + 1
            
            if (Jpg["restartInterval",number] > 0 & Jpg["scanMCU",number] % Jpg["restartInterval",number] == 0 ) {
                Jpg["restartPending",number] = 1
            }

            if (Jpg["scanMCU",number] >= Jpg["scanMCUCount",number]) {         
                
                Jpg["scanEOBRun",number] = 0               
                jpgAlignByte()
                return 1
            }
        }
    }

    return 0
}

function jpgAfterDecodeScan(Result:number) {
    if (Result>0 | Result==-1) {                        
        if (Jpg["progressive",number]) {            
            local Preview = Jpg["progressivePreview",number]
            if (Preview) {                        
                local Ah = Jpg["scanAh",number]
                
                local IsRefineScan = (Ah > 0)
                if (IsRefineScan) {
                    Preview = 0
                } else {
                    if (Preview == 1 & Jpg["scanCount",number] != 2) {
                        Preview = 0
                    }    
                }                
            }
            
            if (Jpg["earlyMarker",number] != 0) {
                Jpg["marker",number] = Jpg["earlyMarker",number]
                Jpg["earlyMarker",number] = 0
                Jpg["parseState",number] = 1
                if (Preview) {
                    Jpg["state",string] = "progressive_build"    
                } else {
                    Jpg["state",string] = "parse_scan"  
                }                  
            } elseif (Jpg["marker",number] == 0xFFDA) {
                Jpg["parseState",number] = 0
                if (Preview) {
                    Jpg["state",string] = "progressive_build"    
                } else {
                    Jpg["state",string] = "parse_scan"          
                }          
            } else {
                Jpg["state",string] = "progressive_build"       
            }
        } else {
            Jpg["state",string] = "assemble"        
        }
        
    } elseif (Result<0) {
        if (JPGDebug) {
            print(
                "BLOCK ERROR",
                "MCU",
                Jpg["scanMCU",number],
                "COMP",
                Jpg["mcuDecode",table]["component",number],
                Result
            )
        }    
    }
}

function number jpgProgressiveBuildBlocksStep()
{
    local State = Jpg["progressiveBuild",table]

    if (!State["active",number]) {

        State["active",number] = 1
        State["component",number] = 1
        State["block",number] = 0
        State["totalBlocks",number] = 0

        State["stage",number] = 0
        State["blockData",table] = table()        
    }

    local ID = State["component",number]


    if (ID > Jpg["componentCount",number]) {
        State["active",number] = 0
        return 1
    }


    local Component =
        Jpg["components",table][ID,table]


    local Count = Component["blockCount",number]    

    local Index = State["block",number]

    local Key = ID * 10000 + Index
    
    local Storage = Jpg["blocks",table][ID,table]
    
    if (Jpg["progressivePreview",number]) {
        local Existing = Storage[Index,table]
        
        if (!Jpg["dirtyBlocks",table][Key,number] & Existing:count()) {
        
            State["stage",number] = 0
            State["blockData",table] = table()
    
            State["block",number] = Index + 1
        
            if (State["block",number] >= Count) {
                State["block",number] = 0
                State["component",number] = ID + 1
            }
        
            return 0
        }
    }

    local QT =
        Jpg["dqt",table][Component["qt",number],table]


    local Coeff =
        Jpg["coefficients",table][ID,table][Index,table]


    local Block = State["blockData",table]


    #
    # load coefficient block
    #

    if (!Block:count()) {
        Block = table()

        for (I = 0,63) {
            Block[I,number] = Coeff[I,number]
        }

        State["blockData",table] = Block

        State["dequantIndex",number] = 0
        Jpg["idct",table] = table()
    }


    #
    # dequant
    #

    if (State["stage",number] == 0) {
        if (!jpgDequantizeBlockStep(
            Block,
            QT, State
        )) {
            return 0
        }


        State["stage",number] = 1
    }


    #
    # IDCT
    #

    if (State["stage",number] == 1) {
    
        if (!jpgIDCTStep(Block)) {
            return 0
        }

        State["stage",number] = 2
    }


    #
    # save
    #

    if (State["stage",number] == 2) {
        Storage[Index,table] = Block

        #
        # next block
        #

        State["block",number] = Index + 1
        State["totalBlocks",number] = State["totalBlocks",number] + 1

        State["stage",number] = 0
        State["blockData",table] = table()


        if (State["block",number] >= Count) {

            State["block",number] = 0
            State["component",number] = ID + 1
        }
    }

    return 0
}
