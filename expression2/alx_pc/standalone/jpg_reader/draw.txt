#include "alx_pc/standalone/jpg_reader/_shared_"

function number jpgDrawStep()
{
    local State = Jpg["draw",table]

    local Width = Jpg["width",number]
    local Height = Jpg["height",number]
    
    local NoKeepAR = Jpg["NoKeepAR",number]

    if (State["active",number] == 0) {
        State["active",number] = 1
        State["index",number] = 0
        
        if (!Jpg["DS_Inited",number]) {
            
            local ScreenWidth = Jpg["screenWidth",number]
            local ScreenHeight = Jpg["screenHeight",number]
            
            Jpg["DS_Inited",number] = 1
            
            Progress["progressDrew",number] = 1
            
            DS[1048569] = 2
            DS[1048574] = 0
            DS[1048575] = 1
            
            if (!NoKeepAR) {
                local SWH = (ScreenWidth>ScreenHeight ? ScreenWidth : ScreenHeight)
                Jpg["Xadd",number] = (ScreenHeight>ScreenWidth ? floor((SWH-ScreenWidth)/2) : 0 )
                Jpg["Yadd",number] = (ScreenWidth>ScreenHeight ? floor((SWH-ScreenHeight)/2) : 0 )
                Jpg["posMul", number] = SWH  
                DS[1048573] = SWH
                DS[1048572] = SWH          
            } else {
                Jpg["Xadd",number] = 0
                Jpg["Yadd",number] = 0
                Jpg["posMul", number] = ScreenWidth
                DS[1048573] = ScreenWidth
                DS[1048572] = ScreenHeight
            }
        }
    }

    local Planes = Jpg["planesFull",table]

    local YPlane = Planes[1,table]
    local CbPlane = Planes[2,table]
    local CrPlane = Planes[3,table]

    local Count = Width * Height

    local Xadd = Jpg["Xadd",number]
    local Yadd = Jpg["Yadd",number]
    
    local PosMul = Jpg["posMul",number]
    
    local IsProgressive = Jpg["progressivePreview",number]
    local TestYCbCr = Jpg["test_YCbCrToRGB",number]
    local PixelsTable = Jpg["pixels",table]
    local Dirty = Jpg["dirtyBlocks",table]

    local CompY = Jpg["components",table][1,table]
    local CompY_BX = CompY["blocksX",number]

    local CompCb = Jpg["components",table][2,table]
    local CompCb_BX = CompCb["blocksX",number]
    local CbScaleX = CompCb["h",number] / Jpg["maxH",number]
    local CbScaleY = CompCb["v",number] / Jpg["maxV",number]

    local CompCr = Jpg["components",table][3,table]
    local CompCr_BX = CompCr["blocksX",number]
    local CrScaleX = CompCr["h",number] / Jpg["maxH",number]
    local CrScaleY = CompCr["v",number] / Jpg["maxV",number]

    local Scale = Jpg["globalScale", number]
    local InvScale = Jpg["invScale", number]
    local StepX = floor(Scale)
    if (StepX < 1) { StepX = 1 }

    local LastX = -1
    local LastY = -1

    while (perf(90)) {

        local I = State["index",number]

        if (I >= Count) {
            State["active",number] = 0
            return 1
        }
        
        local PixelX = I % Width
        local PixelY = floor(I / Width)
        
        if (IsProgressive) {
            local BlockIdx_Y = floor(PixelY >> 3) * CompY_BX + floor(PixelX >> 3)
            local KeyY = 10000 + BlockIdx_Y
            
            if (!Dirty[KeyY,number]) {
                local BlockIdx_Cb = floor((PixelY * CbScaleY) / 8) * CompCb_BX + floor((PixelX * CbScaleX) / 8)
                local KeyCb = 20000 + BlockIdx_Cb
                
                if (!Dirty[KeyCb,number]) {
                    local BlockIdx_Cr = floor((PixelY * CrScaleY) / 8) * CompCr_BX + floor((PixelX * CrScaleX) / 8)
                    local KeyCr = 30000 + BlockIdx_Cr
                    
                    if (!Dirty[KeyCr,number]) {
                        local SkipPixels = 8 - (PixelX % 8)
               
                        if ((PixelX + SkipPixels) > Width) {
                            SkipPixels = Width - PixelX
                        }
            
                        if (Scale > 1) {
                            local RawNextRow = floor((I + SkipPixels) / Width)
                            local AlignedNextRow = (floor(RawNextRow / StepX) + 1) * StepX

                            State["index",number] = AlignedNextRow * Width
                        } else {
                            State["index",number] = I + SkipPixels
                        }
                        continue 
                    }
                }
            }
        }
        
        local Y  = YPlane[I,number]
        local Cb = CbPlane[I,number] - 128
        local Cr = CrPlane[I,number] - 128
        
        local R = clamp(Y + 1.402 * Cr, 0, 255)
        local G = clamp(Y - 0.344136 * Cb - 0.714136 * Cr, 0, 255)
        local B = clamp(Y + 1.772 * Cb, 0, 255)
        
        local DSIndex = I
        
        if (Scale > 1) {
            local FinalX = floor(PixelX * InvScale)
            local FinalY = floor(PixelY * InvScale)
            
            if (FinalX != LastX | FinalY != LastY) {
                LastX = FinalX
                LastY = FinalY
                
                if (!NoKeepAR) {
                    DSIndex = (FinalX + Xadd) + (FinalY + Yadd) * PosMul
                } else {
                    DSIndex = FinalX + FinalY * PosMul
                }
                
                local PackedRGB = (floor(R) << 16) || (floor(G) << 8) || floor(B)
                if (TestYCbCr) {
                    PixelsTable[DSIndex,number] = PackedRGB
                } else {
                    DS[DSIndex] = PackedRGB
                }
            }
        } else {
            if (!NoKeepAR) {
                DSIndex = (PixelX + Xadd) + (PixelY + Yadd) * PosMul
            }
            local PackedRGB = (floor(R) << 16) || (floor(G) << 8) || floor(B)
            if (TestYCbCr) {
                PixelsTable[DSIndex,number] = PackedRGB
            } else {
                DS[DSIndex] = PackedRGB
            }
        }

        if (Scale > 1) {
            local SpaceInRow = Width - PixelX - 1
            if (SpaceInRow >= StepX) {
                State["index",number] = I + StepX
            } else {
                local CurrentRow = floor(I / Width)
                local NextRow = CurrentRow + StepX
                State["index",number] = NextRow * Width
            }
        } else {
            State["index",number] = I + 1
        }
    }

    return 0
}

function number jpgInitFullPlanes()
{
    local State = Jpg["full",table]

    if (State["active",number] == 0) {
        
        if (Jpg["maxH",number] == 1 & Jpg["maxV",number] == 1) {
            State["active",number] = 0
            Jpg["planesFull",table] = Jpg["planes",table]
            return 1
        }

        State["active",number] = 1
        State["index",number] = 0

        local Full = table()

        Full[1,table] = Jpg["planes",table][1,table]
        Full[2,table] = table()
        Full[3,table] = table()

        Jpg["planesFull",table] = Full
    }

    local Width = Jpg["width",number]
    local Height = Jpg["height",number]

    local Planes = Jpg["planes",table]
    local Cb = Planes[2,table]
    local Cr = Planes[3,table]

    local Full = Jpg["planesFull",table]
    local CbFull = Full[2,table]
    local CrFull = Full[3,table]

    local Count = Width * Height
    
    local Scan = Jpg["frameComponents",table]
    local JpgComp = Jpg["components",table]
    
    local CbID = Scan[2,table]["id",number]
    local CrID = Scan[3,table]["id",number]
    
    local CbInfo = JpgComp[CbID,table]
    local CrInfo = JpgComp[CrID,table]
    
    local CbW = CbInfo["width",number]
    local CrW = CrInfo["width",number]
    
    local CbScaleX = CbInfo["h",number] / Jpg["maxH",number]
    local CbScaleY = CbInfo["v",number] / Jpg["maxV",number]
    
    local CrScaleX = CrInfo["h",number] / Jpg["maxH",number]
    local CrScaleY = CrInfo["v",number] / Jpg["maxV",number]

    local Scale = Jpg["globalScale", number]
    local StepX = floor(Scale)
    if (StepX < 1) { StepX = 1 }

    while (perf(90)) {

        local I = State["index",number]

        if (I >= Count) {
            State["active",number] = 0
            return 1
        }

        local X = I % Width
        local Y = floor(I / Width)
        
        local CbX = floor(X * CbScaleX)
        local CbY = floor(Y * CbScaleY)
        
        local CrX = floor(X * CrScaleX)
        local CrY = floor(Y * CrScaleY)
         
        CbFull[I,number] = Cb[CbY * CbW + CbX, number]
        CrFull[I,number] = Cr[CrY * CrW + CrX, number]

        if (Scale > 1) {
            local SpaceInRow = Width - X - 1
            if (SpaceInRow >= StepX) {
                State["index",number] = I + StepX
            } else {
                local CurrentRow = floor(I / Width)
                local NextRow = CurrentRow + StepX
                State["index",number] = NextRow * Width
            }
        } else {
            State["index",number] = I + 1
        }
    }

    return 0
}

function jpgCopyBlockToPlane(
    Block:table,
    ComponentID:number,
    X:number,
    Y:number
)
{    
    local Plane = Jpg["planes",table][ComponentID,table]
    local Component = Jpg["components",table][ComponentID,table]
    
    local Width = Component["width",number]
    local Height = Component["height",number]

    local BlockIndex = 0

    for (BY = 0, 7) {

        local PY = Y + BY
 
        if (PY >= Height) {
            break
        }

        local RowOffset = PY * Width

        for (BX = 0, 7) {

            local PX = X + BX
   
            if (PX >= Width) { 
                BlockIndex = BlockIndex + 1
                continue
            }

            local Value = Block[BlockIndex,number]
            
            Value = clamp(Value, 0, 255)
            Plane[RowOffset + PX,number] = Value  
            BlockIndex = BlockIndex + 1
        }
    }
}

function number jpgAssembleImageStep()
{
    local State = Jpg["assemble", table]

    if (State["active", number] == 0) {
        State["active", number] = 1
        State["mcu", number] = 0
        State["component", number] = 1
        State["block", number] = 0     
        State["processed", number] = 0
    }

    local Components = Jpg["frameComponents", table]
    local CompCount = Components:count()    
    
    local JpgBlocks = Jpg["blocks", table]
    local JpgComponents = Jpg["components", table]
    local McuXCount = Jpg["mcuX", number]
    local McuCount = Jpg["mcuCount", number]
    local IsProgressive = Jpg["progressivePreview", number]
    local DirtyBlocks = Jpg["dirtyBlocks", table] 

    while (perf(90)) {

        local MCU = State["mcu", number]

        if (MCU >= McuCount) {
            State["active", number] = 0
            return 1
        }

        local ComponentIndex = State["component", number]
        local Scan = Components[ComponentIndex, table]
        local ComponentID = Scan["id", number]
        
        local Component = JpgComponents[ComponentID, table]

        local H = Component["h", number]
        local V = Component["v", number]
        local BlocksPerMCU = H * V

        local Storage = JpgBlocks[ComponentID, table]
        local BlockNum = State["block", number]

        local Index = jpgComponentBlockIndex(
            Component,
            MCU,
            BlockNum
        )

        if (IsProgressive) {
            local Key = ComponentID * 10000 + Index
            
            if (!DirtyBlocks[Key, number]) {
                
                State["processed", number] = State["processed", number] + 1
                BlockNum = BlockNum + 1
                State["block", number] = BlockNum
    
                if (BlockNum >= BlocksPerMCU) {
                    State["block", number] = 0
                    ComponentIndex = ComponentIndex + 1
                    State["component", number] = ComponentIndex
    
                    if (ComponentIndex > CompCount) {
                        State["component", number] = 1
                        State["mcu", number] = MCU + 1
                    }
                }
                
                continue 
            }
        }
        
        local Block = Storage[Index, table]  

        local McuX = MCU % McuXCount
        local McuY = floor(MCU / McuXCount)
        
        local BlockX = BlockNum % H
        local BlockY = floor(BlockNum / H)
        
        local FinalX = (McuX * H + BlockX) << 3
        local FinalY = (McuY * V + BlockY) << 3
        
        jpgCopyBlockToPlane(
            Block,
            ComponentID,
            FinalX,
            FinalY
        )
        
        State["processed", number] = State["processed", number] + 1
        BlockNum = BlockNum + 1
        State["block", number] = BlockNum

        if (BlockNum >= BlocksPerMCU) {

            State["block", number] = 0
            ComponentIndex = ComponentIndex + 1
            State["component", number] = ComponentIndex

            if (ComponentIndex > CompCount) {
                State["component", number] = 1
                State["mcu", number] = MCU + 1
            }
        }
    }

    return 0
}
