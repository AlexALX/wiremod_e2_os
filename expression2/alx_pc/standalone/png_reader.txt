@name PNG Reader By AlexALX
@inputs PngSizeLimit:number PngNoKeepAR:number
@persist SizeLimit:number NoKeepAR:number
@strict

#[
    PNG Reader
    Created by AlexALX (c) 2026
    -------------------
    PNG Parser, specifications:
    - Max resolution is unlimited, but will be automatically resized to 512x512 (DS limit) or 1024x1024 (Quad DS)
    - Support Baseline and 7-pass Adam7 Interlace
    - Color Depth: Full range (1 to 32 bits)
    - Types: RGB, RGBA, Grayscale & Palette
    - Slow, no multithread
    
    Usage:
    - Spawn one PNG Reader (this file)
    - Connect this E2 to Digital Screen (this is not EGP screen! located in Wire -> Visuals -> Screens -> Digital Screen)
    - Or connect "DS" input to Quad Digital Screen E2 VirtualScreen output (this is custom device dupe)
    
    Chat commands:
    !png http://url_with_png_file - open png file from the web
    Note: long url may be cut, use input LongURL instead for such case (via string value) 
    
    !pngfile filename.png - open png file from a garrysmod/data/e2files/ folder
    
    !pngabort - abort rendering
    
    !pngsize - set maximum image dimensions before scaling (1 to 512 for DS, 1024 for Quad DS)
    !pngkeepar - set image Aspect Ratio mode (0 or 1)

    Additional Wire Inputs:
    PngSizeLimit - Maximum image dimensions before scale
     This can be used to scale down any image before drawing
    PngNoKeepAR - Do not keep Aspect Ratio of image
]#

#include "alx_pc/standalone/includes/basicreader"
#include "alx_pc/standalone/png_reader/_init_"

if (first() | duped()) {
    Command = "!png"
    
    Funcs = table()
    
    Funcs["readerAbort", function] = function() {
        print(_HUD_PRINTTALK,"PNG rendering canceled.")     
        pngAbort()
    }
    
    Funcs["readerLoad", function] = function(Body:string, Size:number, Path:string) {
        pngInit(Body, SizeLimit, NoKeepAR)
    }
    
    Funcs["readerCheck", function] = function() {
        if (!->DS) {
            print(_HUD_PRINTTALK,"Digital Screen is not connected.") 
            return 0
        }
        
        local ScreenName = DS:entity():getName()
        if (DS:entity():type()!="gmod_wire_digitalscreen" & ScreenName!="Quad Digital Screen") {
            print(_HUD_PRINTTALK,"Entity what you did connected is not Digital Screen.\n"+
            "It is located in Wire -> Visuals -> Screens -> Digital Screen.\nThis is NOT EGPv3 screen.") 
            return 0    
        }
        return 1  
    } 
    
    Funcs["readerError", function] = function(Error:string, Path:string) {
        print(_HUD_PRINTTALK, Error)
    }
    
    SizeLimit = PngSizeLimit
    NoKeepAR = PngNoKeepAR
    
    DS_Out = DS
}

event chat(Player:entity, Said:string, Team:number) {
    if (!Command) { exit() }
    
    if (Player==owner()) { 
        
        local Find = Said:lower():find(" ")
        
        if (Find) {
            local Cmd = Said:left(Find-1)
            local Mode = Said:sub(Find+1):lower()

            if (Cmd==Command+"size") {
                local IsQuadScreen = DS:entity():getName()=="Quad Digital Screen"
                local MaxSize = IsQuadScreen ? 1024 : 512
                
                local Size = Mode:toNumber()                
                SizeLimit = clamp(Size,1,MaxSize)
                print(_HUD_PRINTTALK,
                    "Png max size set to "+SizeLimit
                ) 
                if (Size>MaxSize) {
                    if (IsQuadScreen) {
                        print(_HUD_PRINTTALK,
                            "Size clamped, Quad Digital Screen support max 1024x1024"
                        ) 
                    } else {
                        print(_HUD_PRINTTALK,
                            "Size clamped, Digital Screen support max 512x512"
                        ) 
                    }
                }
            } elseif (Cmd==Command+"keepar") {
                if (Mode=="false" | Mode=="0" | Mode=="nokeep" | Mode=="off") {
                    NoKeepAR = 1    
                    print(_HUD_PRINTTALK,
                        "Aspect Ratio: DISABLED (Stretch to fit)"
                    ) 
                } else {
                    NoKeepAR = 0   
                    print(_HUD_PRINTTALK,
                        "Aspect Ratio: ENABLED (Preserve proportions)"
                    ) 
                }
            }
        }
    }
}

if (~PngSizeLimit) {
    SizeLimit = PngSizeLimit
}

if (~PngNoKeepAR) {
    NoKeepAR = PngNoKeepAR
}

if (~DS) {
    DS_Out = DS    
}
