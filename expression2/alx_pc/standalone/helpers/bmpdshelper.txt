@name BMP Helper for DS
@persist Header:string Inited:number
@inputs BmpError:string BmpPercent:number BmpStep:number BmpDrew:number BmpCurrentStep:string
@strict

#include "alx_pc/standalone/helpers/_shared_"

function displayHelp() {
    displayInit(1)

    local Text = Header+
"
BMP Parser, specifications:
- Max resolution is up to 512x512 (DS limit)
  or 1024x1024 (Quad DS Limit)
- Support most common bmp files: 
- 32/24/8/4 bits and monochrome
- including RGB444/RGB555/RGB565
- Support OS/2 BMP files
- Support RLE Compression
- Support \"flip row order\" function
- Slow, but multithreaded

Usage via chat commands:
!bmp http://url - open bmp file
Note: long url may be cut, use input LongURL 
instead for such case (via string value) 

!bmpfile filename.bmp.txt - open bmp file 
from garrysmod/data/e2files/ folder
file extension must be renamed to .txt

!bmpabort - abort rendering
!bmpsize - limit image scale (1-512)
!bmpkeepar - set Aspect Ratio mode (0 or 1)

If you have own web server with PHP, 
then you can place img2bmp.php from php dir
https://github.com/AlexALX/wiremod_e2_os/
and then use it to convert any image to BMP
with auto resize in url:
http://yoursv/img2bmp.php?http://url/img.jpg
"

    dsDrawTextStart(2, 0, Text)
}

function displayError(Msg:string) {
    displayInit()
    
    local Text = Header+
"
BMP Reader Error:

"+Msg+"
"

    dsDrawTextStart(2, 0, Text)
}

function displayProgressStart() {
    if (Inited) { return }
    
    local ProcessorDS = E2Processor["DS_Out",wirelink]
    local IsSameScreen = ProcessorDS==DS
    
    if (IsSameScreen & (BmpDrew | BmpStep>2)) {
        dsStopQueues()
        Inited = 0
        return
    }
    
    local Threads = E2Processor["BmpThreads",number]
    
    displayInit()
    
    displayBlockDraw(0.5)
    
    local MaxSteps = E2Processor["BmpMaxSteps",number]
    
    local Text = Header+
"
Step:
1 / "+MaxSteps+"

Current task:
Loading BMP

Progress:

"
    
    dsDrawTextStart(2, 0, Text)
}

function displayProgress()
{ 
    local ProcessorDS = E2Processor["DS_Out",wirelink]
    local IsSameScreen = ProcessorDS==DS
    
    if (IsSameScreen & BmpDrew) {
        dsStopQueues()
        return
    }    
    
    if (BlockDraw & BmpCurrentStep!="done") {
        return
    }
    
    if (BmpCurrentStep=="done") {
        Inited = 0    
    }
    
    local FontLine = DsFontHeight + DsFontSpacing
    
    local Threads = E2Processor["BmpThreads",number]
    local ThreadsTxt = ""
    if (BmpStep>2) {    
        ThreadsTxt = " ("+Threads+" Thread"+(Threads>1 ? "s" : "")+")"
    }

    local MaxSteps = E2Processor["BmpMaxSteps",number]

    local Task = E2Processor["BmpCurrentStepText",string]

    displayBlockDraw(1)

    dsDrawCachedText(2, 4 * FontLine, BmpStep+" / "+MaxSteps)
    dsDrawCachedText(2, 7 * FontLine, Task+ThreadsTxt)
    dsDrawCachedText(2, 10 * FontLine, BmpPercent+"%")
}

function displayMode() {
    if (BmpError){ 
        displayError(BmpError)
    } else {
        displayHelp()    
    }   
}

if (first() | duped()) {
    Header = 
"BMP Reader - Multithreaded
Created by AlexALX (c) 2018-2026
-------------------"
    
    displayMode()
}


event input(InputName:string) {    
    if (InputName=="DS" & DS!=nowirelink()) {
        displayMode()
    } elseif (InputName=="BmpError" & BmpError) {
        displayError(BmpError)    
    } elseif (InputName=="BmpStep") {
        if (BmpStep) {        
            displayProgressStart()
        } else {
            Inited = 0    
        }
    } elseif ((InputName=="BmpPercent" | InputName=="BmpCurrentStep") & BmpStep) { 
        displayProgress() 
    } elseif (InputName=="BmpDrew" & BmpDrew) {
        local ProcessorDS = E2Processor["DS_Out",wirelink]
        local IsSameScreen = ProcessorDS==DS
            
        if (IsSameScreen) {
            Inited = 0
            dsStopQueues()    
        }
    }
}
