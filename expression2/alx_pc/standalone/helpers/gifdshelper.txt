@name GIF Helper for DS
@persist Header:string
@inputs CurrentState:string Progress:table ProgressText:string Processing:number Error:string Warning:string
@strict

#include "alx_pc/standalone/helpers/_shared_"

function displayHelp() {
    displayInit(1)

    local Text = Header+
"
GIF Parser, specifications:
- Max resolution is unlimited, but
  will be resized to 512x512 (DS limit) 
  or 1024x1024 (Quad DS Limit) 
- Baseline and Interlace files supported
- Animation with multiple loop modes
- Loop and Disposal support
- Due to digital screen speed issues anims 
   work properly only on very small images
   or you can resize output by GifSizeLimit

Usage via chat commands:
!gif http://url - open gif file from the web
Note: long url may be cut, use input LongURL 
instead for such case (via string value) 

!giffile filename.gif - open gif file from 
garrysmod/data/e2files/ folder
file extension must be renamed to .txt

!gifabort - abort rendering
!gifmode - normal, manual, animate
Modes:
 normal - first frame, no animation
 manual - play next frame via !gifnext
 animate - automatic animation play

!gifsize - limit image scale (1-512)
!gifdelay - frame delay limit (min 0.1)
!gifkeepar - set Aspect Ratio mode (0 or 1)
"

    dsDrawTextStart(2, 0, Text)
}

function displayProgressStart() {
    displayInit()
    
    displayBlockDraw(0.5)

    local Text = Header+
"
Step:
0 / 0

Current task:
Loading GIF

Progress:

"
    
    dsDrawTextStart(2, 0, Text)
}

function displayError(Msg:string) {
    displayInit()
    
    local Text = Header+
"
GIF Reader Error:

"+Msg+"
"

    dsDrawTextStart(2, 0, Text)
}

function displayProgress()
{
    local ProcessorDS = ImageProcessor["DS_Out",wirelink]
    local IsSameScreen = ProcessorDS==DS
    
    if (IsSameScreen & (CurrentState=="draw" | CurrentState=="done")) {
        dsStopQueues()
        return
    }
     
    local IsLastUpdate = (!Processing & CurrentState=="done")
    
    if (BlockDraw & !IsLastUpdate) {
        return
    }
    
    local Step = Progress["step", number]
    local Current = Progress["progressCurrent", number]
    local Max = Progress["progressMax", number]
    local Perc = Progress["progressPercent", number]
    local CurrentTask = Progress["progressTask",string]
    local Drew = Progress["progressDrew",number]
    
    # if DS drew preview
    if (IsSameScreen & Drew) {
        return    
    }
    
    local FontLine = DsFontHeight + DsFontSpacing
    
    dsDrawCachedText(2, 4 * FontLine, Step + " / "+Progress["stepTotal",number])
    dsDrawCachedText(2, 7 * FontLine, CurrentTask)
    
    local Txt = Current + " / " + Max + " ("+Perc+"%)"
    if (IsLastUpdate) {
        Txt = "100%"
    }
    dsDrawCachedText(2, 10 * FontLine, Txt)
    
    displayBlockDraw(2)
    
}

function displayMode() {
    if (Error){ 
        displayError(Error)
    } elseif (Processing) {
        displayProgressStart()
    } else {
        displayHelp()    
    }   
}

function displayWarning(Msg:string) {
    local FontLine = DsFontHeight + DsFontSpacing
        
    local Lines = stringWordWrap(Msg,33)

    dsDrawCachedText(2, 13 * FontLine, "Warning:")
    
    for (I=1,Lines:count()) {
        dsDrawCachedText(2, (13+I) * FontLine, Lines[I,string])     
    }    

    displayBlockDraw(3)
}

if (first() | duped()) {
    Header = 
"GIF Reader
Created by AlexALX (c) 2026
-------------------"
    
    displayMode()
}

event input(InputName:string) {  
    if (InputName=="DS" & DS!=nowirelink()) {
        displayMode()
    } elseif (InputName=="Processing" & Processing) {
        displayProgressStart()
    } elseif (InputName=="Error" & Error) {
        displayError(Error)    
    } elseif (InputName == "Progress" & Progress:count()) {
        displayProgress()
    } elseif (InputName == "Warning" & Warning) {
        displayWarning(Warning)           
    } elseif (InputName == "CurrentState") {
        if (CurrentState=="done") {
            displayProgress()        
        }
    }
}
