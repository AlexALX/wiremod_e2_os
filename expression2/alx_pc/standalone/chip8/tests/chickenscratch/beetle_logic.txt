#include "alx_pc/standalone/chip8/tests/_shared_"

function resetTestEnv(Type:number) {
    c8Reset()
    ChipType = Type
    PC = 0x200
}

# ==============================================================================
# DIAGNOSTIC TEST FOR REGISTERS MATH AND CONDITIONAL FLAGS RETENTION
# ==============================================================================

ExecuteTestStep = function(StepId:number) {
    local Name = ""
    
    if (StepId == 1) {
        Name = "OP 8XY5 (SUB) VF BORROW FLAG ON EQUAL (VX == VY)"
        resetTestEnv(2)
        
        # Test case: 16 - 16. Under CHIP-8 specs, if VX >= VY, VF must be 1.
        V[0, number] = 16
        V[1, number] = 16
        
        # 8015 -> V0 = V0 - V1
        Memory[0x200, number] = 0x80
        Memory[0x201, number] = 0x15
        c8Step()
        
        if (!assertCase(Name, 1, V[0, number], 0)) { return 0 }
        # VF MUST be 1 because 16 >= 16 (No borrow needed)
        if (!assertCase(Name, 2, V[15, number], 1)) { return 0 }
        
        testPass(Name)
    }
    elseif (StepId == 2) {
        Name = "OP 8XY5 (SUB) VF BORROW FLAG ON UNDERFLOW (VX < VY)"
        resetTestEnv(2)
        
        # Test case: 5 - 8. Under CHIP-8 specs, if VX < VY, VF must be 0 (Borrow occurred).
        V[0, number] = 5
        V[1, number] = 8
        
        # 8015 -> V0 = V0 - V1
        Memory[0x200, number] = 0x80
        Memory[0x201, number] = 0x15
        c8Step()
        
        # Unsigned 8-bit wrap: 5 - 8 = -3 -> 253 (0xFD)
        if (!assertCase(Name, 1, V[0, number], 253)) { return 0 }
        # VF MUST be 0 because 5 < 8 (Borrow occurred)
        if (!assertCase(Name, 2, V[15, number], 0)) { return 0 }
        
        testPass(Name)
    }
    elseif (StepId == 3) {
        Name = "OP 8XY7 (SUBN) VF BORROW FLAG ON EQUAL (VY == VX)"
        resetTestEnv(2)
        
        # Test case: V1 - V0 -> 10 - 10. If VY >= VX, VF must be 1.
        V[0, number] = 10
        V[1, number] = 10
        
        # 8017 -> V0 = V1 - V0
        Memory[0x200, number] = 0x80
        Memory[0x201, number] = 0x17
        c8Step()
        
        if (!assertCase(Name, 1, V[0, number], 0)) { return 0 }
        if (!assertCase(Name, 2, V[15, number], 1)) { return 0 }
        
        testPass(Name)
    }
    elseif (StepId == 4) {
        Name = "OP 8XY6 (SHR) CHIP8 VS XOCHIP REGISTER TARGET VALIDATION"
        resetTestEnv(2) # XO-CHIP mode
        
        # In XO-CHIP mode, SHR shifts VX, ignoring VY.
        V[1, number] = 0x03 # VX
        V[2, number] = 0x00 # VY
        
        # 8126 -> SHR V1
        Memory[0x200, number] = 0x81
        Memory[0x201, number] = 0x26
        c8Step()
        
        # V1 (0x03) shifted right by 1 = 1. LSB was 1, so VF = 1.
        if (!assertCase(Name, 1, V[1, number], 1)) { return 0 }
        if (!assertCase(Name, 2, V[15, number], 1)) { return 0 }
        
        testPass(Name)
    }

    return 1
}

TotalCases = 4
runTests()
