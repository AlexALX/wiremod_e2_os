@persist Devices:table TickQuota:number CurrentFS:table AllFs:table LastProgram:string
@persist EGP:wirelink DS:wirelink CS:wirelink

#[
    To use this chip via Remote Executor you need to enable 
    e2 remoteupload extension: wire_expression2_extension_enable remoteupload

    How to execute this program via Remote Executor:
    1. You need copy this file from "garrysmod/data/expression2/alx_pc/gpu/remote" to "garrysmod/data/e2files"
    2. Create file in ALX OS with name "testprog.e2r" and content "file://testprog.txt"
    3. Spawn "remote executor" part from advdupe2, install to PCI slot.
    4. Run file, you should see test dialog.
    
    If you want to run this file as a .e2p:
    1. Uncomment e2p_include below, without this your remote executor will break!
    2. Create file "testprog.e2p" with content "alx_pc/remote/testprog.txt" (this is string value! not a file content)
    3. Run file, you should see test dialog.
    
    There is example how exactly should look .e2p inside:
    https://github.com/AlexALX/wiremod_e2_os/tree/master/examples/example_e2p.txt

if (first()|duped()) {
    
    # app manifest
    AppManifest = table()
    AppManifest["name", string] = "Test Program"
    AppManifest["version", string] = "1.0"
    AppManifest["author", string] = "AlexALX"
    
    # Supported screens, for example:
    # if alx os is running without EGP connected
    # and your program require EGP - it should be canceled
    local Screens = table()
    Screens["egp",number] = 1
    Screens["console",number] = 0
    Screens["digital",number] = 0
    
    # Modes:
    # any - means one of screens are connected
    # all - means screens are REQUIRED (simultaneously)
    Screens["mode",string] = "any"
    
    # this is automatically added to any executables 
    # and REQUIRED to be included, after AppManifest and inside first() check!
    #include "alx_pc/remote/e2_include"
    # in .e2r ANY other include will make error!
}

# use this event as entry point into program!
event remote(Sender:entity, Player:entity, Payload:table) { 
    # protect from broadcast
    if (Sender:id()!=ExecutorID) { exit() }
    
    Devices = Payload["Devices",table]
    TickQuota = Payload["TickQuota",number]
    CurrentFS = Payload["CurrentFS",table]
    AllFs = Payload["AllFs",table]

    # This will have alxos_explorer or alxos_console
    LastProgram = Payload["LastProgram",string]
    
    # This is only for e2z
    #Assets = Payload["Assets",table]
    
    EGP = Devices["GPU_Screens",table]["EGP_WL",wirelink]
    DS = Devices["GPU_Screens",table]["DS_WL",wirelink]
    CS = Devices["GPU_Screens",table]["CS_WL",wirelink]
    
    # we must clear it always! since in isolated mode it will override EGP IDs!
    EGP:egpClear()
    
    local LAST_I = 1
    
    EGP:egpText(LAST_I,"TEST EXECUTE INIT, PRESS ENTER OR F8 TO EXIT",vec2(256,246))
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,255,255,255,255) LAST_I++
    
    EGP:egpText(LAST_I,"PRESS F7 TO EXIT INTO CONSOLE",vec2(256,266))
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,255,255,255,255) LAST_I++
}

# F8 key are RESERVED for Executor to shutdown whole program!
function dsKeyPressed(Key:string, Pressed) {
    if (Key=="enter") {
        # this function automatically added
        exitProg()
    }
    
    if (Key=="f7") {
        # this will exit to directory you was initially running
        dsSendDirect("alxos_runprog",Devices["cpu",entity],table("alxos_console"))    
    }

    # handle keys here
}

if (dsClk()) {
    local Name = dsClkName()

    if (Name=="executorKey") {
        # Handle key press
        local Info = dsGetTable()
        dsKeyPressed(Info[1,string], Info[2,number])
    } elseif (Name=="ScreenConnect") {
        # update connection if its changed
        local Type = dsGetString()
        if (Type=="EGP") {
            EGP = Devices["GPU_Screens",table]["EGP_WL",wirelink]   
        } elseif (Type=="DS") {
            DS = Devices["GPU_Screens",table]["DS_WL",wirelink]  
        } elseif (Type=="CS") {
            CS = Devices["GPU_Screens",table]["CS_WL",wirelink]   
        }
    }
}
