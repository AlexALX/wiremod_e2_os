# use this event as entry point into program!
event remote(Sender:entity, Player:entity, Payload:table) {
    # protect from broadcast
    if (Sender:id()!=ExecutorID) { exit() }
    
    Devices = Payload["Devices",table]
    TickQuota = Payload["TickQuota",number]
    CurrentFS = Payload["CurrentFS",table]
    AllFs = Payload["AllFs",table]

    # This will have alxos_explorer or alxos_console
    LastProgram = Payload["LastProgram",string]
    
    # This is only for e2z
    #Assets = Payload["Assets",table]
    
    EGP = Devices["GPU_Screens",table]["EGP_WL",wirelink]
    DS = Devices["GPU_Screens",table]["DS_WL",wirelink]
    CS = Devices["GPU_Screens",table]["CS_WL",wirelink]
    
    # we must clear it always! since in isolated mode it will override EGP IDs!
    EGP:egpClear()
    
    LAST_I = 1
    
    local Parent = alxos_dialog("Test",410,80,"","")
    
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"TEST EXECUTE INIT, PRESS ENTER OR F8 TO EXIT",vec2(205,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"PRESS F7 TO EXIT INTO CONSOLE",vec2(205,40))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
}

# F8 key are RESERVED for Executor to shutdown whole program!
function dsKeyPressed(Key:string, Pressed) {
    if (Key=="enter") {
        # this function automatically added
        ExitProg()
    }
    
    if (Key=="f7") {
        # this will exit to directory you was initially running
        dsSendDirect("alxos_runprog",Devices["cpu",entity],table("alxos_console"))    
    }
    
    # handle keys here
}

if (dsClk()) {
    local Name = dsClkName()

    if (Name=="executorKey") {
        # Handle key press
        local Info = dsGetTable()
        dsKeyPressed(Info[1,string], Info[2,number])
    } elseif (Name=="DS_Connect") {
        # update DS connection if its changed
        DS = Devices["GPU_Screens",table]["DS_WL",wirelink]   
    } elseif (Name=="CS_Connect") {
        # update CS connection if its changed
        CS = Devices["GPU_Screens",table]["CS_WL",wirelink]   
    }
}
