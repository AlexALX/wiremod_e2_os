@name Image Accelerator
@inputs InA Board:wirelink JPGReader:wirelink GIFReader:wirelink PNGReader:wirelink
@outputs EntID DS:wirelink
@persist BIOS:entity Device_Name:string
@persist E2_GPU:entity Init

#[
    ALX Wiremod E2 PC
    Created by AlexALX
    -------------------
]#

if (first()|duped()) {
    
#include "alx_pc/shared/_main_"
#include "alx_pc/shared/cmd"

dsSetScope(2)

BIOS = getentity(InA)
Board["A",number] = entity():id()
if (!duped()) {
    dsSendDirect("reload",BIOS,"imageacc")
}

Init = 0
Device_Name = "Image Accelerator"

function shutdown() {
    if (!Init) { return }

    stopAllTimers()

    dsSendDirect("abortImage",JPGReader:entity(),1)
    dsSendDirect("abortImage",GIFReader:entity(),1)
    dsSendDirect("abortImage",PNGReader:entity(),1)
    DsHandler = ""
    Init = 0

}

function pci_init() {    
    dsSendDirect("pci",BIOS,table("imageacc",entity(),Device_Name))
}

function usb_init() {
    Init = 1
    local Info = dsGetTable()
    local USB_Device = Info[2,entity]
    dsSendDirect("usbcontroller",USB_Device,table("imageacc",wirelink(),table()))
}

function number img_timeout() {
    dsSendDirect("imageReaderError",E2_GPU,1)
    return 0    
}

}

if (dupefinished()) {
    Board["A",number] = entity():id()
    BIOS = getentity(InA)
}

if (~InA) {
    local New_BIOS = entity(InA)
    if (BIOS!=New_BIOS & Init) {
        shutdown()
    }
    BIOS = New_BIOS
}

if (!BIOS:isValid()) { exit() }

if (dsClk()) {
    local Name = dsClkName()
    
    if (Name=="drawImage") {
        local Info = dsGetTable()
        DS = Info[1,wirelink]
        E2_GPU = Info[2,entity]
        TickQuota = Info[3,number]
        local Type = Info[4,string]
        local ImgInfo = Info[5,table]
        Init = 1
        
        if (Type=="jpg") {
            if (JPGReader == nowirelink()) {
                dsSendDirect("imageReaderError",E2_GPU,1)
            } else {        
                dsSendDirect("drawImage", JPGReader:entity(), ImgInfo)
                runCmd("img_timeout",1000,1)
            }
        } elseif (Type=="gif") {
            if (GIFReader == nowirelink()) {
                dsSendDirect("imageReaderError",E2_GPU,1)
            } else {
                dsSendDirect("drawImage", GIFReader:entity(), ImgInfo)
                runCmd("img_timeout",1000,1)
            }
        } elseif (Type=="png") {
            if (PNGReader == nowirelink()) {
                dsSendDirect("imageReaderError",E2_GPU,1)
            } else {
                dsSendDirect("drawImage", PNGReader:entity(), ImgInfo)
                runCmd("img_timeout",1000,1)
            }
        }
    } elseif (Name=="pci") {
        pci_init()
    } elseif (Name=="usbcontroller") {
        usb_init()
    } elseif (Name=="SHUTDOWN") {
        shutdown()
    } elseif (DsHandler!="") {
        DsHandler(Name)
    } elseif (Name=="initImage") {
        stopCmd()
        dsSendDirect("initImage", E2_GPU, dsGetTable())     
    }
}

if (clk("cmd")) {
    cmdClk()
}
