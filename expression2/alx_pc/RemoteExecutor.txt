@name Remote Executor
@inputs InA Board:wirelink ExecutorID:number
@outputs EntID:number
@persist BIOS:entity Device_Name:string RemoteInfo:table
@persist Devices:table Init Caller:entity CodePrepend:string CodeAppend:string CodeExecute:string CodeInclude:string

#[
    ALX Wiremod E2 PC
    Created by AlexALX
    -------------------
    This device allow to execute e2 code via remoteupload extension from raw code
    Currently support are limited to just code with no includes (or must exists locally)
    And is totally isolated - no way to call gui directly.
    File formats:
    - .e2r contains program code directly
    - .e2p path to expression2 chip (relative to garrysmod/data/expression2), must exists locally
    - .e2z executable archive bundle with assets and includes support
    
    Read more about it on wiki: https://github.com/AlexALX/wiremod_e2_os/wiki/Creating-external-program
]#

if (first()|duped()) {
    
#include "alx_pc/shared/_main_"
#include "alx_pc/shared/cmd"

dsSetScope(2)

BIOS = getentity(InA)
Board["A",number] = entity():id()
if (!duped()) {
    dsSendDirect("reload",BIOS,"rmtexec")
}

Init = 0
Device_Name = "Remote Executor"
Caller = noentity()
EntID = entity():id()

# This code needs always be added to executor chip so will be proper sync
CodePrepend = "@inputs ExecutorID:number\r\n@outputs RemoteEntID:number\r\n@persist ExecutorID:number IsDirectCode:number\r\n"
CodeAppend = "
if (first()|duped()) {
    dsSetScope(2)
    RemoteEntID = entity():id()
    IsDirectCode = 1
    
    ExitProg = function() {
        stopAllTimers()
        dsSendDirect(\"exitProg\", entity(ExecutorID), 1)
        exit()     
    }
}
"

# this code needs to run on program to properly init
CodeExecute = "
function dsInitProg() {
    dsSendDirect(\"initExecutor\", entity(ExecutorID), table(wirelink()))        
}

if (first()) {
    dsInitProg()
} 
"

CodeInclude = "
@name E2P/E2Z Include (REQUIRED)
@inputs ExecutorID:number
@outputs RemoteEntID:number
@persist IsDirectCode:number ExitProg:function

#[
    If you want create .e2p or .e2z file you MUST include this file always
    Or it will not properly init with executor and shutdown your code!

    Note: this file should not be added into e2z zip archive, its added automatically by remote executor.
]#

if (first()|duped()) {
    dsSetScope(2)
    RemoteEntID = entity():id()
    IsDirectCode = 0

    function dsInitProg() {
        dsSendDirect(\"initExecutor\", entity(ExecutorID), table(wirelink()))
    }
    
    function endProg() {
        stopAllTimers()
        dsSendDirect(\"SHUTDOWN\", entity(ExecutorID), 1)
        exit()
    }
    
    function exitProg() {
        stopAllTimers()
        dsSendDirect(\"exitProg\", entity(ExecutorID), 1)
        exit()
    }
    
    # just for unify
    ExitProg = function() { 
        exitProg() 
    }
    
    if (first()) {
        dsInitProg()
    } 

}

"

function clearMemory() {
    RemoteInfo = table()    
}

function setNoCode() {
    #ifdef entity:remoteSetCode(string)
    getentity(ExecutorID):remoteSetCode(CodePrepend + CodeAppend)
    #endif
    
    clearMemory()
}

function shutdown() {
    if (!Init) { return }

    stopAllTimers()

    setNoCode()
    DsHandler = ""
    Init = 0
    Caller = noentity()
}

function pci_init() {    
    dsSendDirect("pci",BIOS,table("rmtexec",entity(),Device_Name))
}

function usb_init() {
    local Info = dsGetTable()
    local USB_Device = Info[2,entity]
    dsSendDirect("usbcontroller",USB_Device,table("rmtexec",wirelink(),table()))
}

function number exec_timeout() {
    dsSendDirect("execError",Caller,1)
    setNoCode()
    return 0
}

}

#ifndef entity:remoteSetCode(string)
    error("This device cannot work without e2 remoteupload extension. Please enable via wire_expression2_extension_enable remoteupload.")
    exit()
#else

if (dupefinished()) {
    Board["A",number] = entity():id()
    BIOS = getentity(InA)
}

if (~InA) {
    local New_BIOS = entity(InA)
    if (BIOS!=New_BIOS & Init) {
        shutdown()
    }
    BIOS = New_BIOS
}

if (!BIOS:isValid()) { exit() }

if (dsClk()) {
    local Name = dsClkName()
    
    if (Name=="executeCode") {
        if (ExecutorID==0 | entity(ExecutorID)==noentity()) {
            dsSendDirect("execError",dsGetSender(),2)
            exit()
        }
        
        RemoteInfo = dsGetTable()
        local FInfo = RemoteInfo["FInfo",table]
        
        local Type = (FInfo["real_ext",string]!="" ? FInfo["real_ext",string] : FInfo["ext",string]):lower()
        local Code = FInfo["content",string]

        Init = 1
        Caller = dsGetSender()
        
        local Executor = getentity(ExecutorID)
        
        #ifdef entity:remoteSetCode(string)
        
        if (Type=="e2z") {
            local Includes = FInfo["includes",table]
            
            # auto append it to file
            Includes["alx_pc/remote/e2p_include",string] = CodeInclude
            
            Executor:remoteSetCode(Code, Includes)
        } elseif (Type=="e2r") {      
            Executor:remoteSetCode(CodePrepend + Code + CodeAppend + CodeExecute)
        } elseif (Type=="e2p") {
            Executor:remoteUpload(Code)
        }
        #endif
        
        runCmd("exec_timeout",1000,1)
    } elseif (Name=="pci") {
        pci_init()
    } elseif (Name=="usbcontroller") {
        usb_init()
    } elseif (Name=="SHUTDOWN") {
        shutdown()
    } elseif (Name=="exitProg") {
        dsSendDirect("exitProg",Caller,1)
        shutdown()
    } elseif (DsHandler!="") {
        DsHandler(Name)
    } elseif (Name=="initExecutor") {        
        stopCmd()
        dsSendDirect("initExecutor", Caller, dsGetTable())  
    } elseif (Name=="executeInitDone") {
        local Executor = getentity(ExecutorID)
        
        RemoteInfo["Assets",table] = RemoteInfo["FInfo",table]["assets",table]
        RemoteInfo:remove("FInfo")
             
        Executor:sendRemoteEvent(RemoteInfo)
        
        # clear memory now
        clearMemory()
    } elseif (Name=="executorKey") {
        local Executor = getentity(ExecutorID)
        dsSendDirect("executorKey", Executor, dsGetTable())    
    } elseif (Name=="DS_Connect" | Name=="CS_Connect") {
        local Executor = getentity(ExecutorID)
        dsSendDirect(Name, Executor, dsGetNumber())  
    }
}

if (clk("cmd")) {
    cmdClk()
}
