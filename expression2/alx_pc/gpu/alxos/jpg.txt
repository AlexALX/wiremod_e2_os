function alxos_jpg_init(Args:table) {
    if (!alxos_img_init(Args, "jpg")) {
        return    
    }
    
    if (CurrentProg=="alxos_console") {
        DialogData = table()
        "alxos_img_postinit"()
        return
    }
    
    local Width = 450
    local Height = 245

    local Parent = "alxos_explorer_dialog"("JPG Settings",Width,Height,"","alxos_img_close")[number]
    
    DialogData[7,number] = LAST_I
    DialogData[-10,table]:pushNumber(LAST_I)
    
    local LAST_Y = 18
    
    Txt = "Please choose JPG Draw Mode:\n\n"+
    "Normal - no preview, fastest render\n"+
    "Preview - first AC scan preview\n"+
    "Full Preview - every scan rerender (slow)"
      
    EGP:egpTextLayout(LAST_I,Txt,vec2(22,LAST_Y),vec2(Width-40,95))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    LAST_Y += 102
    
    local Entries = table()
    local EntID = DialogData[-10,table]
    local JpgMode = GpuPersistData["jpgMode",number]
    
    # 1. Normal Mode Button
    Entries[1,number] = LAST_I
    EntID:pushNumber(LAST_I)   
    local Sz = 65
    local Txt = "Normal"
    local CenterX = (Width / 3) * 0.8
    local Col = vec4(168,168,168,255)
    if (JpgMode==0) { Col = vec4(0,168,168,255) }
    EGP:egpBox(LAST_I,vec2(CenterX - Sz/2,LAST_Y),vec2(Sz,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,Col) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    EntID:pushNumber(LAST_I)        
    EGP:egpText(LAST_I,Txt,EGP:egpPos(LAST_I-1)+vec2(5,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    # 2. Preview Mode Button
    Entries[2,number] = LAST_I
    EntID:pushNumber(LAST_I)  
    local Sz = 70
    local Txt = "Preview"
    local CenterX = (Width / 3) * 1.5
    local Col = vec4(168,168,168,255)
    if (JpgMode==1) { Col = vec4(0,168,168,255) }
    EGP:egpBox(LAST_I,vec2(CenterX - Sz/2,LAST_Y),vec2(Sz,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,Col) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    EntID:pushNumber(LAST_I)        
    EGP:egpText(LAST_I,Txt,EGP:egpPos(LAST_I-1)+vec2(5,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    # 3. Full Preview Mode Button
    Entries[3,number] = LAST_I
    EntID:pushNumber(LAST_I)  
    local Sz = 103
    local Txt = "Full Preview"
    local CenterX = (Width / 3) * 2.3
    local Col = vec4(168,168,168,255)
    if (JpgMode==2) { Col = vec4(0,168,168,255) }
    EGP:egpBox(LAST_I,vec2(CenterX - Sz/2,LAST_Y),vec2(Sz,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,Col) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    EntID:pushNumber(LAST_I)        
    EGP:egpText(LAST_I,Txt,EGP:egpPos(LAST_I-1)+vec2(5,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++

    LAST_Y += 35

    # 4. Max Size Input
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"Size Limit (px):",vec2(22,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"512 max DS, 1024 Quad DS",vec2(220,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpBox(LAST_I,vec2(150,LAST_Y),vec2(60,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,168,168,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    local MaxSize = gpuScreenSize(DS):toString()
    
    DialogData[-10,table]:pushNumber(LAST_I)
    local DefaultMaxSize = GpuPersistData["jpgMaxSize",number] != 0 ? GpuPersistData["jpgMaxSize",number]:toString() : MaxSize
    Entries[4,table] = table(LAST_I,DefaultMaxSize,"^[0-9]$",4)
    EGP:egpText(LAST_I,DefaultMaxSize,vec2(152,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++

    LAST_Y += 35

    # 5. Keep AR Input
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"Keep Aspect Ratio (1/0):",vec2(22,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpBox(LAST_I,vec2(220,LAST_Y),vec2(40,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,168,168,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    local DefaultNoKeepAR = GpuPersistData["jpgNoKeepAR",number] == 1 ? "0" : "1"
    Entries[5,table] = table(LAST_I,DefaultNoKeepAR,"^[0-1]$",1)
    EGP:egpText(LAST_I,DefaultNoKeepAR,vec2(222,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++

    LAST_Y = Height - 25

    # 6. OK Button (In focus by default)
    local SzOK = 30
    local AddOK = -100
    
    DialogData[-10,table]:pushNumber(LAST_I)
    Entries[6,number] = LAST_I
    EGP:egpBox(LAST_I,vec2((SzOK/2)+AddOK+Width/2,Height-22),vec2(SzOK,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,168,0,255)
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    EGP:egpText(LAST_I,"OK",EGP:egpPos(LAST_I-1)+vec2(5,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255)
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++

    # 7. Cancel Button
    local SzCan = 60
    local AddCan = 20
    
    DialogData[-10,table]:pushNumber(LAST_I)
    Entries[7,number] = LAST_I
    EGP:egpBox(LAST_I,vec2((SzCan/2)+AddCan+Width/2,Height-22),vec2(SzCan,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,168,168,168,255)
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    EGP:egpText(LAST_I,"Cancel",EGP:egpPos(LAST_I-1)+vec2(5,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255)
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++

    DialogData[-5,number] = 6
    DialogData[-4,table] = Entries
    DialogData[10,table] = Entries
    DialogData[11,number] = 6
    DialogData[12,number] = 7
    DialogData[-2,string] = "text"
}

function number alxos_jpg_cmd() {
    return 0
}

function alxos_jpg_draw(Data:string, IMGAccelerator:entity) {
    dsSendDirect("drawImage",IMGAccelerator,table(
        DS, entity(), TickQuota, "jpg", table(
        Data,
        GpuPersistData["jpgMode",number],
        GpuPersistData["jpgMaxSize",number],
        GpuPersistData["jpgNoKeepAR",number]
        )
    ))
}

function number alxos_jpg_key(Key:string,Pressed) {
    if (!Pressed) { return 0 }
    
    local Index = DialogData[-5,number]
    local Type = DialogData[-2,string]
    local JpgMode = GpuPersistData["jpgMode",number]
    
    local Title = DialogData["title",string]   
    
    if (Key=="f8") {
        stoptimer("cmd")
        dsSendDirect("clear",GpuData[1,table]["hdd_io",wirelink]:entity(),1)
        local IMGAccelerator = alxos_img_findAccelerator()
        dsSendDirect("SHUTDOWN",IMGAccelerator[3,entity],0)  
        alxos_img_close()
        return 1
    } elseif (Key=="enter" & Title!="Opening file" & Title!="JPG Settings") {
        alxos_jpg_key("f8",1)
        return 1
    }
    
    if (Title!="JPG Settings") { return 0 }
    
    if (Key=="left" | Key=="right" | Key=="tab") {
        local Max = 7
        local Ind = Index
        if (Key=="right" | Key=="tab") {
            Index = (Index==Max ? 1 : Index+1)
        } else {
            Index = (Index==1 ? Max : Index-1)
        }
        DialogData[-5,number] = Index
        DialogData[11,number] = Index
        
        local CurID = DialogData[-4,table][Index,number]     
        local PrevID = DialogData[-4,table][Ind,number]  
        
        if (Ind <= 3) {
            local OldCol = (JpgMode == (Ind-1) ? vec4(0,168,168,255) : vec4(168,168,168,255))
            EGP:egpColor(PrevID, OldCol)
        } elseif (Ind >= 6) {
            EGP:egpColor(PrevID, vec4(168,168,168,255))
        }
        
        if (Index <= 3 | Index >= 6) { EGP:egpColor(CurID, vec4(0,168,0,255)) }
        
        # "_"
        if (Index == 4 | Index == 5) {
            local TxtTbl = DialogData[-4,table][Index,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string]+"_",EGP:egpPos(TxtTbl[1,number]))
        }
        if (Ind == 4 | Ind == 5) {
            local TxtTbl = DialogData[-4,table][Ind,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string],EGP:egpPos(TxtTbl[1,number]))
        }
        return 1
        
    } elseif (Key=="up" | Key=="down") {
        local Ind = Index
        if (Key=="down") {
            if (Index >= 1 & Index <= 3) { Index = 4 }      #      Max Size
            elseif (Index == 4) { Index = 5 }               #  Max Size   Keep AR
            elseif (Index == 5) { Index = 6 }               #  Keep AR    OK
            elseif (Index >= 6 & Index <= 7) { Index = 1 }  #      Normal
        } else { # Key == "up"
            if (Index >= 1 & Index <= 3) { Index = 6 }      #     OK
            elseif (Index == 4) { Index = 1 }               #  Max Size   Normal
            elseif (Index == 5) { Index = 4 }               #  Keep AR   Max Size
            elseif (Index >= 6 & Index <= 7) { Index = 5 }  #      Keep AR
        }
        DialogData[-5,number] = Index
        DialogData[11,number] = Index
        
        local CurID = DialogData[-4,table][Index,number]     
        local PrevID = DialogData[-4,table][Ind,number]  
        
        if (Ind <= 3) {
            local OldCol = (JpgMode == (Ind-1) ? vec4(0,168,168,255) : vec4(168,168,168,255))
            EGP:egpColor(PrevID, OldCol)
        } elseif (Ind >= 6) {
            EGP:egpColor(PrevID, vec4(168,168,168,255))
        }
        
        if (Index <= 3 | Index >= 6) { EGP:egpColor(CurID, vec4(0,168,0,255)) }
        
        if (Index == 4 | Index == 5) {
            local TxtTbl = DialogData[-4,table][Index,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string]+"_",EGP:egpPos(TxtTbl[1,number]))
        }
        if (Ind == 4 | Ind == 5) {
            local TxtTbl = DialogData[-4,table][Ind,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string],EGP:egpPos(TxtTbl[1,number]))
        }
        return 1        
    } elseif (Key=="enter") {     
        
        if (Index >= 1 & Index <= 3) {
            local PrevActiveModeID = DialogData[-4,table][JpgMode+1,number]
            EGP:egpColor(PrevActiveModeID, vec4(168,168,168,255))   
            
            JpgMode = Index - 1
            GpuPersistData["jpgMode",number] = JpgMode
            
            local NewActiveModeID = DialogData[-4,table][Index,number]
            EGP:egpColor(NewActiveModeID, vec4(0,168,0,255))
            return 1
        }
        
        if (Index == 7) {
            alxos_jpg_key("f8",1)
            return 1
        }
        
        if (Index == 6) {
            local InpMaxSize = DialogData[-4,table][4,table][2,string]:toNumber()
            local InpKeepAR = DialogData[-4,table][5,table][2,string]
            
            local MaxSize = gpuScreenSize(DS)
            
            if (InpMaxSize==0) { InpMaxSize = MaxSize }
            if (InpKeepAR=="") { InpKeepAR = "1" }
            
            GpuPersistData["jpgMaxSize",number] = clamp(InpMaxSize,1,MaxSize)
            GpuPersistData["jpgNoKeepAR",number] = (InpKeepAR=="1" ? 0 : 1)
            
            "alxos_explorer_closedialog"()
            alxos_img_postinit()
            return 1
        }
        
        return 0
        
    } else {
        if (Index == 4 | Index == 5) {
            local ITbl = DialogData[-4,table]
            local D_I = ITbl[Index,table][1,number]
            local D_V = ITbl[Index,table][2,string]
            
            if (Key=="backspace") {
                D_V = D_V:sub(0,D_V:length()-1)
                EGP:egpSetText(D_I,D_V+"_")
                ITbl[Index,table][2,string] = D_V
            } elseif (Key=="delete") {
                EGP:egpSetText(D_I,"_")
                ITbl[Index,table][2,string] = ""
            } elseif (Pressed==2) {
                local KMem = toByte(Key)
                local Char = toChar((KMem>=128 ? KMem-80 : KMem))
                local RE = ITbl[Index,table][3,string]
                local MC = ITbl[Index,table][4,number]
                
                if (Char:findRE(RE) & (KMem<=122 | KMem>=128 & KMem<=143)) {
                    if (D_V:length() < MC) {
                        D_V = (D_V+Char)
                        EGP:egpSetText(D_I,D_V+"_")
                        ITbl[Index,table][2,string] = D_V
                    }
                }
            }
        }
    }
    return 0
}
