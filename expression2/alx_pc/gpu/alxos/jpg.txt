function array alxos_jpg_findAccelerator() {
    
    local PCI_Devices = Devices["pci_bus",table]
            
    foreach(K:number,V:array=PCI_Devices) {
        if (V[1,string]=="imageacc") {
            return V    
        }
    }
    
    return array()
}

function alxos_jpgread_progress() {
    local ExplorerFS = GpuData[1,table]
    
    local FS = ExplorerFS["hdd_io",wirelink]
    local Step = DialogData[4,number]

    local CurTask = FS["CurTask",table]
    local I = CurTask[2,number]
    local Max = CurTask[1,number]
    local Last = DialogData[3,number]
    
    if (!Step) { return }

    if (Max>0) {
        local PercID = DialogData[-4,number]
        local BoxID = DialogData[-5,number]
        local BoxW = DialogData[-6,number]
        local CalcP = round(I/Max*100)
        BoxW = ceil(BoxW/100*CalcP)
        if (BoxW<1) { BoxW = 1 }
        EGP:egpSize(BoxID,vec2(BoxW,20))
        EGP:egpSetText(PercID,CalcP+"%")
    }
    
    runCmd("alxos_jpgread_progress",1) 
}

function number alxos_jpg_clear() {
    if (!checkPerf(7000)) {
        runCmd("alxos_jpg_clear",1)
        return 0
    }
    
    local ExplorerFS = GpuData[1,table]
    dsSendDirect("clear",ExplorerFS["hdd_io",wirelink]:entity(),1)
    
    local IMGAccelerator = alxos_jpg_findAccelerator()
    dsSendDirect("SHUTDOWN",IMGAccelerator[3,entity],0)

    "alxos_explorer_handlers"()
    "alxos_explorer_listcurdir"()
    return 0
}

    return 0
}

function number alxos_jpg_finish(Err:string) {
    local Height = 60
    local Pos = vec2(225,18)
    local NewLines = Err:explode("\n") 
    
    if (NewLines:count()>1) {
        Height += (NewLines:count()-1) * 20
    }
    
    if (Err) {
        Pos = vec2(20,18)    
    }
        
    local Parent = "alxos_explorer_dialog"("JPG Reader"+(Err!="" ? " Error" : ""),450,Height,"ok","alxos_jpg_clear")[number]

    if (Err) {
        for(I=1,NewLines:count()) {
            DialogData[-10,table]:pushNumber(LAST_I)
            EGP:egpText(LAST_I,NewLines[I,string],Pos)
            EGP:egpParent(LAST_I,Parent)
            EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
            
            Pos[2] = Pos[2] + 20
        }
    } else {
        local JPGAccelerator = GpuData["JPGAcceleratorWirelink",wirelink]
        local Progress = JPGAccelerator["Progress",table]
        local RenderTime = Progress["renderTime",number]
        
        DialogData[-10,table]:pushNumber(LAST_I)
        EGP:egpText(LAST_I,"Opening JPG file success finished, render time: "+RenderTime+"s.",Pos)
        EGP:egpParent(LAST_I,Parent)
        if (!Err) {
            EGP:egpAlign(LAST_I,1)
        }
        EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    }

    "alxos_explorer_handlers"()
    return 0
}

function number alxos_jpg_finish() {
    if (!checkPerf(7000)) {
        runCmd("alxos_jpg_finish",1)
        return 0
    }
    return alxos_jpg_finish("")
}

function alxos_jpg_warning(Warning:string) {
    DialogData["WarningShown",number] = 1

    local Lines = Warning:explode("\n"):count()*2

    local Width = DialogData["DialogW",number]
    local Height = 90
    local Title = "Warning"
    
    local PosX = 256 - (Width / 2)

function alxos_jpg_progress() {    
    if (DS==nowirelink()) {
        local IMGAccelerator = alxos_jpg_findAccelerator()
        dsSendDirect("SHUTDOWN",IMGAccelerator[3,entity],0)   
        
        "alxos_explorer_error"("Digital screen not connected! Rendering canceled.",420) 
        return
    }
    
    if (CMDLast!="alxos_jpg_progress") {
        runCmd("alxos_jpg_progress",100,1)
        return 
    }

    local JPGAccelerator = GpuData["JPGAcceleratorWirelink",wirelink]
    
    local CurrentState = JPGAccelerator["CurrentState",string]
    local Progress = JPGAccelerator["Progress",table]
    local ProgressText = JPGAccelerator["ProgressText",string]
    local Processing = JPGAccelerator["Processing",number]
    local Error = JPGAccelerator["Error",string]
    
    if (GpuData["JPGInit",number] & Processing==0 | Error) {
        alxos_jpg_finish(Error)
        return
    }
    
    local Warning = JPGAccelerator["Warning",string]
    if (Warning & !DialogData["WarningShown",number]) {
        alxos_jpg_warning(JPGAccelerator["Warning",string])
    }
    
    local TW = DialogData[3,number]
        
    EGP:egpSetText(TW-2, "Current Step: "+Progress["step",number] + " / "+Progress["stepTotal",number])
    
    local PercID = DialogData[5,number]
    local BoxID = DialogData[4,number]
    local BoxW = DialogData[-6,number]
    local CalcP = round(Progress["step",number] * 100 / Progress["stepTotal",number])
    
    BoxW = ceil(BoxW/100*CalcP)
    if (BoxW<1) { BoxW = 1 }
    EGP:egpSize(BoxID,vec2(BoxW,20))
    EGP:egpSetText(PercID,CalcP+"%")
    
    local PercID = DialogData[-4,number]
    local BoxID = DialogData[-5,number]
    local BoxW = DialogData[-6,number]
    local CalcP = Progress["progressPercent",number]
    
    EGP:egpSetText(TW-1, "Current Task: "+Progress["progressTask",string])
    
    BoxW = ceil(BoxW/100*CalcP)
    if (BoxW<1) { BoxW = 1 }
    EGP:egpSize(BoxID,vec2(BoxW,20))
    EGP:egpSetText(PercID,Progress["progressCurrent",number] + " / " + Progress["progressMax",number] + " (" + CalcP+"%)")
    
    runCmd("alxos_jpg_progress",100)    
}

function alxos_jpg_ds(Name:string) {
    if (Name=="fs_readfile") {
        alxos_jpg_draw(dsGetString())
    } elseif (Name=="clear") {
        if (!checkPerf(7000)) {
            runCmd("alxos_jpg_clear",1)
            return
        }
        alxos_jpg_clear()
    } elseif (Name=="imageReaderError") {
        "alxos_explorer_error"("JPG Reader Chip not connected (crashed?)",420)     
    } elseif (Name=="initImage") {
        local Info = dsGetTable()        
        GpuData["JPGAcceleratorWirelink",wirelink] = Info[1,wirelink]
        GpuData["JPGInit",number] = 1
        runCmd("alxos_jpg_progress",100,1)
    }
}

function number alxos_jpg_key(Key:string,Pressed) {
    if (!Pressed) { return 0 }
    
    local Index = DialogData[-5,number]
    local Type = DialogData[-2,string]
    local JpgMode = GpuPersistData["jpgMode",number]
    
    local Title = DialogData["title",string]   
    
    if (Key=="f8") {
        stoptimer("cmd")
        dsSendDirect("clear",GpuData[1,table]["hdd_io",wirelink]:entity(),entity())
        local IMGAccelerator = alxos_jpg_findAccelerator()
        dsSendDirect("SHUTDOWN",IMGAccelerator[3,entity],0)  
        alxos_jpg_close()
        return 1
    } elseif (Key=="enter" & Title!="Opening file" & Title!="JPG Settings") {
        alxos_jpg_key("f8",1)
        return 1
    }
    
    if (Title!="JPG Settings") { return 0 }
    
    if (Key=="left" | Key=="right" | Key=="tab") {
        local Max = 7
        local Ind = Index
        if (Key=="right" | Key=="tab") {
            Index = (Index==Max ? 1 : Index+1)
        } else {
            Index = (Index==1 ? Max : Index-1)
        }
        DialogData[-5,number] = Index
        DialogData[11,number] = Index
        
        local CurID = DialogData[-4,table][Index,number]     
        local PrevID = DialogData[-4,table][Ind,number]  
        
        if (Ind <= 3) {
            local OldCol = (JpgMode == (Ind-1) ? vec4(0,168,168,255) : vec4(168,168,168,255))
            EGP:egpColor(PrevID, OldCol)
        } elseif (Ind >= 6) {
            EGP:egpColor(PrevID, vec4(168,168,168,255))
        }
        
        if (Index <= 3 | Index >= 6) { EGP:egpColor(CurID, vec4(0,168,0,255)) }
        
        # "_"
        if (Index == 4 | Index == 5) {
            local TxtTbl = DialogData[-4,table][Index,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string]+"_",EGP:egpPos(TxtTbl[1,number]))
        }
        if (Ind == 4 | Ind == 5) {
            local TxtTbl = DialogData[-4,table][Ind,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string],EGP:egpPos(TxtTbl[1,number]))
        }
        return 1
        
    } elseif (Key=="up" | Key=="down") {
        local Ind = Index
        if (Key=="down") {
            if (Index >= 1 & Index <= 3) { Index = 4 }      #      Max Size
            elseif (Index == 4) { Index = 5 }               #  Max Size   Keep AR
            elseif (Index == 5) { Index = 6 }               #  Keep AR    OK
            elseif (Index >= 6 & Index <= 7) { Index = 1 }  #      Normal
        } else { # Key == "up"
            if (Index >= 1 & Index <= 3) { Index = 6 }      #     OK
            elseif (Index == 4) { Index = 1 }               #  Max Size   Normal
            elseif (Index == 5) { Index = 4 }               #  Keep AR   Max Size
            elseif (Index >= 6 & Index <= 7) { Index = 5 }  #      Keep AR
        }
        DialogData[-5,number] = Index
        DialogData[11,number] = Index
        
        local CurID = DialogData[-4,table][Index,number]     
        local PrevID = DialogData[-4,table][Ind,number]  
        
        if (Ind <= 3) {
            local OldCol = (JpgMode == (Ind-1) ? vec4(0,168,168,255) : vec4(168,168,168,255))
            EGP:egpColor(PrevID, OldCol)
        } elseif (Ind >= 6) {
            EGP:egpColor(PrevID, vec4(168,168,168,255))
        }
        
        if (Index <= 3 | Index >= 6) { EGP:egpColor(CurID, vec4(0,168,0,255)) }
        
        if (Index == 4 | Index == 5) {
            local TxtTbl = DialogData[-4,table][Index,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string]+"_",EGP:egpPos(TxtTbl[1,number]))
        }
        if (Ind == 4 | Ind == 5) {
            local TxtTbl = DialogData[-4,table][Ind,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string],EGP:egpPos(TxtTbl[1,number]))
        }
        return 1        
    } elseif (Key=="enter") {     
        
        if (Index >= 1 & Index <= 3) {
            local PrevActiveModeID = DialogData[-4,table][JpgMode+1,number]
            EGP:egpColor(PrevActiveModeID, vec4(168,168,168,255))   
            
            JpgMode = Index - 1
            GpuPersistData["jpgMode",number] = JpgMode
            
            local NewActiveModeID = DialogData[-4,table][Index,number]
            EGP:egpColor(NewActiveModeID, vec4(0,168,0,255))
            return 1
        }
        
        if (Index == 7) {
            alxos_jpg_key("f8",1)
            return 1
        }
        
        if (Index == 6) {
            local InpMaxSize = DialogData[-4,table][4,table][2,string]:toNumber()
            local InpKeepAR = DialogData[-4,table][5,table][2,string]
            
            local MaxSize = gpuScreenSize(DS)
            
            if (InpMaxSize==0) { InpMaxSize = MaxSize }
            if (InpKeepAR=="") { InpKeepAR = "1" }
            
            GpuPersistData["jpgMaxSize",number] = clamp(InpMaxSize,1,MaxSize)
            GpuPersistData["jpgNoKeepAR",number] = (InpKeepAR=="1" ? 0 : 1)
            
            "alxos_explorer_closedialog"()
            alxos_jpg_postinit()
            return 1
        }
        
        return 0
        
    } else {
        if (Index == 4 | Index == 5) {
            local ITbl = DialogData[-4,table]
            local D_I = ITbl[Index,table][1,number]
            local D_V = ITbl[Index,table][2,string]
            
            if (Key=="backspace") {
                D_V = D_V:sub(0,D_V:length()-1)
                EGP:egpSetText(D_I,D_V+"_")
                ITbl[Index,table][2,string] = D_V
            } elseif (Key=="delete") {
                EGP:egpSetText(D_I,"_")
                ITbl[Index,table][2,string] = ""
            } elseif (Pressed==2) {
                local KMem = toByte(Key)
                local Char = toChar((KMem>=128 ? KMem-80 : KMem))
                local RE = ITbl[Index,table][3,string]
                local MC = ITbl[Index,table][4,number]
                
                if (Char:findRE(RE) & (KMem<=122 | KMem>=128 & KMem<=143)) {
                    if (D_V:length() < MC) {
                        D_V = (D_V+Char)
                        EGP:egpSetText(D_I,D_V+"_")
                        ITbl[Index,table][2,string] = D_V
                    }
                }
            }
        }
    }
    return 0
}
