function alxos_bmp_postinit() {
    if (!DS:entity():isValid()) {
        if (GpuData["lastProg",string]=="alxos_console") {
            "alxos_console_err"("Digital screen not connected!\nRendering canceled.")
        } else {
            "alxos_explorer_error"("Digital screen not connected! Rendering canceled.",420)
        }
        return
    }

    local FInfo = GpuData[2,table]
    local ExplorerFS = GpuData[1,table]
    
    DsHandler = "alxos_bmp_ds"
    #KeyHandler = "alxos_bmp_key"
    ExplorerCurTask = "bmp"
    
    DS[1048574] = 0
    DS[1048569] = 2
    DS[1048575] = 1
    
    DSEvent = function() {
        bmpSetDS(DS)    
    }
    
    if (!FInfo["content",string]) {        
        dsSendDirect("fs_readfile",ExplorerFS["hdd_io",wirelink]:entity(),table(entity(),ExplorerFS["fs_str_id",string],table(
            ExplorerFS,FInfo,0,66
        )))
        
        GpuData["bmpFull",number] = 0
    }
        
    if (GpuData["lastProg",string]=="alxos_console") {
        if (FInfo["content",string]=="") {
            "alxos_console_msg"("Opening file... Press ENTER or F8 to cancel")
        }
        "alxos_console_msg"("0%")
        "alxos_console_move"()
    } else {
        local Width = 400
        local Height = 80
    
        local Parent = "alxos_explorer_dialog"("Opening file",Width,Height,"progress","alxos_img_clear")[number]
        
        DialogData["DialogW",number] = Width
        DialogData["DialogH",number] = Height
        
        GpuData[5,number] = 0
    
        DialogData[-10,table]:pushNumber(LAST_I)
        EGP:egpText(LAST_I,"Please wait, this can take few minutes...",alxos_dlgPos(0,-22))
        EGP:egpParent(LAST_I,Parent)
        EGP:egpAlign(LAST_I,1)
        EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    }
    
    GpuData["bmpState", string] = "header"
    
    BmpNoPrint = 1
    
    local Body = ""
    local Size = GpuData[2, table]["size", number]
    
    if (FInfo["content",string]) {
        Body = FInfo["content",string]
        GpuData["bmpState", string] = ""
        Size = Body:length()
    }
    
    bmpInit(Body, Size, GpuPersistData["bmpMaxSize",number], GpuPersistData["bmpNoKeepAR",number])
    bmpSetDS(DS)
    
    if (Body) {        
        GpuData["bmpFull",number] = 1
        
        registerTimers("bmp","bmpStep")
        bmpRun()
        
        runCmd("alxos_bmp_progress_full",10)
    }
}

function number alxos_bmp_finish(Err:string) {
    if (GpuData["lastProg",string]=="alxos_console") {
        if (Err) {
            "alxos_console_err"(Err)
        } else {
            local Index = DialogData["WarningShownIndex",number]
            "alxos_console_redraw_msg"(Index,"100%")
            "alxos_console_msg"("Opening BMP file success finished.\nRender time: "+round(BmpRenderTime,2)+"s.")
        }
        
        "alxos_console_handlers"()
        "alxos_console_command"()
        return 0
    }
    
    local Parent = "alxos_explorer_dialog"("BMP Reader"+(Err!="" ? " Error" : ""),450,60,"ok","alxos_img_clear")[number]

    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,(Err!="" ? Err : "Opening BMP file success finished, render time: "+round(BmpRenderTime,2)+"s."),alxos_dlgPos(0,-12))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    
    BmpStep = 0
    BmpData = table()

    return 0
}

function number alxos_bmp_finish() {
    if (!checkPerf(7000)) {
        runCmd("alxos_bmp_finish",1)
        return 0
    }
    return alxos_bmp_finish("")
}

function alxos_bmp_progress_full() {
    local CalcP = BmpPercent
    
    CalcP = round(CalcP,2)

    if (CalcP > 100) {
        CalcP = 100
    }

    local PercID = DialogData[-4,number]
    local BoxID = DialogData[-5,number]
    local BoxW = DialogData[-6,number]

    BoxW = ceil(BoxW / 100 * CalcP)

    if (BoxW < 1) {
        BoxW = 1
    }

    if (GpuData["lastProg",string]=="alxos_console") {
        local Index = DialogData["WarningShownIndex",number]
        "alxos_console_redraw_msg"(Index,CalcP+"%")
    } else {
        EGP:egpSize(BoxID,vec2(BoxW,20))
        EGP:egpSetText(PercID,CalcP+"%")
    }

    if (CalcP<100) {
        runCmd("alxos_bmp_progress_full",10)
    } else {
        alxos_bmp_finish()    
    }
}

function alxos_bmp_progress() {
    local FInfo = GpuData[2,table]
    local FileSize = FInfo["size",number]

    local ExplorerFS = GpuData[1,table]
    local FS = ExplorerFS["hdd_io",wirelink]

    local State = GpuData["bmpState",string]
    local HasRle = BmpData["hasRLE",number]

    local CalcP = 0

    if (State=="header" | State=="read_pallete") {

        local CurTask = FS["CurTask",table]
        local I = CurTask[2,number]
        local Max = CurTask[1,number]

        local DiskP = 0

        if (Max > 0) {
            DiskP = I / Max
        }
        
        if (State=="header") {
            CalcP = DiskP * 2.5
        } else {
            CalcP = 2.5 + DiskP * 2.5
        }

    } elseif ((State=="rle_processing" | State=="rle_readfile") & HasRle) {
        
        local CurTask = FS["CurTask",table]
        local I = CurTask[2,number]
        local Max = CurTask[1,number]
        
        local DiskP = 0
        if (Max > 0) {
            DiskP = I / Max
        }
    
        if (State=="rle_readfile") {
            CalcP = 5 + DiskP * 25
        } else {
            CalcP = 30 + BmpPercent / 100 * 25
        }
        
    } elseif (State=="process_row") {
        
        local CurTask = FS["CurTask",table]
        local I = CurTask[2,number]
        local Max = CurTask[1,number]
    
        if (Max>0) {
            local DiskP = I/Max*100
            
            local StartP = GpuData["bmpChunkStartP",number]
            local ChunkP = GpuData["bmpChunkSizeP",number]/2
            
            CalcP = StartP + ChunkP * DiskP / 100
        }
        
        # not implemented
    } elseif (State=="draw") {
        local StartP = GpuData["bmpChunkStartP",number]
        local ChunkP = GpuData["bmpChunkSizeP",number]/2
        
        local SI = BmpData[9,number]
        local Max = BmpData[10, number]
        
        local DrawP = SI / Max
        
        CalcP = StartP + ChunkP + ChunkP * DrawP
    } elseif (State=="draw_rle") {
        CalcP = 55 + BmpPercent/100 * 45        
    }

    local LastP = GpuData["bmpProgress",number]

    if (CalcP < LastP) {
        CalcP = LastP
    } else {
        GpuData["bmpProgress",number] = CalcP
    }

    CalcP = round(CalcP,2)

    if (CalcP > 100) {
        CalcP = 100
    }

    local PercID = DialogData[-4,number]
    local BoxID = DialogData[-5,number]
    local BoxW = DialogData[-6,number]

    BoxW = ceil(BoxW / 100 * CalcP)

    if (BoxW < 1) {
        BoxW = 1
    }

    if (GpuData["lastProg",string]=="alxos_console") {
        local Index = DialogData["WarningShownIndex",number]
        "alxos_console_redraw_msg"(Index,CalcP+"%")
    } else {
        EGP:egpSize(BoxID,vec2(BoxW,20))
        EGP:egpSetText(PercID,CalcP+"%")
    }

    if (CalcP<100) {
        runCmd("alxos_bmp_progress",10)
    }
}

function number alxos_bmp_step() {
    alxos_bmp_progress()
    
    local State = GpuData["bmpState", string]
    
    if (State=="header") {
        local HeaderResult = bmpParseHeader()
        
        if (HeaderResult==0) {
            if (BmpError) {
                alxos_bmp_finish(BmpError)
            }
            return 0
        }
        
        if (BmpWarning & !DialogData["WarningShown",number]) {
            alxos_img_warning(BmpWarning)
        }
            
        GpuData["realRowBytes", number] = BmpData[28, number]
        GpuData["hddOffset", number] = BmpData[8, number]
        
        if (BmpCurrentStep=="read_pallete") {
            GpuData["bmpState", string] = "read_pallete"
            
            local Colors  = BmpData[23, number]
            local ColorsB = BmpData[5, number]
            local FInfo   = GpuData[2, table]
            local ExplorerFS = GpuData[1, table]
            
            local TotalPaletteBytes = Colors * ColorsB
            GpuData[28,number] = TotalPaletteBytes
            BmpData[10,number] = TotalPaletteBytes
            
            local PaletteOffset = BmpData[8, number]
            BmpData[8, number] = 1
                    
            dsSendDirect("fs_readfile",ExplorerFS["hdd_io",wirelink]:entity(),table(
                entity(),ExplorerFS["fs_str_id",string],table(ExplorerFS,FInfo,PaletteOffset,TotalPaletteBytes)
            ))
        
            return 0
        }
        
        GpuData["bmpState", string] = "request_row"
        return alxos_bmp_step()
    } elseif (State=="read_pallete") {
        local IsPaletteDone = bmpParsePallete()
        
        if (IsPaletteDone==0) {
            runCmd("alxos_bmp_step",1)
            return 0
        } elseif (IsPaletteDone==2) {
            GpuData["bmpState", string] = "rle_readfile"

            local FInfo = GpuData[2, table]
            local ExplorerFS = GpuData[1, table]
            local FullSize = FInfo["size", number]

            dsSendDirect("fs_readfile", ExplorerFS["hdd_io", wirelink]:entity(), table(
                entity(), ExplorerFS["fs_str_id", string], table(ExplorerFS, FInfo, 0, FullSize)
            ))
            
            return 0    
        }
        
        BmpData[10, number] = GpuData["realRowBytes", number]        
        GpuData["hddOffset", number] = BmpData[29, number]
        GpuData["bmpState", string] = "request_row"
        return alxos_bmp_step()
    } elseif (State=="request_row") {
        local FInfo = GpuData[2,table]
        local ExplorerFS = GpuData[1,table]
        
        local CurrentOffset = GpuData["hddOffset", number]
        local RowBytes = GpuData["realRowBytes", number]
    
        local ChunkLines = clamp(floor(8 * sqrt(softQuota() / 10000)),8,128)
        
        local BufferSize = RowBytes * ChunkLines
        
        local TotalSize = FInfo["size", number]
        
        local IsLastChunk = 0
    
        if (CurrentOffset + BufferSize > TotalSize) {
            BufferSize = TotalSize - CurrentOffset
            IsLastChunk = 1
        }
    
        GpuData["bmpLastChunk", number] = IsLastChunk
        BmpData[10, number] = BufferSize
    
        GpuData["bmpState", string] = "process_row"
        
        if (!BmpData["drawInit",number]) {
            bmpDrawInit()
        }
        
        GpuData["bmpProgressMode",number] = 0
        
        local PixelDataSize = GpuData["realRowBytes",number] * BmpData[4,number]
        GpuData["bmpChunkStartP",number] = GpuData["bmpProgress",number]
        GpuData["bmpChunkSizeP",number] = BufferSize / PixelDataSize * 95
        
        dsSendDirect("fs_readfile", ExplorerFS["hdd_io",wirelink]:entity(), 
            table(entity(), ExplorerFS["fs_str_id",string], table(ExplorerFS, FInfo, CurrentOffset, BufferSize))
        )
        
        return 0
    } elseif (State=="process_row") {
        GpuData["bmpProgressMode",number] = 1
        BmpData["max", number] = BmpData[4, number]
        BmpData[8, number] = 1
        BmpData[9, number] = 0
        GpuData["bmpState", string] = "draw"
        runCmd("alxos_bmp_draw",1)
    } elseif (State=="rle_readfile") {
        GpuData["bmpState", string] = "rle_processing"
        return alxos_bmp_step()
    } elseif (State=="rle_processing") {    
        if (!bmp_decompress_rle()) {
            runCmd("alxos_bmp_step",1)
            return 0
        }
        
        BmpData[8, number] = BmpData["RleOffset", number] 
        BmpData[9, number] = 0
        BmpData[10, number] = BmpData["data", string]:length()
        BmpData["max", number] = BmpData[4, number]
        
        GpuData["bmpState",string] = "draw_rle"
        
        if (!BmpData["drawInit",number]) {
            bmpDrawInit()
        }
        
        runCmd("alxos_bmp_draw", 1)
        return 0
    }
    
    return 0
}

function number alxos_bmp_draw() {    
    local IsDone = bmpDraw()
    
    alxos_bmp_progress()
    
    if (IsDone) {
        local IsLastChunk = GpuData["bmpLastChunk", number]
        
        if (IsLastChunk) {        
            GpuData["bmpState", string] = "finished"
            bmpFinish()
            alxos_bmp_finish()
            stoptimer("cmd")
            return 1
        }
        
        GpuData["hddOffset", number] = GpuData["hddOffset", number] + BmpData[10, number]
        
        GpuData["bmpState", string] = "request_row"
        alxos_bmp_step()
    } else {
        runCmd("alxos_bmp_draw",1)
    }
    return 0
}

function alxos_bmp_step(Data:string) {    
    BmpData["data", string] = Data
    alxos_bmp_step()        
}

function alxos_bmp_ds(Name:string) {
    if (Name=="fs_readfile") {
        alxos_bmp_step(dsGetString())
    } elseif (Name=="clear") {
        if (!checkPerf(7000)) {
            runCmd("alxos_img_clear",1)
            return
        }
        alxos_img_clear()
    }
}

function number alxos_bmp_key(Key:string,Pressed) {
    if (!Pressed) { return 0 }
    
    local Index = DialogData[-5,number]
    local Type = DialogData[-2,string]
    
    local Title = DialogData["title",string]   
    
    if (Key=="f8") {
        stoptimer("cmd")
        dsSendDirect("clear",GpuData[1,table]["hdd_io",wirelink]:entity(),1)
        alxos_img_close()
        return 1
    } elseif (Key=="enter" & Title!="Opening file" & Title!="BMP Settings") {
        alxos_bmp_key("f8",1)
        return 1
    }
    
    if (Title!="BMP Settings") { return 0 }
    
    if (Key=="left" | Key=="right" | Key=="tab") {
        local Max = 4
        local Ind = Index
        if (Key=="right" | Key=="tab") {
            Index = (Index==Max ? 1 : Index+1)
        } else {
            Index = (Index==1 ? Max : Index-1)
        }
        DialogData[-5,number] = Index
        DialogData[11,number] = Index
        
        local CurID = DialogData[-4,table][Index,number]     
        local PrevID = DialogData[-4,table][Ind,number]  
        
        if (Ind >= 3) {
            EGP:egpColor(PrevID, vec4(168,168,168,255))
        }
        
        if (Index >= 3) { EGP:egpColor(CurID, vec4(0,168,0,255)) }
        
        # "_"
        if (Index == 1 | Index == 2) {
            local TxtTbl = DialogData[-4,table][Index,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string]+"_",EGP:egpPos(TxtTbl[1,number]))
        }
        if (Ind == 1 | Ind == 2) {
            local TxtTbl = DialogData[-4,table][Ind,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string],EGP:egpPos(TxtTbl[1,number]))
        }
        return 1
        
    } elseif (Key=="up" | Key=="down") {
        local Ind = Index
        if (Key=="down") {
            if (Index < 4) { Index++ } 
            elseif (Index >= 4) { Index = 1 }
        } else { # Key == "up"
            if (Index <= 1) { Index = 4 } 
            elseif (Index > 1) { Index-- }
        }
        DialogData[-5,number] = Index
        DialogData[11,number] = Index
        
        local CurID = DialogData[-4,table][Index,number]     
        local PrevID = DialogData[-4,table][Ind,number]  
        
        if (Ind >= 3) {
            EGP:egpColor(PrevID, vec4(168,168,168,255))
        }
        
        if (Index >= 3) { EGP:egpColor(CurID, vec4(0,168,0,255)) }
        
        if (Index == 1 | Index == 2) {
            local TxtTbl = DialogData[-4,table][Index,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string]+"_",EGP:egpPos(TxtTbl[1,number]))
        }
        if (Ind == 1 | Ind == 2) {
            local TxtTbl = DialogData[-4,table][Ind,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string],EGP:egpPos(TxtTbl[1,number]))
        }
        return 1        
    } elseif (Key=="enter") {     
        
        if (Index == 4) {
            alxos_bmp_key("f8",1)
            return 1
        }
        
        if (Index == 3) {
            local InpMaxSize = DialogData[-4,table][1,table][2,string]:toNumber()
            local InpKeepAR = DialogData[-4,table][2,table][2,string]
            
            local MaxSize = gpuScreenSize(DS)
            
            if (InpMaxSize==0) { InpMaxSize = MaxSize }
            if (InpKeepAR=="") { InpKeepAR = "1" }
            
            GpuPersistData["bmpMaxSize",number] = clamp(InpMaxSize,1,MaxSize)
            GpuPersistData["bmpNoKeepAR",number] = (InpKeepAR=="1" ? 0 : 1)
            
            "alxos_explorer_closedialog"()
            alxos_bmp_postinit()
            return 1
        }
        
        return 0
        
    } else {
        if (Index == 1 | Index == 2) {
            local ITbl = DialogData[-4,table]
            local D_I = ITbl[Index,table][1,number]
            local D_V = ITbl[Index,table][2,string]
            
            if (Key=="backspace") {
                D_V = D_V:sub(0,D_V:length()-1)
                EGP:egpSetText(D_I,D_V+"_")
                ITbl[Index,table][2,string] = D_V
            } elseif (Key=="delete") {
                EGP:egpSetText(D_I,"_")
                ITbl[Index,table][2,string] = ""
            } elseif (Pressed==2) {
                local KMem = toByte(Key)
                local Char = toChar((KMem>=128 ? KMem-80 : KMem))
                local RE = ITbl[Index,table][3,string]
                local MC = ITbl[Index,table][4,number]
                
                if (Char:findRE(RE) & (KMem<=122 | KMem>=128 & KMem<=143)) {
                    if (D_V:length() < MC) {
                        D_V = (D_V+Char)
                        EGP:egpSetText(D_I,D_V+"_")
                        ITbl[Index,table][2,string] = D_V
                    }
                }
            }
        }
    }
    
    return 0
}
