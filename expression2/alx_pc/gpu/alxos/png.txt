function array alxos_png_findAccelerator() {
    
    local PCI_Devices = Devices["pci_bus",table]
            
    foreach(K:number,V:array=PCI_Devices) {
        if (V[1,string]=="imageacc") {
            return V    
        }
    }
    
    return array()
}

function alxos_png_init(Args:table) {

function alxos_prog_png_unplug() {
    stopAllTimers()
    ExplorerDsH = ""
    "alxos_explorer_handlers"()
    "alxos_explorer_msg"("Disk was unplugged.",200,"Information")
}

function alxos_png_postinit() {

    DsHandler = "alxos_png_ds"

    ExplorerCurTask = "alxos_prog_png"
    
    local ExplorerFS = GpuData[1,table]
    local FInfo = GpuData[2,table]

    DS[1048574] = 0
    DS[1048569] = 2
    DS[1048575] = 1

    if (FInfo["content",string]) {
        "alxos_png_draw"(FInfo["content",string])
        return   
    }

    dsSendDirect("fs_readfile",ExplorerFS["hdd_io",wirelink]:entity(),table(entity(),ExplorerFS["fs_str_id",string],table(
        ExplorerFS,FInfo,0,FInfo["size",number]
    )))     

    local Parent = "alxos_explorer_dialog"("Opening file",400,80,"progress","alxos_png_clear")[number]

    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"Please wait, this can take few minutes...",alxos_dlgPos(0,-22))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    
    DialogData[1,number] = 1
    DialogData[3,number] = 1
    DialogData[4,number] = 1
    runCmd("alxos_pngread_progress",1)
}

function alxos_pngread_progress() {
    local ExplorerFS = GpuData[1,table]
    
    local FS = ExplorerFS["hdd_io",wirelink]
    local Step = DialogData[4,number]

    local CurTask = FS["CurTask",table]
    local I = CurTask[2,number]
    local Max = CurTask[1,number]
    local Last = DialogData[3,number]
    
    if (!Step) { return }

    if (Max>0) {
        local PercID = DialogData[-4,number]
        local BoxID = DialogData[-5,number]
        local BoxW = DialogData[-6,number]
        local CalcP = round(I/Max*100)
        BoxW = ceil(BoxW/100*CalcP)
        if (BoxW<1) { BoxW = 1 }
        EGP:egpSize(BoxID,vec2(BoxW,20))
        EGP:egpSetText(PercID,CalcP+"%")
    }
    
    runCmd("alxos_pngread_progress",1) 
}

function number alxos_png_clear() {
    if (!checkPerf(7000)) {
        runCmd("alxos_png_clear",1)
        return 0
    }
    
    local ExplorerFS = GpuData[1,table]
    dsSendDirect("clear",ExplorerFS["hdd_io",wirelink]:entity(),1)
    
    local IMGAccelerator = alxos_png_findAccelerator()
    dsSendDirect("SHUTDOWN",IMGAccelerator[3,entity],0)

    "alxos_explorer_handlers"()
    "alxos_explorer_listcurdir"()
    return 0
}

function number alxos_png_close() {
    "alxos_explorer_handlers"()
    "alxos_explorer_listcurdir"()
    return 0
}

function number alxos_png_finish(Err:string) {
    local Height = 60
    local Pos = vec2(225,18)
    local NewLines = Err:explode("\n") 
    
    if (NewLines:count()>1) {
        Height += (NewLines:count()-1) * 20
    }
    
    if (Err) {
        Pos = vec2(20,18)    
    }
        
    local Parent = "alxos_explorer_dialog"("PNG Reader"+(Err!="" ? " Error" : ""),450,Height,"ok","alxos_png_clear")[number]

    if (Err) {
        for(I=1,NewLines:count()) {
            DialogData[-10,table]:pushNumber(LAST_I)
            EGP:egpText(LAST_I,NewLines[I,string],Pos)
            EGP:egpParent(LAST_I,Parent)
            EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
            
            Pos[2] = Pos[2] + 20
        }
    } else {
        local PNGAccelerator = GpuData["PNGAcceleratorWirelink",wirelink]
        local Progress = PNGAccelerator["Progress",table]
        local RenderTime = Progress["renderTime",number]
        
        DialogData[-10,table]:pushNumber(LAST_I)
        EGP:egpText(LAST_I,"Opening PNG file success finished, render time: "+RenderTime+"s.",Pos)
        EGP:egpParent(LAST_I,Parent)
        if (!Err) {
            EGP:egpAlign(LAST_I,1)
        }
        EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    }

    "alxos_explorer_handlers"()
    return 0
}

function number alxos_png_finish() {
    if (!checkPerf(7000)) {
        runCmd("alxos_png_finish",1)
        return 0
    }
    return alxos_png_finish("")
}

function alxos_png_draw(Data:string) {
    DialogData[4,number] = 0
    
    local IMGAccelerator = alxos_png_findAccelerator()
    
    if (!IMGAccelerator:count()) {
        "alxos_explorer_error"("Image Accelerator not found, please insert PCI card and reboot.",500)
        return
    }
    
    ExplorerCurTask = ""

    GpuData["imgAccelerator",array] = IMGAccelerator
    
    dsSendDirect("drawImage",IMGAccelerator[3,entity],table(
        DS, entity(), TickQuota, "png", table(
            Data, 
            GpuPersistData["pngMaxSize",number],
            GpuPersistData["pngNoKeepAR",number]
        )
    ))

    local Width = 400
    local Height = 120
    
    local LAST_Y = -22
    local LeftPos = 20-Width/2
    
    local Parent = "alxos_explorer_dialog"("Reading PNG",Width,Height,"progress","alxos_png_clear")[number]

    DialogData["DialogW",number] = Width
    DialogData["DialogH",number] = Height

    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"Current Step: 0 / 0",alxos_dlgPos(-180,-42))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) LAST_I++

    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"Current Task: Opening PNG",alxos_dlgPos(-180,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    DialogData[3,number] = LAST_I
    EGP:egpText(LAST_I,"",alxos_dlgPos(-3,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpBox(LAST_I,alxos_dlgPos(LeftPos,LAST_Y),vec2(Width-40,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,125,125,125,255) LAST_I++

    DialogData[4,number] = LAST_I
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpBox(LAST_I,alxos_dlgPos(LeftPos,LAST_Y),vec2(1,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    
    DialogData[5,number] = LAST_I
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"0%",alxos_dlgPos(0,LAST_Y))
    EGP:egpAlign(LAST_I,1)
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,200,200,200,255) LAST_I++
    
    # change position of main progress bar

    local BoxID = DialogData[-5,number]

    #EGP:egpPos(BoxID,alxos_dlgPos(LeftPos, LAST_Y))
    EGP:egpPos(PercID,alxos_dlgPos(0, LAST_Y+45))
}

function alxos_png_warning(Warning:string) {
    DialogData["WarningShown",number] = 1

    local Lines = Warning:explode("\n"):count()*2

    local Width = DialogData["DialogW",number]
    local Height = 90
    local Title = "Warning"
    
    local PosX = 256 - (Width / 2)
    local PosY = (256 + DialogData["DialogH",number] / 2) + 20
    
    alxos_dialog_warning(Title, Width, Height, vec2(PosX,PosY), Warning, Lines*20)
}

function alxos_png_progress() {    
    if (DS==nowirelink()) {
        local IMGAccelerator = alxos_png_findAccelerator()
        dsSendDirect("SHUTDOWN",IMGAccelerator[3,entity],0)   
        
        "alxos_explorer_error"("Digital screen not connected! Rendering canceled.",420) 
        return
    }
    
    if (CMDLast!="alxos_png_progress") {
        runCmd("alxos_png_progress",100,1)
        return 
    }

    local PNGAccelerator = GpuData["PNGAcceleratorWirelink",wirelink]
    
    local CurrentState = PNGAccelerator["CurrentState",string]
    local Progress = PNGAccelerator["Progress",table]
    local ProgressText = PNGAccelerator["ProgressText",string]
    local Processing = PNGAccelerator["Processing",number]
    local Error = PNGAccelerator["Error",string]
    
    if (GpuData["PNGInit",number] & Processing==0 | Error) {
        alxos_png_finish(Error)
        return
    }
    
    local Warning = PNGAccelerator["Warning",string]
    if (Warning & !DialogData["WarningShown",number]) {
        alxos_png_warning(PNGAccelerator["Warning",string])
    }
    
    local TW = DialogData[3,number]
        
    EGP:egpSetText(TW-2, "Current Step: "+Progress["step",number] + " / "+Progress["stepTotal",number])
    
    local PercID = DialogData[5,number]
    local BoxID = DialogData[4,number]
    local BoxW = DialogData[-6,number]
    local CalcP = round(Progress["step",number] * 100 / Progress["stepTotal",number])
    
    BoxW = ceil(BoxW/100*CalcP)
    if (BoxW<1) { BoxW = 1 }
    EGP:egpSize(BoxID,vec2(BoxW,20))
    EGP:egpSetText(PercID,CalcP+"%")
    
    local PercID = DialogData[-4,number]
    local BoxID = DialogData[-5,number]
    local BoxW = DialogData[-6,number]
    local CalcP = Progress["progressPercent",number]
    
    EGP:egpSetText(TW-1, "Current Task: "+Progress["progressTask",string])
    
    BoxW = ceil(BoxW/100*CalcP)
    if (BoxW<1) { BoxW = 1 }
    EGP:egpSize(BoxID,vec2(BoxW,20))
    EGP:egpSetText(PercID,Progress["progressCurrent",number] + " / " + Progress["progressMax",number] + " (" + CalcP+"%)")
    
    runCmd("alxos_png_progress",100)    
}

function alxos_png_ds(Name:string) {
    if (Name=="fs_readfile") {
        alxos_png_draw(dsGetString())
    } elseif (Name=="clear") {
        if (!checkPerf(7000)) {
            runCmd("alxos_png_clear",1)
            return
        }
        alxos_png_clear()
    } elseif (Name=="imageReaderError") {
        "alxos_explorer_error"("PNG Reader Chip not connected (crashed?)",420)     
    } elseif (Name=="initImage") {
        local Info = dsGetTable()        
        GpuData["PNGAcceleratorWirelink",wirelink] = Info[1,wirelink]
        GpuData["PNGInit",number] = 1
        runCmd("alxos_png_progress",100,1)
    }
}

function number alxos_png_key(Key:string,Pressed) {
    if (!Pressed) { return 0 }
    
    local Index = DialogData[-5,number]
    local Type = DialogData[-2,string]
    
    local Title = DialogData["title",string]   
    
    if (Key=="f8") {
        stoptimer("cmd")
        dsSendDirect("clear",GpuData[1,table]["hdd_io",wirelink]:entity(),entity())
        local IMGAccelerator = alxos_png_findAccelerator()
        dsSendDirect("SHUTDOWN",IMGAccelerator[3,entity],0)  
        alxos_png_close()
        return 1
    } elseif (Key=="enter" & Title!="Opening file" & Title!="PNG Settings") {
        alxos_png_key("f8",1)
        return 1
    }
    
    if (Title!="PNG Settings") { return 0 }
    
    if (Key=="left" | Key=="right" | Key=="tab") {
        local Max = 4
        local Ind = Index
        if (Key=="right" | Key=="tab") {
            Index = (Index==Max ? 1 : Index+1)
        } else {
            Index = (Index==1 ? Max : Index-1)
        }
        DialogData[-5,number] = Index
        DialogData[11,number] = Index
        
        local CurID = DialogData[-4,table][Index,number]     
        local PrevID = DialogData[-4,table][Ind,number]  
        
        if (Ind >= 3) {
            EGP:egpColor(PrevID, vec4(168,168,168,255))
        }
        
        if (Index >= 3) { EGP:egpColor(CurID, vec4(0,168,0,255)) }
        
        # "_"
        if (Index == 1 | Index == 2) {
            local TxtTbl = DialogData[-4,table][Index,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string]+"_",EGP:egpPos(TxtTbl[1,number]))
        }
        if (Ind == 1 | Ind == 2) {
            local TxtTbl = DialogData[-4,table][Ind,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string],EGP:egpPos(TxtTbl[1,number]))
        }
        return 1
        
    } elseif (Key=="up" | Key=="down") {
        local Ind = Index
        if (Key=="down") {
            if (Index < 4) { Index++ } 
            elseif (Index >= 4) { Index = 1 }
        } else { # Key == "up"
            if (Index <= 1) { Index = 4 } 
            elseif (Index > 1) { Index-- }
        }
        DialogData[-5,number] = Index
        DialogData[11,number] = Index
        
        local CurID = DialogData[-4,table][Index,number]     
        local PrevID = DialogData[-4,table][Ind,number]  
        
        if (Ind >= 3) {
            EGP:egpColor(PrevID, vec4(168,168,168,255))
        }
        
        if (Index >= 3) { EGP:egpColor(CurID, vec4(0,168,0,255)) }
        
        if (Index == 1 | Index == 2) {
            local TxtTbl = DialogData[-4,table][Index,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string]+"_",EGP:egpPos(TxtTbl[1,number]))
        }
        if (Ind == 1 | Ind == 2) {
            local TxtTbl = DialogData[-4,table][Ind,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string],EGP:egpPos(TxtTbl[1,number]))
        }
        return 1        
    } elseif (Key=="enter") {     
        
        if (Index == 4) {
            alxos_png_key("f8",1)
            return 1
        }
        
        if (Index == 3) {
            local InpMaxSize = DialogData[-4,table][1,table][2,string]:toNumber()
            local InpKeepAR = DialogData[-4,table][2,table][2,string]
            
            local MaxSize = gpuScreenSize(DS)
            
            if (InpMaxSize==0) { InpMaxSize = MaxSize }
            if (InpKeepAR=="") { InpKeepAR = "1" }
            
            GpuPersistData["pngMaxSize",number] = clamp(InpMaxSize,1,MaxSize)
            GpuPersistData["pngNoKeepAR",number] = (InpKeepAR=="1" ? 0 : 1)
            
            "alxos_explorer_closedialog"()
            alxos_png_postinit()
            return 1
        }
        
        return 0
        
    } else {
        if (Index == 1 | Index == 2) {
            local ITbl = DialogData[-4,table]
            local D_I = ITbl[Index,table][1,number]
            local D_V = ITbl[Index,table][2,string]
            
            if (Key=="backspace") {
                D_V = D_V:sub(0,D_V:length()-1)
                EGP:egpSetText(D_I,D_V+"_")
                ITbl[Index,table][2,string] = D_V
            } elseif (Key=="delete") {
                EGP:egpSetText(D_I,"_")
                ITbl[Index,table][2,string] = ""
            } elseif (Pressed==2) {
                local KMem = toByte(Key)
                local Char = toChar((KMem>=128 ? KMem-80 : KMem))
                local RE = ITbl[Index,table][3,string]
                local MC = ITbl[Index,table][4,number]
                
                if (Char:findRE(RE) & (KMem<=122 | KMem>=128 & KMem<=143)) {
                    if (D_V:length() < MC) {
                        D_V = (D_V+Char)
                        EGP:egpSetText(D_I,D_V+"_")
                        ITbl[Index,table][2,string] = D_V
                    }
                }
            }
        }
    }
    return 0
}
