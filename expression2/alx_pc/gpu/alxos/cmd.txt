# custom drew flag from image loaders
function alxos_console_ds_drew() {
    return DS[1048499]    
}

function alxos_toggle_cursor(Show:number) {
    if (CmdScreen==1) { return }
    
    CmdCursor = Show
    
    if (CmdScreen == 2) {
        if (alxos_console_ds_drew()) { return }
             
        local Path = (KeyHandler == "alxos_console_inputkey" ? GpuData["inputMsg",string] : GpuData[-3,string])
        local Cmd = GpuData[-4,string]
        local FullLine = Path + Cmd
        
        local CursorX = FullLine:length() * DsFontCharWidth
        local CursorY = (CmdLinesC-1) * DsFontLineHeight
        
        if (Show) {
            dsDrawChar(CursorX, CursorY, 95, 11053224) # 0xA8A8A8
        } else {
            dsDrawChar(CursorX, CursorY, 95, 0)
        }
    } else {
        local Cursor = GpuData[3,number]
        EGP:egpSetText(Cursor,(Show ? "_" : ""))
    }
}

function alxos_upd_cursor() {
    if (CmdScreen==1) {
        local XPos = (GpuData[-2, number] - BIOS_X) / 10
        CS[2045] = (CmdLinesC - 1) * 60 + XPos * 2
    } elseif (CmdScreen==2) {
        alxos_toggle_cursor(0)
    } else {
        local P = GpuData[3,number]
        local XPos = GpuData[-2,number]
        EGP:egpPos(P,vec2(XPos,LAST_Y-20))
    }
}

function alxos_screen_write(X:number, Y:number, Str:string) {
    csScreenWrite(X,Y-1,Str,666)
}

function alxos_console_move() {
    if (CmdLinesC>=CmdMaxLines) {
        local LinesToShift = CmdLinesC - CmdMaxLines
        
        if (CmdScreen==1) {
            csScreenMove(LinesToShift)
            CmdLinesC = CmdMaxLines
        } elseif (CmdScreen==2) {
            if (alxos_console_ds_drew()) { return }
            
            dsClear()
            dsStopQueues()
            
            local NLines = array()
            local NewIdx = 1
             
            for (I = (LinesToShift + 1), CmdLinesC) {
                local Str = CmdLines[I, string]
                NLines[NewIdx, string] = Str
                dsDrawTextStart(0, (NewIdx - 1) * DsFontLineHeight, Str, vec(168))
                NewIdx++
            }
            
            CmdLines = NLines    
            CmdLinesC = NewIdx-1   
        } else {        
            local RM = LinesToShift
            local NLines = array()
            for (I=1,CmdLinesC) {
                local ID = CmdLines[I,number]
                if (I<=RM) {
                    EGP:egpRemove(ID)   
                } else {
                    #CmdLines[I-1,number] = ID
                    NLines:pushNumber(ID)
                    EGP:egpPos(ID,EGP:egpPos(ID)-vec2(0,20*RM))    
                }
            }
            LAST_Y -= RM*20
            CmdLines = NLines
            CmdLinesC = CmdMaxLines
            if (LAST_I>100) { LAST_I = GpuData[3,number]+1 }
        }
    }
    
    if (GpuData[3,number]) { 
        alxos_toggle_cursor(0)
    }
}

function alxos_console_msg(Str:string) {
    #local Num = GpuData[2,number]
    local Arr = Str:explode(NL)
    
    if (CmdScreen==1) {
        for (I=1,Arr:count()) {
            alxos_screen_write(0, CmdLinesC, Arr[I, string])
            CmdLinesC++
            alxos_console_move()
        }
        
        alxos_upd_cursor()
    } elseif (CmdScreen==2) {
        for (I=1,Arr:count()) {
            CmdLines[CmdLinesC,string] = Arr[I, string]
            if (!alxos_console_ds_drew()) {
                dsDrawTextStart(0, (CmdLinesC-1)*DsFontLineHeight, Arr[I, string], vec(168))
            }
            CmdLinesC++
        }
        #alxos_console_move()
    } else {
        for (I=1,Arr:count()) {
            #GpuData[1,table][Num,array] = array(LAST_I,Arr[I,string])
            #GpuData[2,number] = Num+1
            CmdLines[CmdLinesC,number] = LAST_I
            EGP:egpText(LAST_I,Arr[I,string],vec2(BIOS_X,LAST_Y))
            EGP:egpFont(LAST_I,"Courier New")
            EGP:egpColor(LAST_I,168,168,168,255) LAST_I++ LAST_Y+=20
            CmdLinesC++
            #alxos_console_move()
        }
    }
}

function alxos_console_rawmsg(Str:string) {
    if (CmdScreen==1) {
        alxos_screen_write(0, CmdLinesC, Str)
        alxos_console_move()
    } elseif (CmdScreen==2) {
        CmdLines[CmdLinesC,string] = Str
        if (!alxos_console_ds_drew()) {
            dsDrawTextStart(0, (CmdLinesC-1)*DsFontLineHeight, Str, vec(168))
        }
        alxos_console_move()
    } else {
        CmdLines[CmdLinesC,number] = LAST_I

        EGP:egpFont(LAST_I,"Courier New")
        EGP:egpColor(LAST_I,168,168,168,255) LAST_I++ LAST_Y+=20
    }
    CmdLinesC++
}

function alxos_show_cursor() {
    if (CmdScreen==1) {
        CS[2046] = 1
    } else {
        alxos_toggle_cursor(!CmdCursor) 
        timer("cursor_draw",500)
    }
}

function alxos_hide_cursor() {
    if (CmdScreen==1) {
        CS[2046] = 0  
    } elseif (CmdScreen==2) {
        alxos_toggle_cursor(0)
    } else {
        CmdCursor = 0
        local Cursor = GpuData[3,number]
        EGP:egpSetText(Cursor,"")
    }
    stoptimer("cursor_draw")
}

function alxos_draw_cursor() {
    if (CmdScreen==1) {
        CS[2046] = 1 
    } elseif (CmdScreen==2) {
        registerTimers("cursor_draw","alxos_show_cursor")
        timer("cursor_draw",500)
    } else {
        GpuData[3,number] = LAST_I
        local PathX = GpuData[-2,number]
        EGP:egpText(LAST_I,"_",vec2(PathX,LAST_Y-20))
        EGP:egpFont(LAST_I,"Courier New")
        EGP:egpColor(LAST_I,168,168,168,255) LAST_I++
        
        registerTimers("cursor_draw","alxos_show_cursor")
        timer("cursor_draw",500)
    }
}

function alxos_console_command() {
    alxos_console_move()
    
    GpuData[-1,number] = LAST_I
    local Path = CmdPath:concat("/")+">"
    GpuData[-2,number] = BIOS_X + Path:length()*10
    GpuData[-3,string] = Path
    GpuData[-4,string] = ""
    
    if (CmdScreen==1) {
        alxos_screen_write(0, CmdLinesC, Path)
        alxos_upd_cursor()
        alxos_show_cursor()
    } elseif (CmdScreen==2) {
        CmdLines[CmdLinesC,string] = Path
        if (!alxos_console_ds_drew()) {
            dsDrawTextStart(0, (CmdLinesC-1)*DsFontLineHeight, Path, vec(168))
        }
        alxos_upd_cursor()
        alxos_show_cursor()
    } else {
        CmdLines[CmdLinesC,number] = LAST_I
        EGP:egpText(LAST_I,Path,vec2(BIOS_X,LAST_Y))
        EGP:egpFont(LAST_I,"Courier New")
        EGP:egpColor(LAST_I,168,168,168,255) LAST_I++
        LAST_Y += 20
        CmdLinesC++

        if (GpuData[3,number]!=0) {
            alxos_upd_cursor()
        }
    }
}

function alxos_console_input(Msg:string, Func:function, RE:string) {
    CmdInputFunc = Func
    CmdInputRE = RE
    
    alxos_console_msg(Msg)
    
    GpuData[-1,number] = LAST_I
    GpuData[-2,number] = BIOS_X + Msg:length()*10
    GpuData["inputLen",number] = Msg:length()
    GpuData["inputMsg",string] = Msg
    GpuData[-4,string] = ""
    
    alxos_console_move()
    
    if (!CmdScreen) {
        EGP:egpText(LAST_I,"",vec2(GpuData[-2,number]-BIOS_X,0))
        EGP:egpFont(LAST_I,"Courier New")
        EGP:egpParent(LAST_I,LAST_I-1)
        EGP:egpColor(LAST_I,168,168,168,255) LAST_I++
    } else {
        CmdLinesC--
    }

    KeyHandler = "alxos_console_inputkey"
    alxos_upd_cursor()
    alxos_show_cursor()
}
function alxos_console_input(Msg:string, Func:function) {
    alxos_console_input(Msg, Func, "^[A-Za-z0-9_ /.:-]$")
}

function alxos_console_err(Err:string) {
    alxos_console_handlers()
    alxos_console_msg(Err)
    alxos_console_command()
    KeyHandler = "alxos_console_key"
    alxos_show_cursor()
}

function alxos_console_cls_safe() {
    if (CmdScreen==1) {
        CS[2041] = 1
    } elseif (CmdScreen==2) {
        dsClear()
        dsStopQueues()
        dsInit(DS,256,256)
        dsCacheDelay(1.5)
    } else {
        LAST_I = 1
        LAST_Y = 0
        EGP:egpClear()
    }
    CmdLinesC = 1
    CmdLines = array()  
    CmdCursor = 0
}

function alxos_console_cls() {
    egpClear()
    alxos_console_cls_safe()
    alxos_console_command()
    alxos_draw_cursor()
}

function alxos_console_screen_vars(Screen:number) {
    CmdHeader = "ALX Wiremod E2 OS - Console v1.1\n(c) 2018-2026 by AlexALX"
    
    CSEvent = function() {}
    DSEvent = function() {}
    
    if (Screen==1) {
        CmdMaxLines = 17
        CS[2043] = 0.5
        CS[2044] = 0.2

        CmdHeader = "ALX Wiremod E2 OS Console v1.1\n(c) 2018-2026 by AlexALX"
        
        CSEvent = function() {
            if (EGP:entity():isValid()) {
                CmdScreen = 0
            } else {
                CmdScreen = 2
            }
            
            alxos_console_screen_vars(CmdScreen)
            alxos_console_cls_safe()
            alxos_console_msg("Console screen was disconnected!")
            alxos_console_command()
            alxos_draw_cursor()
            CSEvent = function() {}
        }        
    } elseif (Screen==2) {
        dsInit(DS, 256, 256)
        dsSetFont(dsFont5x7())
        dsCacheDelay(1.5)
        CmdMaxLines = 31
        
        DSEvent = function() {
            if (EGP:entity():isValid()) {
                CmdScreen = 0
            } else {
                CmdScreen = 1    
            }
            alxos_console_screen_vars(CmdScreen)
            alxos_console_cls_safe()
            alxos_console_msg("Digital screen was disconnected!")
            alxos_console_command()
            alxos_draw_cursor()
            DSEvent = function() {}
        }
    } else {
        CmdMaxLines = 25
    }
}

function alxos_console_screen(Screen:number) {
    if (Screen==1) {
        if (!CS:entity():isValid()) {
            alxos_console_msg("Console screen is not connected!")
            alxos_console_command()
            return
        }
    } elseif (Screen==2) {
        if (!DS:entity():isValid()) {
            alxos_console_msg("Digital screen is not connected!")
            alxos_console_command()
            return
        }
    } else {
        if (!EGP:entity():isValid()) {
            alxos_console_msg("EGP screen is not connected!")
            alxos_console_command()
            return
        }
    }
    
    CmdScreen = Screen
    
    alxos_console_screen_vars(Screen)
        
    egpClear()
    csClear()
    dsClear()
    dsStopQueues()

    alxos_console_cls_safe()
    alxos_console_msg(CmdHeader)
    alxos_console_command()
    alxos_draw_cursor()
}

function number alxos_console_listdir_draw(Info:table) {
    
    local CmdRun = GpuData[-5,number]
    
    if (CmdRun==0) {
    
        local InfoC = Info:count()
        if (InfoC<=2 & CmdDir[1,number]!=CmdFS["root_dir_cluster",number] | InfoC<=1) {
            local Err = Info[1,string]
            if (Err=="") {
                Err = "Directory is empty."   
            }
            alxos_console_err(Err)
            return 0
        }
        
        GpuData[4,table] = Info
        CmdDir[3,table] = Info
        
        GpuData[-5,number] = 1
        runCmd("alxos_console_listdir_draw",1,Info) 
        return 1
    
    } elseif (CmdRun==1) {
        local Max = Info:count()-1
        local I = GpuData[9,number]
        local TotalDirs = GpuData[3,number]
        while(perf()) {
            if (I>Max) { 
                break    
            }
            local Entry = Info[I,table]
            I++
            if (Entry["name",string]=="." | Entry["name",string]=="..") { continue }
            
            local Sz = alxos_nicesize(Entry["size",number])
            alxos_console_msg((alxos_wm1_isdir(Entry) ? "<DIR>      " : Sz+
            " ":repeat(11-Sz:length()))+Entry["name",string]+(Entry["ext",string]!="" ? "."+Entry["ext",string] : ""))

            #local Date = Entry["time_create",number]
            #local DateStr = alxos_wm1_date(Date)
            #local DateStr = alxos_wm1_time(Date)
            
            #TotalDirs++
        }
        #GpuData[3,number] = TotalDirs
        
        if (I<=Max) { 
            GpuData[9,number] = I
            runCmd("alxos_console_listdir_draw",1,Info) 
            return 1
        } else {
            GpuData[-5,number] = 0
            GpuData[10,number] = 0
            GpuData[9,number] = 0
        }
    
    }
    
    alxos_console_command()
    KeyHandler = "alxos_console_key"
    return 0
}

function table alxos_console_indir(Name:string,Type) {
    local C = CmdDir[3,table]:count()-1
    for(I=0,C) {
        local Entry = CmdDir[3,table][I,table]
        local IsDir = alxos_wm1_isdir(Entry)
        if (Entry["name",string]==Name & (IsDir & Type==0 | !IsDir & Type==1 )) {
            return Entry 
        }
    }
    return table()
}

function alxos_console_cdpost(Name:string) {
    CmdDir[3,table] = dsGetTable()
    local Dir = GpuData[-10,string]
    if (Dir=="..") {
        CmdPath:pop()
    } elseif (Dir:length()==2 && Dir[2]==":") {
        CmdPath = table(Dir)
    } else {
        CmdPath:pushString(Dir)
    }
    GpuData[-10,string] = ""
    if (GpuData[-12,number]==0) {
        alxos_console_command()
        CmdDsH = ""
        KeyHandler = "alxos_console_key"
    } else {
        runCmd("alxos_console_readpath",0,GpuData[-11,table])
    }
}

function alxos_console_cdpre(Path:string,PArr:array,Entry:table) {
    if (Path!=".") {
        CmdDsH = "alxos_console_cdpost"
        GpuData[-10,string] = Path
        if (Entry["fs_str_id",string]!="") {
            CmdDir[1,number] = Entry["root_dir_cluster",number]
            CmdDir[2,number] = 0
            CmdFS = Entry
            alxos_console_listdir(CmdFS,CmdDir[1,number],CmdDir[2,number])  
        } else {
            CmdDir[1,number] = Entry["cluster",number]
            CmdDir[2,number] = Entry["data_sector",number]
            alxos_console_listdir(CmdFS,Entry["cluster",number],Entry["data_sector",number])  
        }
        return
    }
    if (GpuData[-12,number]==0) {
        alxos_console_command()
        KeyHandler = "alxos_console_key"
    } else {
        runCmd("alxos_console_readpath",0,GpuData[-11,table])
    }
}

function number alxos_console_readpath_single(Path:string) {
    if (Path==".." & CmdDir[1,number] == CmdFS["root_dir_cluster",number] | Path=="") {
        return 0
    } elseif (Path:length()==2 & Path[2]==":") {
        if (CmdAllFs[Path[1],table]:count()) {
            GpuData[-11,table] = CmdAllFs[Path[1],table]
        } else {
            return -2
        }
        return 1
    }
    local Ent = alxos_console_indir(Path,0)
    if (!Ent:count()) {
        return -1
    }
    GpuData[-11,table] = Ent
    return 1
}

function number alxos_console_readpath(Args:table) {
    local Path = Args[1,string]
    local Callback = Args[2,string]
    local Step = Args[3,number]
    if (Step == 0) {
        KeyHandler = ""
        local Tmp = Path:explode("/")
        local CTmp = Tmp:count()
        if (CTmp==1) {
            local Ret = alxos_console_readpath_single(Path)
            if (Ret==-1) {
                alxos_console_err("Failed to change folder, invalid path.")
            } elseif (Ret==-2) {
                alxos_console_err("Failed to change folder, invalid volume.")
            } elseif (Ret==1) {
                local Ent = GpuData[-11,table]
                GpuData[-11,table] = table()
                Callback(Path,Tmp,Ent)
            } elseif (Ret==0) {
                alxos_console_command()
                KeyHandler = "alxos_console_key"
            }
        } else {
            Args[3,number] = 1
            Args[4,array] = Tmp
            Args[5,number] = CTmp
            Args[6,number] = 1
            Args[7,table] = CmdDir:clone()
            Args[8,table] = CmdPath:clone()
            Args[9,table] = CmdFS
            GpuData[-12,number] = 1
            runCmd("alxos_console_readpath",0,Args)    
        }
    } elseif (Step == 1) {
        local Tmp = Args[4,array]
        local CTmp = Args[5,number]
        local CurI = Args[6,number]

        while (perf(80)) {
            if (CurI>CTmp) {
                alxos_console_command()
                CmdDsH = ""
                KeyHandler = "alxos_console_key"
                GpuData[-11,table] = table()
                GpuData[-12,number] = 0
                return 0    
            }
            local Path = Tmp[CurI,string]
            local Ret = alxos_console_readpath_single(Path)
    
            if (Ret==-1) {
                CmdDir = Args[7,table]
                CmdPath = Args[8,table]
                CmdFS = Args[9,table]
                GpuData[-12,number] = 0
                CmdDsH = ""
                alxos_console_err("Failed to change folder, invalid path.")
                return 0
            } elseif (Ret==-2) {
                CmdDir = Args[7,table]
                CmdPath = Args[8,table]
                CmdFS = Args[9,table]
                GpuData[-12,number] = 0
                CmdDsH = ""
                alxos_console_err("Failed to change folder, invalid volume.")
                return 0
            } elseif (Ret==1) {
                local Ent = GpuData[-11,table]
                Args[6,number] = CurI+1
                GpuData[-11,table] = Args
                Callback(Path,Tmp,Ent)
                return 1
            } elseif (Ret==0) {
                #Args[6,number] = CurI+1
                #runCmd("alxos_console_readpath",0,Args) 
            }
            CurI++
        }

        Args[6,number] = CurI
        runCmd("alxos_console_readpath",1,Args)
        return 1
    }
    return 0
}

# redraw last msg
function alxos_console_redraw_last(Msg:string) {
    if (CmdScreen==1) {
        alxos_screen_write(0, CmdLinesC-1, Msg+" ":repeat(30-Msg:length()))
    } elseif (CmdScreen==2) {
        CmdLines[CmdLinesC-1, string] = Msg
        if (!alxos_console_ds_drew()) {
            dsDrawCachedText(0, (CmdLinesC-2)*DsFontLineHeight, Msg, vec(168))
        }
    } else {
        EGP:egpSetText(LAST_I-1,Msg)
    }
}

function alxos_console_redraw_last(Msg:string, Force:number) {
    if (CmdScreen==2) {
        dsDrawCachedForce()    
    }
    alxos_console_redraw_last(Msg)
}

# redraw msg by index (0 - last, 1 - 2 msgs before, and so)
function alxos_console_redraw_msg(Index, Msg:string) {
    if (CmdScreen==1) {
        alxos_screen_write(0, CmdLinesC-1-Index, Msg+" ":repeat(30-Msg:length()))
    } elseif (CmdScreen==2) {
        CmdLines[CmdLinesC-1-Index, string] = Msg
        if (!alxos_console_ds_drew()) {
            dsDrawCachedText(0, (CmdLinesC-2-Index)*DsFontLineHeight, Msg, vec(168))
        }
    } else {
        EGP:egpSetText(LAST_I-1-Index,Msg)
    }
}

function alxos_console_redraw_msg(Index, Msg:string, Force:number) {
    if (CmdScreen==2) {
        dsDrawCachedForce()    
    }
    alxos_console_redraw_msg(Index, Msg)
}

function number alxos_console_cmd() {
    local CurTask = CmdFS["hdd_io",wirelink]["CurTask",table]
    
    if (CurTask:count()) { runCmd("alxos_console_cmd",100) }
    else { return 0 }
    
    local Max = CurTask[1,number]
    local Cur = CurTask[2,number]
    if (Max==0) { return 0 }
    
    local CalcP = round(Cur/Max*100)
    alxos_console_redraw_last("Removing file... "+CalcP+"%")

    return 0
}

function alxos_explorer_fput_post(Name:string) {
    local Ret = dsGetNumber()
    if (Ret==-2) {
        alxos_console_err("Not enough space on disk.")
    } elseif (Ret!=1) {
        alxos_console_err("File system internal error.")
    } else {
        alxos_console_err("File succesfully written.")
    }
    CmdDsH = ""
}

function alxos_explorer_fget_post(Name:string) {
    local Data = dsGetString()
    alxos_console_err(Data) 
    CmdDsH = ""
}

function alxos_console_run(Command:string) {
    local Tmp = Command:find(" ")
    local Cmd = Command:sub(0,Tmp-1)
    local Arg = Command:sub(Tmp+1,Command:length())
    if (Tmp==0) {
        Cmd = Command
        Arg = ""   
    }
    
    if (Command!="") {
        CmdHistory:pushString(Command)
        if (CmdHistory:count()>15) {
            CmdHistory:remove(1)    
        }
    }
    GpuData[100,number] = 0

    switch (Cmd:lower()) {
        case "help",
            if (Arg!="") {
                alxos_console_err("Invalid command arguments.")
                return   
            }
            alxos_console_msg("Available commands:")
            if (CmdScreen!=1) {
                alxos_console_msg("help - display help information")
            }
            alxos_console_msg("cls - clear console")
            alxos_console_msg("cs - draw on console screen")
            alxos_console_msg("ds - draw on digital screen")
            alxos_console_msg("egp - draw on egp screen")
            alxos_console_msg("reboot - reboot PC")
            alxos_console_msg("shutdown - turn off PC")
            #alxos_console_msg("info - display information about PC")
            alxos_console_msg("volumes - show list of all available volumes")
            alxos_console_msg("ls - show list of files and folders")
            alxos_console_msg("cd path - move to this path")
            alxos_console_msg("mkdir DirName - create folder with given name")
            alxos_console_msg("mkfile FileName.ext - create empty file")
            alxos_console_msg("rm FileName - remove file or folder")
            alxos_console_msg("fput FileName content - write to file with content")
            alxos_console_msg("fget FileName - read first 48 bytes from file")
            alxos_console_msg("open FileName.e2e - run E2 executable")
            if (CmdScreen!=1) {
                alxos_console_msg(" e2r, e2p, e2z also supported by this command")
                alxos_console_msg("open FileName.jpg - load image on Digital Screen")
                alxos_console_msg(" Supported images: jpg, gif, bmp, png")
                alxos_console_msg("open FileName.zip - open zip")
            }
            alxos_console_msg(" Open command can open any supported file in OS")
            alxos_console_command()
        break
        case "volumes",
        case "vls",
        case "vol",
            if (Arg!="") {
                alxos_console_err("Invalid command arguments.")
                return   
            }
            local MenuTree = table()
            foreach(K,V:table=CmdAllFs) {
                if (K=="C") {
                    MenuTree:unshiftString(K+": ["+V["volume_name",string]+"]")
                } else {
                    MenuTree:pushString(K+": ["+V["volume_name",string]+"]")
                }
            }
            
            alxos_console_msg("Available volumes:")            
            for(I=1,MenuTree:count()) {
                alxos_console_msg(MenuTree[I,string])
            }
            
            alxos_console_command()
        break
        case "cls",
        case "clear",
            alxos_console_cls()
        break
        case "echo",
        case "print",
            alxos_console_err(Arg)
        break
        case "ls",
        case "dir",
            if (Arg!="") {
                alxos_console_err("Invalid command arguments.")
                return   
            }
            alxos_console_msg("Directory content:")
            KeyHandler = ""
            #if (Arg=="") {
                alxos_console_listcurdir()
            #[} else {
                alxos_console_readpath(Arg,"alxos_console_listdir_path")
            }]#
        break
        case "cd",
            alxos_console_readpath(table(Arg,"alxos_console_cdpre"))
        break
        case "reboot",
            if (Arg!="") {
                alxos_console_err("Invalid command arguments.")
                return   
            }
            reboot()
        break
        case "exit",
        case "quit",
            if (DEBUG) { DEBUG = 1 }
            stoptimer("cursor_draw")
            csClear()
            "alxos_execute"("alxos_explorer")
        break
        case "shutdown",
            if (Arg!="") {
                alxos_console_err("Invalid command arguments.")
                return   
            }
            dsSendDirect("SHUTDOWN",BIOS,0)  
        break
        case "mkdir",
            if (Arg=="") {
                alxos_console_err("Please enter folder name.")
                return   
            }
            
            if (CmdFS["read_only",number]) {
                alxos_console_ro()
                return
            }
            
            for(I=0,CmdDir[3,table]:count()) {
                local Entry = CmdDir[3,table][I,table]
                local FullName = Entry["name",string]+(Entry["ext",string]!="" ? "."+Entry["ext",string] : "")
                if (FullName==Arg) {
                    alxos_console_err("Folder or file with this name already exists.")
                    return
                }
            }
            
            KeyHandler = ""            
            CmdDsH = "alxos_console_mkdirpost"
            dsSendDirect("fs_mkfile",CmdFS["hdd_io",wirelink]:entity(),table(entity(),CmdFS["fs_str_id",string],table(CmdFS,Arg,"",1,CmdDir[1,number])))   
        break
        case "mkfile",
            if (Arg=="") {
                alxos_console_err("Please enter file name.")
                return   
            }
            
            if (CmdFS["read_only",number]) {
                alxos_console_ro()
                return
            }
            
            local Tmp = Arg:explode(".")
            if (Tmp:count()>2) {
                alxos_console_err("Invalid file name, too many dots.")
                return 
            }
            
            if (Arg:findRE("[^A-Za-z0-9_!@#$^ .-]")) {
                alxos_console_err("Invalid file name, forbidden chars.")
                return
            }
            
            if (Tmp[1,string]:length()>10) {
                alxos_console_err("Invalid file name, max 10 chars.")
                return 
            }
            
            if (Tmp[2,string]:length()>3) {
                alxos_console_err("Invalid file extension, max 3 chars.")
                return 
            }
            
            for(I=0,CmdDir[3,table]:count()) {
                local Entry = CmdDir[3,table][I,table]
                local FullName = Entry["name",string]+(Entry["ext",string]!="" ? "."+Entry["ext",string] : "")
                if (FullName==Arg) {
                    alxos_console_err("Folder or file with this name already exists.")
                    return
                }
            }
            
            KeyHandler = ""
            CmdDsH = "alxos_console_mkfilepost"
            dsSendDirect("fs_mkfile",CmdFS["hdd_io",wirelink]:entity(),table(entity(),CmdFS["fs_str_id",string],table(CmdFS,Tmp[1,string],Tmp[2,string],0,CmdDir[1,number]))) 
        break
        case "rm",
        case "remove",
            if (Arg=="") {
                alxos_console_err("Please enter file name.")
                return   
            }
            
            if (CmdFS["read_only",number]) {
                alxos_console_ro()
                return
            }
        
            local File = table()
            local Num = 0
        
            for(I=0,CmdDir[3,table]:count()) {
                local Entry = CmdDir[3,table][I,table]
                local FullName = Entry["name",string]+(Entry["ext",string]!="" ? "."+Entry["ext",string] : "")
                if (FullName==Arg) {
                    File = Entry
                    Num = I
                    break
                }
            }
            
            if (!File:count()) {
                alxos_console_err("Folder or file with this name not exists.")
                return
            }
            
            GpuData[-11,number] = Num
            
            KeyHandler = ""
            
            if (alxos_wm1_isdir(File)) {
                GpuData[-10,table] = File
                CmdDsH = "alxos_console_delpre"
                dsSendDirect("fs_listdir",CmdFS["hdd_io",wirelink]:entity(),table(entity(),CmdFS["fs_str_id",string],table(CmdFS,File["cluster",number],File["data_sector",number])))    
            } else {
                alxos_console_msg("Removing file... 0%")
                
                runCmd("alxos_console_cmd",100)
            
                CmdDsH = "alxos_console_delpost"
                dsSendDirect("fs_removefile",CmdFS["hdd_io",wirelink]:entity(),table(entity(),CmdFS["fs_str_id",string],table(CmdFS,File["cluster",number],File["data_sector",number])))
            }       
        break
        case "version",
        case "ver",
            alxos_console_msg(CmdHeader)
            alxos_console_command()
        break
        case "fput",
        case "fwrite",
            local Tmp2 = Arg:find(" ")
            local Arg1 = Arg:sub(0,Tmp2-1)
            local Arg2 = Arg:sub(Tmp2+1,Arg:length())
            if (Tmp2==0) {
                Arg1 = Arg
                Arg2 = ""   
            }
        
            if (Arg1=="") {
                alxos_console_err("Please enter file name.")
                return   
            }
            
            if (Arg2=="") {
                alxos_console_err("Please enter file content.")
                return   
            }
            
            local File = table()
        
            for(I=0,CmdDir[3,table]:count()) {
                local Entry = CmdDir[3,table][I,table]
                local FullName = Entry["name",string]+(Entry["ext",string]!="" ? "."+Entry["ext",string] : "")
                if (FullName==Arg1) {
                    File = Entry
                    break
                }
            }
            
            if (!File:count()) {
                alxos_console_err("File with this name not exists.")
                return
            }
            
            if (alxos_wm1_isdir(File)) {
                alxos_console_err("This is folder!")
                return
            }
            
            if (alxos_wm1_isdir(File)) {
                alxos_console_err("This is folder!")
                return
            }
            
            KeyHandler = ""        

            local C = File["cluster",number]
            local Sd = File["data_sector",number]
            CmdDsH = "alxos_explorer_fput_post"
            
            alxos_console_msg("Writing to file...")
            
            #runCmd("alxos_explorer_mkfile_progress",100)
            
            dsSendDirect("fs_writefile",CmdFS["hdd_io",wirelink]:entity(),table(entity(),CmdFS["fs_str_id",string],table(CmdFS,C,Sd,Arg2)))    
        
        break
        case "cat",
        case "fread",
        case "fget",
            if (Arg=="") {
                alxos_console_err("Please enter file name.")
                return   
            }
        
            local File = table()
        
            for(I=0,CmdDir[3,table]:count()) {
                local Entry = CmdDir[3,table][I,table]
                local FullName = Entry["name",string]+(Entry["ext",string]!="" ? "."+Entry["ext",string] : "")
                if (FullName==Arg) {
                    File = Entry
                    break
                }
            }
            
            if (!File:count()) {
                alxos_console_err("File with this name not exists.")
                return
            }
            
            if (alxos_wm1_isdir(File)) {
                alxos_console_err("This is folder!")
                return
            }
            
            KeyHandler = ""
            
            alxos_console_msg("File first 48 bytes:")   
            CmdDsH = "alxos_explorer_fget_post"         
            dsSendDirect("fs_readfile",CmdFS["hdd_io",wirelink]:entity(),table(entity(),CmdFS["fs_str_id",string],table(CmdFS,File,0,48)))  
        break
        case "open",
        case "openfile",
        case "start",
        case "run",
            if (Arg=="") {
                alxos_console_err("Please enter file name.")
                return   
            }
        
            local File = table()
        
            for(I=0,CmdDir[3,table]:count()) {
                local Entry = CmdDir[3,table][I,table]
                local FullName = Entry["name",string]+(Entry["ext",string]!="" ? "."+Entry["ext",string] : "")
                if (FullName==Arg) {
                    File = Entry
                    break
                }
            }
            
            if (!File:count()) {
                alxos_console_err("File with this name not exists.")
                return
            }
            
            local Ext = File["ext",string]:lower()
            
            if (alxos_wm1_isdir(File)) {
                alxos_console_err("This is directory.")
                return
            }

            KeyHandler = ""
            CurrentProg = "alxos_console" # force update
            if (Ext=="e2e") {
                dsSendDirect("alxos_execute",E2_CPU,table(entity(),CmdFS,File))
            } else {
                dsSendDirect("alxos_openfile",E2_CPU,table(entity(),File))    
            }
        break
        case "consolescreen",
        case "cs",
            alxos_console_screen(1)
        break
        case "digitalscreen",
        case "ds",
            alxos_console_screen(2)
        break
        case "egpscreen",
        case "egp",
            alxos_console_screen(0)
        break
        default,
            if (Command!="") {
                alxos_console_msg("Unknown command.")
            }
            alxos_console_command()
        break
    }
}

function alxos_console_noinput() {
    CmdInputFunc = function(Input:string) { 
        return 1 
    }
    alxos_hide_cursor()
}

function alxos_console_inputkey(Key:string,Pressed) {
    if (!Pressed) { return } 
    
    local MsgLen = GpuData["inputLen",number]
    
    if (Key=="enter") {
        if (CmdScreen) {
            if (CmdScreen == 2) {
                alxos_toggle_cursor(0)
                CmdLines[CmdLinesC, string] = GpuData["inputMsg",string]+GpuData[-4,string]
            }
            
            CmdLinesC++
        }
        
        local Result = CmdInputFunc(GpuData[-4,string])[number]
        #[
        if (Result) {
            alxos_console_noinput()
        } ]#
    } elseif (Key=="f8") {
        if (CmdScreen) {
            CmdLinesC++
        }
        
        CmdInputFunc("_EXIT_")[number]
        alxos_console_handlers()
        alxos_console_command()
    } elseif (Key=="delete") { 
        local I = GpuData[-1,number]
        local Cmd = GpuData[-4,string]
    
        if (CmdScreen==1) {
            alxos_screen_write(MsgLen, CmdLinesC, " ":repeat(Cmd:length())) 
        } elseif (CmdScreen==2) {
            local FullCmd = GpuData["inputMsg",string]
            dsTextClearLine((CmdLinesC-1)*DsFontLineHeight)
            dsDrawTextStart(0, (CmdLinesC-1)*DsFontLineHeight, FullCmd, vec(168))
        } else {
            EGP:egpSetText(I,"")
        }
        GpuData[-4,string] = ""
        GpuData[-2,number] = GpuData[-2,number] - Cmd:length()*10
        alxos_upd_cursor()
    } elseif (Key=="backspace") { 
        local I = GpuData[-1,number]
        local Cmd = GpuData[-4,string]
        if (Cmd=="") { return }        
        Cmd = Cmd:sub(0,Cmd:length()-1)
        if (CmdScreen==1) {
            alxos_screen_write(MsgLen, CmdLinesC, Cmd + " ") 
        } elseif (CmdScreen==2) {
            local FullCmd = GpuData["inputMsg",string]
            dsTextClearLine((CmdLinesC-1)*DsFontLineHeight)
            dsDrawTextStart(0, (CmdLinesC-1)*DsFontLineHeight, FullCmd + Cmd, vec(168))
        } else {
            EGP:egpSetText(I,Cmd)
        }
        GpuData[-4,string] = Cmd
        GpuData[-2,number] = GpuData[-2,number] - 10
        alxos_upd_cursor()        
    } elseif (Pressed==2) {
        local KMem = toByte(Key) # handle capital/small without writting own code
        local Char = toChar((KMem>=128 ? KMem-80 : KMem))
        
        if (Char:findRE(CmdInputRE) & (KMem<=122 | KMem>=128 & KMem<=143)) {
            local I = GpuData[-1,number]
            local Cmd = GpuData[-4,string]
            Cmd += Char
            if (CmdScreen==1) {
                alxos_screen_write(MsgLen, CmdLinesC, Cmd)
            } elseif (CmdScreen==2) {      
                alxos_toggle_cursor(0)        
                dsDrawTextStart(MsgLen*DsFontCharWidth, (CmdLinesC-1)*DsFontLineHeight, Cmd, vec(168))
            } else {
                EGP:egpSetText(I,Cmd)
            }
            GpuData[-4,string] = Cmd
            GpuData[-2,number] = GpuData[-2,number] + 10
            alxos_upd_cursor()
        }
    }    
}

function alxos_console_key(Key:string,Pressed) {
    if (!Pressed) { return } 
    
    if (alxos_console_ds_drew()) {
        if (Key=="enter" | Key=="f8") {
            alxos_console_cls()
        }
        return
    }
    
    if (Key=="enter") {
        if (CmdScreen) {
            if (CmdScreen == 2) {                
                alxos_toggle_cursor(0)
                CmdLines[CmdLinesC, string] = GpuData[-3,string] + GpuData[-4,string]
            }
            
            CmdLinesC++
        }
        alxos_console_run(GpuData[-4,string])
    } elseif (Key=="up" | Key=="down") {
    
        if (CmdHistory:count()) {
            
            local Ind = GpuData[100,number]
            if (Key=="up") {
                Ind--    
            } else {
                Ind++
            }
            
            local Max = CmdHistory:count()
            
            if (Ind<1) {
                Ind = Max  
            } elseif (Ind>Max) { 
                Ind = 1
            }
            
            GpuData[100,number] = Ind
    
            local I = GpuData[-1,number]
            local Path = GpuData[-3,string]
            local Cmd = CmdHistory[Ind,string]
        
            if (CmdScreen==1) {
                local Msg = Path + Cmd
                alxos_screen_write(0, CmdLinesC, Msg + " ":repeat(30-Msg:length()))
            } elseif (CmdScreen==2) {
                dsTextClearLine((CmdLinesC-1)*DsFontLineHeight)
                dsDrawTextStart(0, (CmdLinesC-1)*DsFontLineHeight, Path + Cmd, vec(168))
            } else {
                EGP:egpSetText(I,Path+Cmd)    
            }
            GpuData[-4,string] = Cmd
            GpuData[-2,number] = (Path+Cmd):length()*10 + 10
            alxos_upd_cursor()
        }
    } elseif (Key=="delete") { 
        local I = GpuData[-1,number]
        local Path = GpuData[-3,string]
        local Cmd = GpuData[-4,string]
        
        if (CmdScreen==1) { 
            alxos_screen_write(0, CmdLinesC, Path + " ":repeat(Cmd:length())) 
        } elseif (CmdScreen==2) {
            dsTextClearLine((CmdLinesC-1)*DsFontLineHeight)
            dsDrawTextStart(0, (CmdLinesC-1)*DsFontLineHeight, Path + Cmd, vec(168))
        } else {
            EGP:egpSetText(I,Path)
        }
        GpuData[-4,string] = ""
        GpuData[-2,number] = (Path):length()*10 + 10
        alxos_upd_cursor()
    } elseif (Key=="backspace") { 
        local I = GpuData[-1,number]
        local Path = GpuData[-3,string]
        local Cmd = GpuData[-4,string]
        if (Cmd=="") { return }
        Cmd = Cmd:sub(0,Cmd:length()-1)
        if (CmdScreen==1) {
            alxos_screen_write(0, CmdLinesC, Path + Cmd + " ")
        } elseif (CmdScreen==2) {
            dsTextClearLine((CmdLinesC-1)*DsFontLineHeight)
            dsDrawTextStart(0, (CmdLinesC-1)*DsFontLineHeight, Path + Cmd, vec(168))
        } else {
            EGP:egpSetText(I,Path+Cmd)
        }
        GpuData[-4,string] = Cmd
        GpuData[-2,number] = GpuData[-2,number] - 10
        alxos_upd_cursor()
    } elseif (Pressed==2) {
        local KMem = toByte(Key) # handle capital/small without writting own code
        local Char = toChar((KMem>=128 ? KMem-80 : KMem))
        local RE = "^[A-Za-z0-9_ /.:-]$"
        
        if (Char:findRE(RE) & (KMem<=122 | KMem>=128 & KMem<=143)) {
            local I = GpuData[-1,number]
            local Path = GpuData[-3,string]
            local Cmd = GpuData[-4,string]
            Cmd += Char
            if (CmdScreen==1) {
                alxos_screen_write(0, CmdLinesC, Path + Cmd)
            } elseif (CmdScreen==2) {
                alxos_toggle_cursor(0)
                dsDrawTextStart(0, (CmdLinesC-1)*DsFontLineHeight, Path + Cmd, vec(168))
            } else {
                EGP:egpSetText(I,Path+Cmd)
            }
            GpuData[-4,string] = Cmd
            GpuData[-2,number] = GpuData[-2,number] + 10
            alxos_upd_cursor()
        }
    }
}

function alxos_console_ds(Name:string) {        
    if (CmdDsH!="") {
        CmdDsH(Name)
    } elseif (Name=="fs_listdir") {
        alxos_console_listdir_draw(dsGetTable())   
    } elseif (Name=="alxos_runprog") {
        stoptimer("cursor_draw")
        local Info = dsGetTable()
        
        local TInfo = table(Info[2,table],CmdFS,CmdAllFs,CmdPath,CmdDir)
        "alxos_execute"(Info[1,string],TInfo)
    } elseif (Name=="alxos_fail") {
        alxos_console_err(dsGetString())
    } elseif (Name=="alxos_openfile") {
        local Info = dsGetTable()
        if (Info[1,number]==-1) {
            alxos_console_err("Failed to open file, unsupported format")     
        }    
    }
}

function alxos_console_postinit(Name:string) {
    CmdDir[3,table] = dsGetTable()
    
    alxos_console_command()
    
    CmdCursor = 0
    alxos_draw_cursor()    
    
    KeyHandler = "alxos_console_key"  

    CmdDsH = ""
}

    return CmdPath
}

function table alxos_console_getdir() {
    return CmdDir
}

function alxos_console_init(Args:table) {
    egpClear()
    csClear()
    CmdFS = Args[2,table]
    CmdAllFs = Args[3,table]
    CmdPath = Args[4,table]
    CmdDir = Args[5,table]
    
    if (!CmdDir:count()) {
        CmdDir[1,number] = CmdFS["root_dir_cluster",number]
        CmdDir[2,number] = 0
    }
    
    if (
        (CmdScreen==0 & !EGP:entity():isValid()) |
        (CmdScreen==1 & !CS:entity():isValid()) |
        (CmdScreen==2 & !DS:entity():isValid())
    ) {
        if (EGP:entity():isValid()) {
            CmdScreen = 0
        } elseif (CS:entity():isValid()) {
            CmdScreen = 1
        } elseif (DS:entity():isValid()) {
            CmdScreen = 2
        }
    }
    
    alxos_console_screen_vars(CmdScreen)
    
    CmdCopy = table()
    CmdDsH = ""
    CmdKeyH = ""
    KeyHandler = ""
    DsHandler = "alxos_console_ds"
    #CmdPath = table("C:")
    CmdLinesC = 1
    CmdLines = array()
    if (!CmdHistory:count()) { CmdHistory = array("") }
    PlugHandler = "alxos_console_onplug"
    
    GpuData[1,table] = table()
    GpuData[2,number] = 0
    
    alxos_console_msg(CmdHeader)
    if (!EGP:entity():isValid()) {
        alxos_console_msg("Warning: EGP is not connected.")
    }
    alxos_console_command()  
    
    CmdDsH = "alxos_console_postinit"
    alxos_console_listdir(CmdFS,CmdDir[1,number],CmdDir[2,number])       
}

function alxos_console_onplug(Name:string) {
    if (Name=="sata_onplug" | Name=="hddio_unplug" | Name=="cd_newdisc") {
        if (!CmdFS["hddinfo",table]:count() | CmdFS["start_sector",number]==-1) {
            CmdFS = CmdAllFs["C",table]    
            CmdPath = table("C:")
            CmdDir = table(CmdFS["root_dir_cluster",number],0)
            alxos_console_msg("Disk was unplugged.")
            alxos_console_command()
            KeyHandler = "alxos_console_key" 
            CmdDsH = ""
            stopCmd()
        }
    }
}
