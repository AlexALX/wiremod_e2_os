#include "alx_pc/gpu/alxos/_main_"

function alxos_execute(Name:string) {
    if (ALXOS_PROGS:exists(Name)) {
        Name(table())
        CurrentProg = Name   
    } else {
        "alxos_explorer_error"("Function "+Name+" is missing.")
    }
}

function alxos_execute(Name:string,Args:table) {
    if (ALXOS_PROGS:exists(Name)) {
        Name(Args)
        CurrentProg = Name
    } else {
        "alxos_explorer_error"("Function "+Name+" is missing.")
    }
}
