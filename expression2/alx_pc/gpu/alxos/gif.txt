function alxos_gif_init(Args:table) {
    if (!alxos_img_init(Args, "gif")) {
        return    
    }
    
    if (CurrentProg=="alxos_console") {
        DialogData = table()
        "alxos_img_postinit"()
        return
    }

    local Height = 290

    DialogData[-10,table]:pushNumber(LAST_I)
    
    local LAST_Y = 18
    
    Txt = "Please choose GIF Animation Mode:\n\n"+
    "Normal - no animation, just first frame\n"+
    "Manual - play animation manually\n"+
    "Animate - replay animation, but it's slow,\n"+
    "work properly only on very small images"

    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    LAST_Y += 122
    
    local Entries = table()
    local EntID = DialogData[-10,table]
    
    local GifMode = GpuPersistData["gifMode",number]
    
    # 1. Normal Mode Button
    Entries[1,number] = LAST_I
    EntID:pushNumber(LAST_I)   
    local Sz = 65
    local Txt = "Normal"
    local CenterX = (Width / 3) * 0.8
    local Col = vec4(168,168,168,255)
    if (GifMode==0) { Col = vec4(0,168,168,255) }
    EGP:egpBox(LAST_I,vec2(CenterX - Sz/2,LAST_Y),vec2(Sz,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,Col) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    EntID:pushNumber(LAST_I)        
    EGP:egpText(LAST_I,Txt,EGP:egpPos(LAST_I-1)+vec2(5,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    # 2. Manual Mode Button
    Entries[2,number] = LAST_I
    EntID:pushNumber(LAST_I)  
    local Sz = 65
    local Txt = "Manual"
    local CenterX = (Width / 3) * 1.5
    local Col = vec4(168,168,168,255)
    if (GifMode==1) { Col = vec4(0,168,168,255) }
    EGP:egpBox(LAST_I,vec2(CenterX - Sz/2,LAST_Y),vec2(Sz,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,Col) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    EntID:pushNumber(LAST_I)        
    EGP:egpText(LAST_I,Txt,EGP:egpPos(LAST_I-1)+vec2(5,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    # 3. Animate Mode Button
    Entries[3,number] = LAST_I
    EntID:pushNumber(LAST_I)  
    local Sz = 70
    local Txt = "Animate"
    local CenterX = (Width / 3) * 2.3
    local Col = vec4(168,168,168,255)
    if (GifMode==2) { Col = vec4(0,168,168,255) }
    EGP:egpBox(LAST_I,vec2(CenterX - Sz/2,LAST_Y),vec2(Sz,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpAlign(LAST_I,1)
    EGP:egpColor(LAST_I,Col) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    EntID:pushNumber(LAST_I)        
    EGP:egpText(LAST_I,Txt,EGP:egpPos(LAST_I-1)+vec2(5,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++

    LAST_Y += 35

    # 4. Max Size Input
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"Size Limit (px):",vec2(22,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"512 max DS, 1024 Quad DS",vec2(220,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++

    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpBox(LAST_I,vec2(220,LAST_Y),vec2(40,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,168,168,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    local DefaultNoKeepAR = GpuPersistData["gifNoKeepAR",number] == 1 ? "0" : "1"
    Entries[5,table] = table(LAST_I,DefaultNoKeepAR,"^[0-1]$",1)
    EGP:egpText(LAST_I,DefaultNoKeepAR,vec2(222,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    LAST_Y += 35

    # 5. Min Delay
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"Minumum Frame Delay:",vec2(22,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpText(LAST_I,"can be float (eg 0.5)",vec2(270,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)
    EGP:egpBox(LAST_I,vec2(205,LAST_Y),vec2(55,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,168,168,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    DialogData[-10,table]:pushNumber(LAST_I)

    Entries[6,table] = table(LAST_I,GifMinDelay:toString(),"^[0-9.]$",4)
    EGP:egpText(LAST_I,GifMinDelay:toString(),vec2(207,LAST_Y))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255) 
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++

    LAST_Y = Height - 25

    # 6. OK Button (In focus by default)
    local SzOK = 30
    local AddOK = -100
    
    DialogData[-10,table]:pushNumber(LAST_I)
    Entries[7,number] = LAST_I
    EGP:egpBox(LAST_I,vec2((SzOK/2)+AddOK+Width/2,Height-22),vec2(SzOK,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,168,0,255)
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    EGP:egpText(LAST_I,"OK",EGP:egpPos(LAST_I-1)+vec2(5,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255)
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++

    # 7. Cancel Button
    local SzCan = 60
    local AddCan = 20
    
    DialogData[-10,table]:pushNumber(LAST_I)
    Entries[8,number] = LAST_I
    EGP:egpBox(LAST_I,vec2((SzCan/2)+AddCan+Width/2,Height-22),vec2(SzCan,20))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,168,168,168,255)
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++
    
    EGP:egpText(LAST_I,"Cancel",EGP:egpPos(LAST_I-1)+vec2(5,0))
    EGP:egpParent(LAST_I,Parent)
    EGP:egpColor(LAST_I,0,0,0,255)
    DialogData[-10,table]:pushNumber(LAST_I) LAST_I++

    DialogData[-5,number] = 7
    DialogData[-4,table] = Entries
    DialogData[10,table] = Entries
    DialogData[11,number] = 7
    DialogData[12,number] = 8
    DialogData[-2,string] = "text"
}

function number alxos_gif_cmd() {
    return 0
}

function alxos_gif_draw(Data:string, IMGAccelerator:entity) {
    local GifMode = GpuPersistData["gifMode",number]
    
    dsSendDirect("drawImage",IMGAccelerator,table(
        DS, entity(), TickQuota, "gif", table(
            Data, 
            GifMode, 
            GpuPersistData["gifMaxSize",number],
            GpuPersistData["gifNoKeepAR",number],
            GpuPersistData["gifMinDelay",number]
        )
    ))
}

function alxos_gif_proccessMode() {
    local GifMode = GpuPersistData["gifMode",number]
    
    if (GifMode==1) {
        local IMGAccelerator = GpuData["IMGAcceleratorWirelink",wirelink]
        dsSendDirect("nextFrame",IMGAccelerator:entity(),1)
    } elseif (GifMode==2) {
        alxos_img_stop()
        alxos_img_finish()
    }
}

function number alxos_gif_key(Key:string,Pressed) {
    if (!Pressed) { return 0 }
    
    local Index = DialogData[-5,number]
    local Type = DialogData[-2,string]
    local GifMode = GpuPersistData["gifMode",number]
    
    local Title = DialogData["title",string]
    
    if ((GpuData["lastProg",string]=="alxos_console" | Title=="Reading GIF") & Key=="enter") {        
        alxos_gif_proccessMode()
        return 0
    }
    
    if (Key=="f8") {
        stoptimer("cmd")
        dsSendDirect("clear",GpuData[1,table]["hdd_io",wirelink]:entity(),1)
        local IMGAccelerator = alxos_img_findAccelerator()
        dsSendDirect("SHUTDOWN",IMGAccelerator[3,entity],0)  
        alxos_img_close()
        return 1
    } elseif (Key=="enter" & Title!="Opening file" & Title!="GIF Settings") {
        alxos_gif_key("f8",1)
        return 1
    }
    
    if (Title!="GIF Settings") { return 0 }
    
    if (Key=="left" | Key=="right" | Key=="tab") {
        local Max = 8
        local Ind = Index
        if (Key=="right" | Key=="tab") {
            Index = (Index==Max ? 1 : Index+1)
        } else {
            Index = (Index==1 ? Max : Index-1)
        }
        DialogData[-5,number] = Index
        DialogData[11,number] = Index
        
        local CurID = DialogData[-4,table][Index,number]     
        local PrevID = DialogData[-4,table][Ind,number]  
        
        if (Ind <= 3) {
            local OldCol = (GifMode == (Ind-1) ? vec4(0,168,168,255) : vec4(168,168,168,255))
            EGP:egpColor(PrevID, OldCol)
        } elseif (Ind >= 7) {
            EGP:egpColor(PrevID, vec4(168,168,168,255))
        }
        
        if (Index <= 3 | Index >= 7) { EGP:egpColor(CurID, vec4(0,168,0,255)) }
        
        # "_"
        if (Index >= 4 & Index <= 6) {
            local TxtTbl = DialogData[-4,table][Index,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string]+"_",EGP:egpPos(TxtTbl[1,number]))
        }
        if (Ind >= 4 & Ind <= 6) {
            local TxtTbl = DialogData[-4,table][Ind,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string],EGP:egpPos(TxtTbl[1,number]))
        }
        return 1
        
    } elseif (Key=="up" | Key=="down") {
        local Ind = Index
        if (Key=="down") {
            if (Index >= 1 & Index <= 3) { Index = 4 }      #      Max Size
            elseif (Index == 4) { Index = 5 }               #  Max Size   Keep AR
            elseif (Index == 5) { Index = 6 }               #  Keep AR    Min Delay
            elseif (Index == 6) { Index = 7 }               #  Min Delay    OK
            elseif (Index >= 7 & Index <= 8) { Index = 1 }  #      Normal
        } else { # Key == "up"
            if (Index >= 1 & Index <= 3) { Index = 7 }      #     OK
            elseif (Index == 4) { Index = 1 }               #  Max Size   Normal
            elseif (Index == 5) { Index = 4 }               #  Keep AR   Max Size
            elseif (Index == 6) { Index = 5 }               #  Keep AR   Min Delay
            elseif (Index >= 7 & Index <= 8) { Index = 6 }  #      Min Delay
        }
        DialogData[-5,number] = Index
        DialogData[11,number] = Index
        
        local CurID = DialogData[-4,table][Index,number]     
        local PrevID = DialogData[-4,table][Ind,number]  
        
        if (Ind <= 3) {
            local OldCol = (GifMode == (Ind-1) ? vec4(0,168,168,255) : vec4(168,168,168,255))
            EGP:egpColor(PrevID, OldCol)
        } elseif (Ind >= 7) {
            EGP:egpColor(PrevID, vec4(168,168,168,255))
        }
        
        if (Index <= 3 | Index >= 7) { EGP:egpColor(CurID, vec4(0,168,0,255)) }
        
        if (Index >= 4 & Index <= 6) {
            local TxtTbl = DialogData[-4,table][Index,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string]+"_",EGP:egpPos(TxtTbl[1,number]))
        }
        if (Ind >= 4 & Ind <= 6) {
            local TxtTbl = DialogData[-4,table][Ind,table]
            EGP:egpText(TxtTbl[1,number],TxtTbl[2,string],EGP:egpPos(TxtTbl[1,number]))
        }
        return 1        
    } elseif (Key=="enter") {     
        
        if (Index >= 1 & Index <= 3) {
            local PrevActiveModeID = DialogData[-4,table][GifMode+1,number]
            EGP:egpColor(PrevActiveModeID, vec4(168,168,168,255))   
            
            GifMode = Index - 1
            GpuPersistData["gifMode",number] = GifMode
            
            local NewActiveModeID = DialogData[-4,table][Index,number]
            EGP:egpColor(NewActiveModeID, vec4(0,168,0,255))
            return 1
        }
        
        if (Index == 8) {
            alxos_gif_key("f8",1)
            return 1
        }
        
        if (Index == 7) {
            local InpMaxSize = DialogData[-4,table][4,table][2,string]:toNumber()
            local InpKeepAR = DialogData[-4,table][5,table][2,string]
            local InpMinDelay = DialogData[-4,table][6,table][2,string]:toNumber()
            
            local MaxSize = gpuScreenSize(DS)
            
            # default
            if (InpMaxSize==0) { InpMaxSize = MaxSize }
            if (InpKeepAR=="") { InpKeepAR = "1" }
            if (InpMinDelay==0) { InpMinDelay = 2 }
            
            GpuPersistData["gifMaxSize",number] = clamp(InpMaxSize,1,MaxSize)
            GpuPersistData["gifNoKeepAR",number] = (InpKeepAR=="1" ? 0 : 1)
            GpuPersistData["gifMinDelay",number] = max(0.1,InpMinDelay)
            
            "alxos_explorer_closedialog"()
            alxos_img_postinit()
            return 1
        }
        
        return 0
        
    } else {
        if (Index >= 4 & Index <= 6) {
            local ITbl = DialogData[-4,table]
            local D_I = ITbl[Index,table][1,number]
            local D_V = ITbl[Index,table][2,string]
            
            if (Key=="backspace") {
                D_V = D_V:sub(0,D_V:length()-1)
                EGP:egpSetText(D_I,D_V+"_")
                ITbl[Index,table][2,string] = D_V
            } elseif (Key=="delete") {
                EGP:egpSetText(D_I,"_")
                ITbl[Index,table][2,string] = ""
            } elseif (Pressed==2) {
                local KMem = toByte(Key)
                local Char = toChar((KMem>=128 ? KMem-80 : KMem))
                local RE = ITbl[Index,table][3,string]
                local MC = ITbl[Index,table][4,number]
                
                if (Char:findRE(RE) & (KMem<=122 | KMem>=128 & KMem<=143)) {
                    if (D_V:length() < MC) {
                        D_V = (D_V+Char)
                        EGP:egpSetText(D_I,D_V+"_")
                        ITbl[Index,table][2,string] = D_V
                    }
                }
            }
        }
    }
    return 0
}
