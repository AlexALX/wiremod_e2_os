#include "alx_pc/gpu/alxos/_main_"

function array alxos_findExecutor() {
    
    local PCI_Devices = Devices["pci_bus",table]
            
    foreach(K:number,V:array=PCI_Devices) {
        if (V[1,string]=="rmtexec") {
            return V    
        }
    }
    
    return array()
}

function alxos_executor_error(ErrorMsg:string, Width) {
    if (GpuData["lastProg", string]=="alxos_console") {
        "alxos_console_handlers"()
        "alxos_console_err"(ErrorMsg)
    } else {        
        "alxos_explorer_error"(ErrorMsg,Width)
    }
}

function alxos_executor_init(Args:table) {
    #ifndef entity:remoteSetCode(string)
        if (1) {
            local Error = "Remote Upload E2 Extension is disabled,\n"+
                "cannot run code.\n\n"+
                "You can enable it in server console:\n"+
                "wire_expression2_extension_enable remoteupload"
            
            alxos_executor_error(Error,500)
            return
        }
    #endif
    
    local Executor = alxos_findExecutor()
    if (!Executor:count()) {
        local Error = "Remote Executor not found, please insert PCI card and reboot."
        
        alxos_executor_error(Error,500)
        return
    }    
    
    GpuData["ExecInit",number] = 0
    KeyHandler = "alxos_executor_key"
    DsHandler = "alxos_executor_ds"
    ExplorerCurTask = "alxos_prog_executor"
    
    local FInfo = Args[1,table]
    local ExplorerFS = Args[2,table]
    
    local RealExt = FInfo["real_ext",string]!="" ? FInfo["real_ext",string] : FInfo["ext",string]
    
    ExplorerFile = FInfo["name",string]+"."+RealExt
    
    GpuData[1,table] = ExplorerFS
    GpuData[2,table] = FInfo
    GpuData[5,number] = 0

    if (FInfo["content",string]) {
        "alxos_executor_run"(FInfo["content",string])
        return
    }
    
    dsSendDirect("fs_readfile",ExplorerFS["hdd_io",wirelink]:entity(),table(entity(),ExplorerFS["fs_str_id",string],table(
        ExplorerFS,FInfo,0,FInfo["size",number]
    )))
    
    GpuData["lastProg", string] = CurrentProg
    
    if (CurrentProg=="alxos_console") {
        "alxos_hide_cursor"()                
        "alxos_console_msg"("Opening file... Press ENTER or F8 to cancel")
        "alxos_console_msg"("0%")
        "alxos_console_move"()
    } else {
        local Parent = "alxos_explorer_dialog"("Opening file",400,80,"progress","alxos_executor_clear")[number]
    
        DialogData[-10,table]:pushNumber(LAST_I)
        EGP:egpText(LAST_I,"Please wait, this can take few minutes...",alxos_dlgPos(0,-22))
        EGP:egpParent(LAST_I,Parent)
        EGP:egpAlign(LAST_I,1)
        EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    }
    
    DialogData[1,number] = 1
    DialogData[3,number] = 1
    DialogData[4,number] = 1
    runCmd("alxos_executor_progress",1)
}

function alxos_executor_manifest() {
    local AppManifest = GpuData["AppManifest",table]
    
    if (GpuData["lastProg", string]=="alxos_console") {
        if (AppManifest:count()) {
            "alxos_console_msg"(AppManifest["name",string])
            "alxos_console_msg"("Author: "+AppManifest["author",string])
            "alxos_console_msg"("Version: "+AppManifest["version",string])
        } else {
            "alxos_console_msg"("Warning: no AppManifest.")
        }
        
        "alxos_console_msg"("Executing in a second... press ENTER to abort")
        "alxos_console_move"()
    } else {
        local Height = (AppManifest:count() ? 125 : 85)
        
        local Info = table()
        Info["OK",array] = array("Abort",50,0)
        
        local Parent = "alxos_explorer_dialog"("App Manifest",400,Height,"ok","alxos_executor_clear",Info)[number]
    
        local LAST_Y = 20
        
        if (AppManifest:count()) {
    
            DialogData[-10,table]:pushNumber(LAST_I)
            EGP:egpText(LAST_I,AppManifest["name",string],vec2(20,LAST_Y))
            EGP:egpParent(LAST_I,Parent)
            EGP:egpColor(LAST_I,0,0,0,255) LAST_I++ LAST_Y+=20
            
            DialogData[-10,table]:pushNumber(LAST_I)
            EGP:egpText(LAST_I,"Author: "+AppManifest["author",string],vec2(20,LAST_Y))
            EGP:egpParent(LAST_I,Parent)
            EGP:egpColor(LAST_I,0,0,0,255) LAST_I++ LAST_Y+=20
            
            DialogData[-10,table]:pushNumber(LAST_I)
            EGP:egpText(LAST_I,"Version: "+AppManifest["version",string],vec2(20,LAST_Y))
            EGP:egpParent(LAST_I,Parent)
            EGP:egpColor(LAST_I,0,0,0,255) LAST_I++ LAST_Y+=20
        
        } else {
            DialogData[-10,table]:pushNumber(LAST_I)
            EGP:egpText(LAST_I,"Warning: no AppManifest.",vec2(20,LAST_Y))
            EGP:egpParent(LAST_I,Parent)
            EGP:egpColor(LAST_I,0,0,0,255) LAST_I++ LAST_Y+=20
        }
        
        DialogData[-10,table]:pushNumber(LAST_I)
        EGP:egpText(LAST_I,"Executing in a second...",vec2(20,LAST_Y))
        EGP:egpParent(LAST_I,Parent)
        EGP:egpColor(LAST_I,0,0,0,255) LAST_I++ LAST_Y+=20
    }

    stopCmd()
    runCmd("alxos_executor_done",1200,1)
}

function alxos_executor_done() {
    if (GpuData["lastProg", string]=="alxos_console") {
        "alxos_hide_cursor"()
        "alxos_console_redraw_last"("Code is executing... Press F8 to cancel",1)
        "alxos_console_move"()
    } else {
        local Parent = "alxos_explorer_dialog"("Code Executed",400,60,"","")[number]
        
        DialogData[-10,table]:pushNumber(LAST_I)
        EGP:egpText(LAST_I,"Code is executing...",vec2(200,20))
        EGP:egpParent(LAST_I,Parent)
        EGP:egpAlign(LAST_I,1)
        EGP:egpColor(LAST_I,0,0,0,255) LAST_I++
    }
    
    local Executor = alxos_findExecutor()
    dsSendDirect("executeInitDone",Executor[3,entity],1)
}

function alxos_executor_progress() {
    local ExplorerFS = GpuData[1,table]
    
    local FS = ExplorerFS["hdd_io",wirelink]
    local Step = DialogData[4,number]

    local CurTask = FS["CurTask",table]
    local I = CurTask[2,number]
    local Max = CurTask[1,number]
    local Last = DialogData[3,number]
    
    if (!Step) { return }

    if (Max>0) {
        local PercID = DialogData[-4,number]
        local BoxID = DialogData[-5,number]
        local BoxW = DialogData[-6,number]
        local CalcP = round(I/Max*100)
        BoxW = ceil(BoxW/100*CalcP)
        if (BoxW<1) { BoxW = 1 }     
        
        if (GpuData["lastProg",string]=="alxos_console") {            
            "alxos_console_redraw_last"(CalcP+"%")
        } else {
            EGP:egpSize(BoxID,vec2(BoxW,20))
            EGP:egpSetText(PercID,CalcP+"%")
        }
    }
    
    runCmd("alxos_executor_progress",1) 
}

function number alxos_executor_clear() {
    if (!checkPerf(7000)) {
        runCmd("alxos_executor_clear",1)
        return 0
    }
    stoptimer("cmd")
    
    local ExplorerFS = GpuData[1,table]
    dsSendDirect("clear",ExplorerFS["hdd_io",wirelink]:entity(),1)
    
    local Executor = alxos_findExecutor()
    dsSendDirect("SHUTDOWN",Executor[3,entity],0)

    EGPEvent = function() {}
    DSEvent = function() {}
    CSEvent = function() {}
    
    if (GpuData["lastProg", string]=="alxos_console") {
        local CmdPath = "alxos_console_getpath"()[table]
        local CmdDir = "alxos_console_getdir"()[table]
        "alxos_execute"("alxos_console",table(table(),ExplorerFS,AllFs,CmdPath,CmdDir))
    } else {
        "alxos_explorer_handlers"()
        "alxos_explorer_listcurdir"()
    }
    return 0
}

function alxos_executor_end() {
    stoptimer("cmd")
    local ExplorerFS = GpuData[1,table]
    dsSendDirect("clear",ExplorerFS["hdd_io",wirelink]:entity(),1)
    
    EGPEvent = function() {}
    DSEvent = function() {}
    CSEvent = function() {}

    local Executor = alxos_findExecutor()
    dsSendDirect("SHUTDOWN",Executor[3,entity],0)
}

function number alxos_executor_close() {
    DSEvent = function() {}
    CSEvent = function() {}
    
    if (GpuData["lastProg", string]=="alxos_console") {
        local ExplorerFS = GpuData[1,table]
        local CmdPath = "alxos_console_getpath"()[table]
        local CmdDir = "alxos_console_getdir"()[table]
        "alxos_execute"("alxos_console",table(table(),ExplorerFS,AllFs,CmdPath,CmdDir))
    } else {
        "alxos_explorer_handlers"()
        "alxos_explorer_listcurdir"()
    }
    return 0
}

function alxos_executor_run(Code:string) {
    local Executor = alxos_findExecutor()
    
    local ExplorerFS = GpuData[1,table]
    local FInfo = GpuData[2,table]:copy()
    
    FInfo["content",string] = Code
    
    EGPEvent = function() {
        local Executor = alxos_findExecutor()
        dsSendDirect("ScreenConnect",Executor[3,entity],"EGP")  
    }
    
    DSEvent = function() {
        local Executor = alxos_findExecutor()
        dsSendDirect("ScreenConnect",Executor[3,entity],"DS")  
    }
    
    CSEvent = function() {
        local Executor = alxos_findExecutor()
        dsSendDirect("ScreenConnect",Executor[3,entity],"CS")  
    }
    
    local Info = table()
    Info["Devices",table] = Devices
    Info["TickQuota",number] = TickQuota
    Info["ExplorerFS",table] = ExplorerFS
    Info["AllFs",table] = AllFs
    Info["FInfo",table] = FInfo
    Info["LastProgram",string] = GpuData["lastProg",string]
    
    local PercID = DialogData[-4,number]
    local BoxID = DialogData[-5,number]
    local BoxW = DialogData[-6,number]
    local CalcP = 100
        
    if (GpuData["lastProg",string]=="alxos_console") {            
        "alxos_console_redraw_last"(CalcP+"%",1)
    } else {
        EGP:egpSize(BoxID,vec2(BoxW,20))
        EGP:egpSetText(PercID,CalcP+"%")
    }
    
    stopCmd()
    dsSendDirect("executeCode",Executor[3,entity],Info)
}

function alxos_executor_ds(Name:string) {      
    if (Name=="fs_readfile") {
        alxos_executor_run(dsGetString())
    } elseif (Name=="clear") {
        if (!checkPerf(7000)) {
            runCmd("alxos_png_clear",1)
            return
        }
        alxos_executor_clear()
    } elseif (Name=="execError") {
        local ErrType = dsGetNumber()
        local ErrorMsg = "Remote Executor timeout (chip crashed?)"
        
        if (ErrType==2) {
            ErrorMsg = "Remote Executor ExecutorID is invalid!\nPlease check wire inputs on pci card e2 chips."    
        }
        
        alxos_executor_error(ErrorMsg,420)
    } elseif (Name=="initExecutor") {
        local Info = dsGetTable()
        
        local AppManifest = Info[2,table]
        
        local ErrorMsg = ""
        
        if (AppManifest:count()) {
            local ScreenModes = AppManifest["screen_modes", table]
            
            local ScreenMode = ScreenModes["mode",string]
            
            if (ScreenMode=="any") {
                local Supported = 0
                local SupportedArr = array()
                
                if (ScreenModes["egp",number]==1) {
                    if (EGP:entity():isValid()) {
                        Supported = 1
                    } else {
                        SupportedArr:pushString("EGP")
                    }
                } 
                
                if (ScreenModes["console",number]==1) {
                    if (CS:entity():isValid()) {
                        Supported = 1
                    } else {
                        SupportedArr:pushString("Console")
                    }
                } 
                
                if (ScreenModes["digital",number]==1 & DS:entity():isValid()) {
                    if (DS:entity():isValid()) {
                        Supported = 1
                    } else {
                        SupportedArr:pushString("Digital")
                    }
                }
                
                if (!Supported) {
                    if (SupportedArr:count()==1) {
                        ErrorMsg = "This program doesn't support\nexecuting without "+SupportedArr[1,string]+"\nScreen connected."
                    } else {
                        ErrorMsg = "This program doesn't support\nexecuting without any of next\n"
                            +"screens connected:\n"+SupportedArr:concat(", ")
                    }
                }
                
            } else {
                if (ScreenModes["egp",number]==1 & !EGP:entity():isValid()) {
                    ErrorMsg = "This program doesn't support\nexecuting without EGP\nScreen connected."
                } elseif (ScreenModes["console",number]==1 & !CS:entity():isValid()) {
                    ErrorMsg = "This program doesn't support\nexecuting without Console\nScreen connected."
                } elseif (ScreenModes["digital",number]==1 & !DS:entity():isValid()) {
                    ErrorMsg = "This program doesn't support\nexecuting without Digital\nScreen connected."
                }
            }
        } else {
            # if no manifest provided - assume its legacy EGP only
            if (!EGP:entity():isValid()) {
                ErrorMsg = "This program doesn't support\nexecuting without EGP\nScreen connected."   
            }
        }
    
        if (ErrorMsg) {
            alxos_executor_error(ErrorMsg,420)
            alxos_executor_end()
            return
        }

        GpuData["ExecutorWirelink",wirelink] = Info[1,wirelink]
        GpuData["ExecInit",number] = 1
        GpuData["AppManifest",table] = AppManifest
        
        alxos_executor_manifest()
    } elseif (Name=="exitProg") {    
        alxos_executor_end()
        alxos_executor_close()
    } elseif (Name=="alxos_runprog") {
        alxos_executor_end()        
        "alxos_explorer_ds"(Name)
    }
}

function number alxos_executor_key(Key:string,Pressed) {
    if (!Pressed) { return 0 }
    
    local Index = DialogData[-5,number]
    local Type = DialogData[-2,string]
    
    if (Key=="f8") {  
        alxos_executor_clear()
        return 1
    } elseif (Key=="enter" & !GpuData["ExecInit",number]) {
        alxos_executor_key("f8",1)
        return 1
    }

    if (GpuData["ExecInit",number]) {
        local Executor = alxos_findExecutor()
        dsSendDirect("executorKey", Executor[3,entity], table(Key,Pressed))
    }

    return 0
}
