#include "alx_pc/gpu/_draw_"

function init() {
    initClear()
    initDrawFunc()
    
    stopAllTimers()
    KeyHandler = ""
    DsHandler = ""
    ERR_C = 0
    ERR_N = 0
    initDrawLine(Devices["names",table]["gpu",string]+" - GPU Initialization...")
    runCmd("initpost",300)
}

registerTimers("boot_timer","hdd_timer")

function hdd_timer() {
    hddload(0)      
}

function initpost() {
    initClear()
    runCmd("initbiospre",100)
}

function initbiospre() {
    initDrawLine("Simple BIOS v1.2")
    runCmd("initbios",2000)
}

function initbios() {
    BIOS_INIT = 1
    
    local Status = "OK"
    if (softQuota()<5000 | TickQuota<19000) {
        Status = "Error"
        ERR_C = 1
    } else { Status = "OK" }
    local Quota = (softQuota()>TickQuota ? TickQuota : softQuota())
    
    initDrawLine("CPU "+Devices["names",table]["cpu",string]+" ("+round(Quota/1000,2)+" Mhz) - "+Status)
    if (ERR_C) { 
        initDrawLine("ERROR: CPU is too slow.")
        if (TickQuota<19000) {
            initDrawLine("Minimum e2 tickquota is 20000 on server.")
            initDrawLine("Current value is approx "+TickQuota+" (hard or tick).")
        } else {
            initDrawLine("Minimum e2 softquota is 5000 on server.")
            initDrawLine("Current value is "+softQuota()+".")
        }
        err() 
    }
    
    if (!EGP:entity():isValid()) {
        if (CS:entity():isValid() | DS:entity():isValid()) {
            initDrawLine("Warning: EGP is not connected!")
        } else {
            ERR_C = 7
            err()
        }
    }

    if (egpMaxObjects()<300 & EGP:entity():isValid()) {
        Status = "Error"
        ERR_C = 2
    } else { Status = "OK" }
    
    initDrawLine("GPU "+Devices["names",table]["gpu",string]+" ("+ceil(egpMaxObjects()/128)*1024+" KB) - "+Status)

    if (ERR_C) { 
        initDrawLine("ERROR: GPU memory is too small.")
        initDrawLine("Minimum EGP Objects is 300 on server.")
        initDrawLine("Current value is "+egpMaxObjects()+".")
        
        err()
    }
    
    if (!Devices["spu",entity]:isValid()) {
        Status = "Not connected"
    } else { Status = "Connected" }
    #ifndef soundStop(number)
        if (Status=="Connected") {
            Status = "Malfunction"
        }
    #endif
    initDrawLine("Sound "+Devices["names",table]["spu",string]+" - "+Status)
    
    #ifndef soundStop(number)
    initDrawLine("Error: E2 Extension Sound is not enabled!")
    #endif
    
    if (!DS:entity():isValid()) {
        Status = "Not connected"
    } else { Status = "Connected" }
    
    local ScreenName = gpuScreenShortName(DS)
    if (ScreenName) {
        Status += " ("+ScreenName+")"
    }
    
    initDrawLine("Digital Screen - "+Status)
    
    if (!CS:entity():isValid()) {
        Status = "Not connected"
    } else { Status = "Connected" }
    
    initDrawLine("Console Screen - "+Status)
    
    if (Keyboard[1,entity]==noentity()) {
        ERR_C = 4
        
        initDrawLine("Keyboard - "+Status)
        initDrawLine("ERROR: No keyboard detected")
        
        err() 
    } else { 
        initDrawLine(Devices["names",table]["keyboard",string]+" - Connected")
    }
    
    if (toByte(NB)!=0) {
        ERR_C = 5
        initDrawLine("ERROR: Null byte is not correct!")
        initDrawLine("Please connect empty \"To Character\" gate")
        initDrawLine("\"Out\" output to \"NullByte\" input on BIOS,")
        initDrawLine("Or update your wiremod.")
        err() 
    }
    
    #Y+=Sz
    #EGP:egpText(I,"Connected HDD's: ",vec2(X,Y))
    #EGP:egpSize(I,Sz) I++ Y+=Sz
    
    hddload(1) 
    DsHandler = "boot_ds"
    dsSendDirect("gethdds",BIOS,entity())
}

function inithdd() {
    local Info = dsGetTable()
    HDDS = Info[1,table]
    FS_TYPES = Info[2,table]
    
    local X = BIOS_X
    local Sz = BIOS_SZ
    
    if (!HDDS:count()|!HDD_IO:count()) {
        initDrawLine("")
        initDrawLine("Warning: No hdd's found.")
    } else {
        
        local UsbDevices = table()
        
        local II = 0
        local MaxLines = InitMaxLines - InitLines
        
        for (H=1,Devices["hdd_io",table]:count()) {
            if (II>MaxLines) { break } # gui limitation
            local Inf = Devices["hdd_io",table][H,table]
            for(I=1,Inf[4,number]) {
                local Item = Inf[5,table][I,table]
            
                if (I==1) {
                    local Name = Inf[3,string]+" #"
                    if (Item["usb",entity]) {
                        UsbDevices:pushTable(table(Name,Item))
                        break
                    }
                    
                    initDrawLine(Name+H+": ")
                    if (II>MaxLines) { break } # gui limitation
                }
                if (Item:count()) {
                    if (Item["usb",entity]) {
                        local Name = Inf[3,string]+" #"
                        UsbDevices:pushTable(table(Name,Item))
                        continue
                    }
                    if (Item["type",number]==1) {
                        initDrawLine("HDD"+(I-1)+" - "+Item["name",string])
                    } else {
                        initDrawLine(
                            "HDD"+(I-1)+" - "+Item["name",string]+" - Size "
                            +(Item["size",number]*Item["sector_size",number])/1024+" KB"
                        )
                    }
                } else {
                    initDrawLine("HDD"+(I-1)+" - Not connected")
                }
                II++
                if (II>MaxLines) { break } # gui limitation
            }
        }
        
        if (II<=MaxLines) {
            for(U=1,UsbDevices:count()) {
                local Item = UsbDevices[U,table][2,table]
                local Name = UsbDevices[U,table][1,string]
                
                initDrawLine(Name+(U-1)+" - Size "+(Item["size",number]*Item["sector_size",number])/1024+" KB")
                II++
                if (II>MaxLines) { break } # gui limitation   
            }
            
        }
        
    }
    
    KeyHandler = "bios_key"
    
    if (InitDrawMode=="egp") {
        EGP:egpText(LAST_I,"Press DEL to enter SETUP.",vec2(X,512-Sz))
        EGP:egpSize(LAST_I,Sz) LAST_I++
    } elseif(InitDrawMode=="console") {
        #InitLines = InitMaxLines
        #initDrawLine("Press DEL to enter SETUP.")
    }
    
    #soundPlay(12,0.25,"synth/square_440.wav")
    ERR_C = -1
    err()
    
    timer("boot_timer",100)
    runCmd("boot",2000)
    
    if (DEBUG==-1 | DEBUG & Keyboard["Memory",number]==148) { "bios_key"("delete",1) }
}

function bios_key(Key:string,Pressed) {
    if (!Pressed) { return }
    if (Key=="delete") {
        if (InitDrawMode=="console") {
            return    
        }
        
        #include "alx_pc/gpu/bios"
        stoptimer("cmd")
        bios_init()   
    }
}

function noboot(Msg:string) {
    
    initDrawLine(Msg)
    
    initDrawLine("")
    #initDrawLine("Press ENTER to run Setup program.")
    initDrawLine("Press F1 to reboot.")
    
    KeyHandler = "bootmenu_key"   
}

function boot_ds(Name:string) {
    if(Name=="gethdds") {
        DsHandler = ""
        inithdd()
    } elseif(Name=="readfstbl") {
        DsHandler = ""
        "bootmenu"()
    } elseif(Name=="fs_readfs") {
        "bootmenu_post"(0)
    } elseif(Name=="bootmenu") {
        "bootmenu_post"(1)
    } elseif(Name=="read_disk") {
        local Ret = dsGetTable()
        if (Ret[1,string]=="OK") {
            "bootmenu_post_cd"(0)
        } else {
            "nobootCD"(Ret[1,string])
        }
    } elseif (Name=="boot_func") {
        #include "alx_pc/gpu/_boot_"
        local Info = dsGetTable()
        if (!initboot(Info)) {
            noboot("Function "+Info[1,string]+" is missing.")   
        }
    }
}

function nobootCD(Msg:string) {
    local Arr = Msg:explode(NL)
    for (I=1,Arr:count()) {
        initDrawLine(Arr[I,string])
    }
    
    initDrawLine("")
    initDrawLine("Press F1 to reboot.")
    
    local BootHDD = BOOT_HDD[1,table]

    dsSendDirect("stopdisk",BootHDD["ent",entity],0)

    KeyHandler = "bootmenu_key"   
}

function table findboot() {
    local Ent = Devices[BOOT_HDD[2,string]+"_bus",table][BOOT_HDD[3,number],array][3,entity]
    if (Ent) {
        if (BOOT_HDD[4,string]!="") {
            for(I=1,Devices["usb",table]:count()) {
                local UsbCont = Devices["usb",table][I,table]
                if (UsbCont[1,entity]==Ent) {
                    if (UsbCont[4,table][BOOT_HDD[4,string],table]:count()) {
                        Ent = UsbCont[4,table][BOOT_HDD[4,string],table][3,wirelink]:entity()
                        break
                    }
                }    
            }
        }
        local HDD_IOS = Devices["hdd_io",table]
        for(I=1,HDD_IOS:count()) {
            if (HDD_IOS[I,table][1,wirelink]:entity()==Ent) {
                return HDD_IOS[I,table][5,table][BOOT_HDD[1,number],table] 
            }    
        }
    }
    return table()
}

function boot() {
    initClear()
    
    local BootHDD = findboot()
    BOOT_HDD[5,table] = BootHDD
    
    if (!BootHDD:count()|!HDD_IO:count()) {
        initDrawLine("No bootable device")
        initDrawLine("Press any key to reboot.")
        
        if (!EGP:entity():isValid()) {
            initDrawLine("")    
            initDrawLine("Warning:")    
            initDrawLine("Changing bootable device")    
            initDrawLine("Supported only on EGP")    
            initDrawLine("Inside BIOS")
        }

        KeyHandler = "reboot"
    } else {
        KeyHandler = ""
    
        if (BootHDD["type",number]==1) {
            initDrawLine(BootHDD["name",string]+" initialization,")
            initDrawLine("Please wait...")
            
            DsHandler = "boot_ds"
            dsSendDirect("read_disk",BootHDD["hdd_io",wirelink]:entity(),table("init",entity(),BootHDD))     
        } else {
            initDrawLine("Loading boot loader...")

            DsHandler = "boot_ds"
            dsSendDirect("readfstbl",BootHDD["hdd_io",wirelink]:entity(),table(entity(),BootHDD))
        }
    }
}

function reboot(Key:string,Pressed) {
    if (!Pressed) { return }
    KeyHandler = ""
    reboot()
}

function bootmenu() {
    local BootHDD = BOOT_HDD[5,table]
    local FsTbl = dsGetTable()

    if (FsTbl:count()) {
        for(I=0,FsTbl:count()-1) {
            if (FsTbl[I,table]["active",number]==1) {
                DsHandler = "boot_ds"
                dsSendDirect("fs_readfs",BootHDD["hdd_io",wirelink]:entity(),table(entity(),"wm1",table(BootHDD,FsTbl[I,table])))
                return
            }   
        }
    }        
    
    noboot("No Bootable Volume")
    
    #if (DEBUG) { runCmd("bootpost",1) }
        
}

function bootmenu_post_cd(Step:number) {
    if (Step==0) {
        local CurFs = dsGetTable()[2,table]
        
        if (CurFs["boot_func",string]!="") {
            dsSendDirect("bootmenu",E2_CPU,CurFs)   
        } else {
            nobootCD("This disc is not bootable.")
        }
    } elseif (Step==1) {
        local Type = dsGetType()
        if (Type=="string") {
            nobootCD(dsGetString())
        } else {
            #include "alx_pc/gpu/_boot_"
            local Info = dsGetTable()
            if (!initboot(Info)) {
                noboot("Function "+Info[1,string]+" is missing.")   
            }
        }
    }
}

function bootmenu_post(Step:number) {
    if (Step==0) {
        local CurFs = dsGetTable()
        
        if (CurFs["boot_func",string]!="") {
            dsSendDirect("bootmenu",E2_CPU,CurFs)   
        } else {
            noboot("Boot loader not found.")
            DEBUG = 0
        }
    } elseif (Step==1) {
        local Type = dsGetType()
        if (Type=="string") {
            noboot(dsGetString())
        } else {
            #include "alx_pc/gpu/_boot_"
            local Info = dsGetTable()
            if (!initboot(Info)) {
                noboot("Function "+Info[1,string]+" is missing.")   
            }
        }
    }
}
