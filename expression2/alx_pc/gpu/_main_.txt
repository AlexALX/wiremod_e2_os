function number in_array(Arr:array, Value) {
    for (I=1,Arr:count()) {
        if (Arr[I,number]==Value) { return 1 }
    }       
    return 0
}

function err() {
    dsSendDirect("err",BIOS,ERR_C)
    if (ERR_C!=-1) { exit() }
}

function egpClear() {
    EGP:egpClear()
    EGP:egpDrawTopLeft(1)
    LAST_I = 1 
    LAST_Y = 0 
    GpuData = table()
    if (EgpCleanHandler!="") {
        EgpCleanHandler()
    }
}

function dsClear() {
    DS[1048574] = 0
    DS[1048575] = 1
}

function csClear() {
    CS[2041] = 1
    CS[2046] = 0
}

# write to console screen
function csScreenWrite(X:number, Y:number, Str:string, ColorParam:number) {       
    local StartAddr = Y * 60 + X * 2
    local StrArr = Str:explode("")
    
    local MaxY = Y * 60 + 60
    
    for(I = 1, StrArr:count()) {        
        local CharCode = toByte(StrArr[I,string])
        
        CS[StartAddr] = CharCode
        CS[StartAddr + 1] = ColorParam
        
        StartAddr += 2
        
        if (StartAddr >= MaxY) { break }
    }
}

function csScreenMove(LinesToShift:number) {
    CS[2038] = LinesToShift
}

function reboot() {
    dsSendDirect("REBOOT",BIOS,0)   
}

function hddload(Num) {
    dsSendDirect("HDDInd",BIOS,Num)    
}

function number gpuScreenSize(DS:wirelink) {
    if (DS:entity():type()=="gmod_wire_digitalscreen") {
        return 512
    }

    local ScreenName = DS:entity():getName()
    if (ScreenName=="Quad Digital Screen") {
        return 1024
    } elseif (ScreenName=="ZGPU Screen Renderer") {
        return 512
    }
    
    return 0
}

function string gpuScreenName(DS:wirelink) {
    local ScreenName = DS:entity():getName()
    
    if (ScreenName=="Quad Digital Screen" | ScreenName=="ZGPU Screen Renderer") {
        return ScreenName        
    }
    return ""
}

function string gpuScreenShortName(DS:wirelink) {
    local ScreenName = DS:entity():getName()
    
    if (ScreenName=="Quad Digital Screen") {
        return "Quad"        
    } elseif (ScreenName=="ZGPU Screen Renderer") {
        return "ZGPU"    
    }
    return ""
}
