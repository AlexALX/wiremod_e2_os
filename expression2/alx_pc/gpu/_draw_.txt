@persist InitDrawLine:function InitClearLine:function InitDrawChar:function
@persist InitMaxLines:number InitDrawMode:string InitLines:number InitMaxChars:number

#include "alx_pc/gpu/_main_"
#include "alx_pc/standalone/ds_gfx/_init_"
#include "alx_pc/standalone/ds_gfx/font_5x7"
#include "alx_pc/gpu/includes/consolescreen"

function initDrawFunc() {
    InitDrawLine = function(Msg:string) {}
    InitClearLine = function(Line:number) {}
    InitDrawChar = function(X:number,Line:number,Char:string) {}
    if (EGP:entity():isValid()) {
        InitDrawLine = function(Msg:string) {
            local LocalY = BIOS_SZ*InitLines
            EGP:egpText(LAST_I,Msg,vec2(BIOS_X,LocalY))
            EGP:egpSize(LAST_I,BIOS_SZ) LAST_I++
            #LAST_Y += BIOS_SZ
            InitLines++
        }
        InitMaxChars = 50
        InitMaxLines = 20
        InitDrawMode = "egp"
    } elseif (CS:entity():isValid()) {
        InitDrawLine = function(Msg:string) {
            csScreenWrite(0,InitLines,Msg,999)
            InitLines++
        }
        InitClearLine = function(Line:number) {
            csClearLine(Line+1)
        }
        InitDrawChar = function(X:number,Line:number,Char:string) {
            csScreenWrite(X,Line,Char,999)    
        }
        InitMaxChars = CsMaxChars
        InitMaxLines = CsMaxLines-1
        InitDrawMode = "console"
    } elseif (DS:entity():isValid()) {    
        dsInit(DSContainer, 256, 256)
        dsSetFont(dsFont5x7())
        InitDrawLine = function(Msg:string) {
            local LAST_Y = DsFontLineHeight*InitLines
            local Lines = Msg:explode("\n"):count()
            dsDrawTextStart(0,LAST_Y,Msg)
            InitLines += Lines
        }
        InitClearLine = function(Line:number) {
            dsTextClearLine(Line*DsFontLineHeight)
        }
        InitDrawChar = function(X:number,Line:number,Char:string) {
            if (Char==" ") {
                dsClearChar(DS,X*DsFontCharWidth,Line*DsFontLineHeight)
            } else {            
                dsDrawChar(DS,X*DsFontCharWidth,Line*DsFontLineHeight,toByte(Char))
            }
        }
        InitMaxChars = 50
        InitMaxLines = ceil(256 / DsFontLineHeight)-1
        InitDrawMode = "ds"
    }
}

function initDrawLine(Msg:string) {
    InitDrawLine(Msg)    
}

function initClearLine(Line:number) {
    InitClearLine(Line)
}

function initDrawChar(X:number,Line:number,Char:string) {
    InitDrawChar(X,Line,Char)    
}

function initClear() {
    egpClear()
    csClear()
    dsClear()
    dsStopQueues()
    
    LAST_I = 1
    LAST_Y = 0
    InitLines = 0
}
