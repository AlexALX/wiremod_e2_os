@name Quad Digital Screen
@inputs LinkTopLeft:wirelink LinkTopRight:wirelink LinkBottomLeft:wirelink LinkBottomRight:wirelink DsShift:number
@outputs VirtualScreen:wirelink DsImageDrew HitPerf DsShifted:number
@persist ScrW ScrH VirtW VirtH DsPixelAddr
@persist MaxPixelsPerTick PixelsThisTick
@trigger none 
@strict

#[
    ALX Wiremod E2 PC
    Created by AlexALX (c) 2026
]#

# When DsShift set then all commands must add this shift to its address!
# This fixes edge 1024*1024 case
# If you want to make it work as normal DS - set DsShift = 0, can work up to 1023*1023
# Also make sure your script break while(perf()) by DS["HitPerf",number]==1
function dsShift() {
    DsShifted = DsShift
    if (DsShift) {
        DsPixelAddr = 1048576
    } else {
        DsPixelAddr = 1048500
    }
}

function dsHWClear() {  
    LinkTopLeft[1048574, number] = 0
    LinkTopRight[1048574, number] = 0
    LinkBottomLeft[1048574, number] = 0
    LinkBottomRight[1048574, number] = 0   
    DsImageDrew = 0
    dsShift()
}

if (first() | duped()) {    
    ScrW = 512
    ScrH = 512
    
    VirtW = ScrW * 2
    VirtH = ScrH * 2
    
    # throttle to fix fps
    MaxPixelsPerTick = 512
        
    VirtualScreen = wirelink()
    
    dsHWClear()
    
    Screen = VirtualScreen
    #include "alx_pc/shared/wirelinkcontroller"
}

if (dupefinished()) {
    dsShift()    
}

event tick() {
    HitPerf = 0
    PixelsThisTick = 0
}

event writeCell(Address:number, Value:number) {    
    if (Address < DsPixelAddr) {
        PixelsThisTick++

        local GlobalX = Address % VirtW
        local GlobalY = (Address - GlobalX) / VirtW
      
        if (GlobalY < ScrH) {
            local LocalRowOffset = GlobalY * ScrW
            if (GlobalX < ScrW) {
                LinkTopLeft[GlobalX + LocalRowOffset, number] = Value
            } else {
                LinkTopRight[(GlobalX - ScrW) + LocalRowOffset, number] = Value
            }
        } else {
            local LocalRowOffset = (GlobalY - ScrH) * ScrW
            if (GlobalX < ScrW) {
                LinkBottomLeft[GlobalX + LocalRowOffset, number] = Value
            } else {
                LinkBottomRight[(GlobalX - ScrW) + LocalRowOffset, number] = Value
            }
        }
    } else {
        Address = Address - DsShift
        
        # special custom image drew flag
        if (Address == 1048500) {
            DsImageDrew = Value
        } elseif (Address == 1048574) {
            dsHWClear()
        } elseif (Address == 1048573) {
            VirtW = Value
            ScrW = floor(Value / 2)
            
            LinkTopLeft[1048573, number] = ScrW
            LinkTopRight[1048573, number] = ScrW
            LinkBottomLeft[1048573, number] = ScrW
            LinkBottomRight[1048573, number] = ScrW
        } elseif (Address == 1048572) {
            VirtH = Value
            ScrH = floor(Value / 2)
            
            LinkTopLeft[1048572, number] = ScrH
            LinkTopRight[1048572, number] = ScrH
            LinkBottomLeft[1048572, number] = ScrH
            LinkBottomRight[1048572, number] = ScrH
        } elseif (Address == 1048570) {
            if (Value < ScrH) {
                LinkTopLeft[1048570, number] = Value
                LinkTopRight[1048570, number] = Value
            } else {
                local LocalY = Value - ScrH
                LinkBottomLeft[1048570, number] = LocalY
                LinkBottomRight[1048570, number] = LocalY
            }
        } elseif (Address == 1048571) {
            if (Value < ScrW) {
                LinkTopLeft[1048571, number] = Value
                LinkBottomLeft[1048571, number] = Value
            } else {
                local LocalX = Value - ScrW
                LinkTopRight[1048571, number] = LocalX
                LinkBottomRight[1048571, number] = LocalX
            }
        } else {  
            LinkTopLeft[Address, number] = Value
            LinkTopRight[Address, number] = Value
            LinkBottomLeft[Address, number] = Value
            LinkBottomRight[Address, number] = Value
        }
    }
    
    if (PixelsThisTick >= MaxPixelsPerTick | !perf(90)) { HitPerf = 1 }
}

event input(InputName:string) {
    if (InputName=="InG") {
        dsHWClear()    
    } elseif (InputName=="DsShift") {
        dsShift() 
    }
}

if (dsClk("wirelink") & InG==dsGetEntity():id()) {
    wirelinkInit()  
}
